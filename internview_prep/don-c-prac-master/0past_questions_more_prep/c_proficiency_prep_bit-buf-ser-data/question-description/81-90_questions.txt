// 81. 패킷 레이트 리미터 (토큰 버킷/누출 버킷)
/*
설명:
임베디드 네트워크, 무선 통신, 센서 게이트웨이 등에서 과도한 패킷 송출을 제한하기 위해 레이트 리미터가 필요합니다.
정수 기반 시간 스케일, 오버플로 방지, 버스트 허용 등 robust한 구현이 요구됩니다.

- 토큰 버킷/누출 버킷 알고리즘 중 택1
- 정수 기반 시간/토큰 처리(부동소수점 금지)
- 버스트 허용량, 최대 속도 파라미터
- 오버플로/언더플로 방지
- API: init, allow, update
*/

#define RATE_LIMITER_H
#include <stdint.h>
typedef struct {
    uint32_t tokens;
    uint32_t max_tokens;
    uint32_t rate_per_tick;
    uint32_t last_tick;
} token_bucket_t;
void token_bucket_init(token_bucket_t* b, uint32_t rate_per_tick, uint32_t max_tokens, uint32_t now_tick);
int  token_bucket_allow(token_bucket_t* b, uint32_t now_tick); // 1=허용, 0=거부
#endif

/*

상태 저장, 타임아웃, 최대 시도 등 robust한 정책 구현이 요구됩니다.

*/

    uint8_t retries;
    uint8_t max_retries;
    uint32_t backoff_ticks;
    uint32_t next_try_tick;
} bus_backoff_t;
void bus_backoff_init(bus_backoff_t* b, uint8_t max_retries, uint32_t base_backoff);
int  bus_backoff_on_fail(bus_backoff_t* b, uint32_t now_tick);

#endif
