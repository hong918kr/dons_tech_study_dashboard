// 77. 바이트 오더 변환 매크로
/*
설명:
임베디드에서 네트워크/저장소/MCU 간 바이트 오더 변환이 필요할 때, bswap16/32/64 매크로가 널리 쓰입니다.
내장함수/폴백, 정적 인라인, 이식성 등이 중요합니다.
예시 상황:
- 네트워크 패킷 직렬화/역직렬화
- 플래시/EEPROM 데이터 변환
요구사항:
- bswap16/32/64 매크로/정적 인라인
- 내장함수(__builtin_bswap 등) 사용/폴백
- 이식성/성능 주석
*/
#ifndef BSWAP_H
#define BSWAP_H
#include <stdint.h>
static inline uint16_t bswap16(uint16_t v) {
#if defined(__GNUC__)
    return __builtin_bswap16(v);
#else
    return (v << 8) | (v >> 8);
#endif
}
static inline uint32_t bswap32(uint32_t v) {
#if defined(__GNUC__)
    return __builtin_bswap32(v);
#else
    return ((v & 0xFF) << 24) | ((v & 0xFF00) << 8) | ((v >> 8) & 0xFF00) | ((v >> 24) & 0xFF);
#endif
}
static inline uint64_t bswap64(uint64_t v) {
#if defined(__GNUC__)
    return __builtin_bswap64(v);
#else
    return ((uint64_t)bswap32(v) << 32) | bswap32(v >> 32);
#endif
}
#endif
