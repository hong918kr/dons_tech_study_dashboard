// 51. 비트 회전(rotl/rotr)
/*
설명:
암호화, 체크섬, 해시, 센서 데이터 처리 등에서 비트 순환 시프트(rotate left/right)가 자주 필요합니다.
임베디드 환경에서는 시프트 폭 정규화, 인라인 최적화, 매크로/함수 선택 등 다양한 구현이 요구됩니다.
예시 상황:
- CRC/해시 함수 내부 비트 조작
- 센서 데이터 LFSR 변환
요구사항:
- 8/16/32/64비트 rotl/rotr 지원
- 시프트 폭 정규화(0~비트폭-1)
- 매크로/정적 인라인 함수 제공
- 경계값/성능 테스트
*/
#ifndef BITROT_H
#define BITROT_H
#include <stdint.h>
static inline uint8_t  rotl8(uint8_t v, unsigned n)  { return (v << (n & 7)) | (v >> ((8 - n) & 7)); }
static inline uint8_t  rotr8(uint8_t v, unsigned n)  { return (v >> (n & 7)) | (v << ((8 - n) & 7)); }
static inline uint16_t rotl16(uint16_t v, unsigned n){ return (v << (n & 15)) | (v >> ((16 - n) & 15)); }
static inline uint16_t rotr16(uint16_t v, unsigned n){ return (v >> (n & 15)) | (v << ((16 - n) & 15)); }
static inline uint32_t rotl32(uint32_t v, unsigned n){ return (v << (n & 31)) | (v >> ((32 - n) & 31)); }
static inline uint32_t rotr32(uint32_t v, unsigned n){ return (v >> (n & 31)) | (v << ((32 - n) & 31)); }
static inline uint64_t rotl64(uint64_t v, unsigned n){ return (v << (n & 63)) | (v >> ((64 - n) & 63)); }
static inline uint64_t rotr64(uint64_t v, unsigned n){ return (v >> (n & 63)) | (v << ((64 - n) & 63)); }
#endif
