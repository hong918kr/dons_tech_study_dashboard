// 41. 비트 온도 변환기(센서 포맷)
/*
설명:
저가형 온도 센서, ADC, I2C/SPI 센서 등에서 12비트 two’s complement(부호 있는) 값을 받아 섭씨 온도로 변환해야 하는 경우가 많습니다.
임베디드 환경에서는 비트 마스킹, 사인 확장, 스케일링(예: 0.0625°C/LSB) 등 저수준 처리가 필수입니다.
예시 상황:
- TMP102, MCP9808 등 I2C 온도 센서 데이터 해석
- 12비트 ADC로 온도 측정 후 변환
요구사항:
- 12비트 two’s complement → int32_t 사인 확장
- 스케일링(예: 0.0625°C/LSB, 고정소수점 지원)
- 입력 범위/오버플로 방지
- 변환 함수 API
*/
#ifndef TEMP12BIT_H
#define TEMP12BIT_H
#include <stdint.h>
int32_t temp12bit_to_celsius(int16_t raw12);
int32_t temp12bit_to_celsius_q16(int16_t raw12); // Q16.16 고정소수점
#endif

// 42. UART 프레이밍 파서
/*
설명:
임베디드 UART 통신에서 STX(0x02)/ETX(0x03)/ESC(0x1B) 기반 프레이밍이 자주 쓰입니다.
바이트 스터핑, 타임아웃, 부분 입력 등 robust한 파서가 필요합니다.
예시 상황:
- UART로 패킷 기반 프로토콜 구현
- 센서/모듈 간 신뢰성 있는 메시지 수신
요구사항:
- STX/ETX 프레임 경계 인식
- ESC 바이트 스터핑 처리(0x02, 0x03, 0x1B)
- 타임아웃/리셋 훅 지원
- 부분 입력 스트리밍(feed API)
- 최대 길이/오버런 방지
*/
#ifndef UART_FRAME_PARSER_H
#define UART_FRAME_PARSER_H
#include <stddef.h>
#include <stdint.h>
typedef void (*uart_frame_cb_t)(const uint8_t* frame, size_t len, void* user);
typedef struct uart_frame_parser uart_frame_parser_t;
uart_frame_parser_t* uart_frame_parser_create(uart_frame_cb_t cb, void* user, size_t max_len);
void uart_frame_parser_destroy(uart_frame_parser_t* p);
void uart_frame_parser_feed(uart_frame_parser_t* p, const uint8_t* data, size_t len, uint32_t now_tick);
void uart_frame_parser_reset(uart_frame_parser_t* p);
void uart_frame_parser_set_timeout(uart_frame_parser_t* p, uint32_t timeout_ticks);
#endif

/*

요구사항:

- DLC(데이터 길이) 및 데이터 필드 직렬화

int    can_frame_parse(const uint8_t* in, size_t in_len, can_frame_t* frame);
#endif
