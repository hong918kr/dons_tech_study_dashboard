42static long now_ms(void) {
43 struct timespec ts;
44 clock_gettime(CLOCK_MONOTONIC, &ts);
45 return (long)ts.tv_sec * 1000L + ts.tv_nsec / 1000000L;
46}

48static void sleep_ms(int ms) {
49 struct timespec ts;
50 ts.tv_sec = ms / 1000;
51 ts.tv_nsec = (long)(ms % 1000) * 1000000L;
52 nanosleep(&ts, NULL);
53}

56static bool wait_flag(_Atomic int *flag, int want, int timeout_ms) {
57 long deadline = now_ms() + timeout_ms;
