52// ===========================================================================
53// Q39. read_full / write_full — 부분 읽기·쓰기 루프 + EINTR 재시도
54// ---------------------------------------------------------------------------
55// POSIX 에서 read()/write() 는 "요청한 만큼"을 보장하지 않는다. 파이프·소켓·tty 는
56// 커널 버퍼 한도만큼만 처리하고 돌아온다(short read / short write). 시그널이 오면
57// -1/EINTR 로 조기 복귀한다. 이 둘을 처리하지 않으면 모뎀 응답이 잘리고, 클라우드로
58// 보낸 프레임이 반쪽만 나간다 — 게이트웨이 필드 버그 1순위.
59// read_full : n 바이트를 다 읽거나 EOF 를 만날 때까지 반복. 리턴 = 실제 읽은 수
60// (n 보다 작으면 EOF. 그 자체는 에러가 아니다.) 에러면 -1.
61// write_full: n 바이트를 전부 밀어넣는다. 리턴 = n, 에러면 -1.
62// ===========================================================================
63ssize_t read_full(int fd, void *buf, size_t n) {
64 if (!buf && n) { errno = EINVAL; return -1; }
65 unsigned char *p = (unsigned char *)buf;
66 size_t got = 0;
67 while (got < n) {
68 ssize_t k = read(fd, p + got, n - got);
69 if (k < 0) {
70 if (errno == EINTR) continue; // 시그널에 깨진 것뿐 — 재시도
71 return -1; // 진짜 에러
72 }
73 if (k == 0) break; // EOF: 짧게 리턴하되 에러 아님
74 got += (size_t)k;
75 }
76 return (ssize_t)got;
77}
78 
79ssize_t write_full(int fd, const void *buf, size_t n) {
80 if (!buf && n) { errno = EINVAL; return -1; }
81 const unsigned char *p = (const unsigned char *)buf;
82 size_t sent = 0;
83 while (sent < n) {
84 ssize_t k = write(fd, p + sent, n - sent);
85 if (k < 0) {
86 if (errno == EINTR) continue; // 재시도
87 return -1;
88 }
89 sent += (size_t)k; // 부분 쓰기 → 남은 만큼 다시
90 }
91 return (ssize_t)sent;
92}

94// ===========================================================================
95// Q40. 논블로킹 fd (fcntl O_NONBLOCK) + EAGAIN/EWOULDBLOCK 처리
96// ---------------------------------------------------------------------------
97// 이벤트 루프의 모든 fd 는 논블로킹이어야 한다. poll 이 "읽을 수 있다"고 해도
98// (스퓨리어스 wakeup, 다른 소비자가 먼저 읽음) 막상 read 가 블록될 수 있기 때문.
99// 핵심은 F_GETFL → OR → F_SETFL 의 read-modify-write: 기존 플래그를 덮어쓰면 안 된다.
100// EAGAIN 은 "지금은 데이터 없음"이지 에러가 아니다. EWOULDBLOCK 과 같은 값일 수도,
101// 아닐 수도 있으므로 두 개를 모두 검사하는 게 이식성 있는 관용구다.
102// ===========================================================================
103typedef enum {
104 IO_OK = 0, // out_n 바이트를 읽었다
105 IO_WOULDBLOCK = 1, // 지금은 데이터 없음 (EAGAIN/EWOULDBLOCK)
106 IO_EOF = 2, // 상대가 닫음 (read == 0)
107 IO_ERROR = -1 // 진짜 에러
108} io_res_t;
109 
110int set_nonblocking(int fd) {
111 int flags = fcntl(fd, F_GETFL, 0);
112 if (flags < 0) return -1;
113 if (flags & O_NONBLOCK) return 0; // 이미 설정 — 멱등
114 if (fcntl(fd, F_SETFL, flags | O_NONBLOCK) < 0) // '=' 아니고 '|=' 가 핵심
115 return -1;
116 return 0;
117}
118 
119io_res_t read_nb(int fd, void *buf, size_t n, size_t *out_n) {
120 if (out_n) *out_n = 0;
121 if (!buf && n) { errno = EINVAL; return IO_ERROR; }
122 for (;;) {
123 ssize_t k = read(fd, buf, n);
124 if (k > 0) { if (out_n) *out_n = (size_t)k; return IO_OK; }
125 if (k == 0) return IO_EOF;
126 if (errno == EINTR) continue; // 재시도
127 if (errno == EAGAIN || errno == EWOULDBLOCK)
128 return IO_WOULDBLOCK; // 정상 — 루프로 돌아가라
129 return IO_ERROR;
130 }
131}

134// Q41. poll() 기반 이벤트 루프 — 두 fd 멀티플렉싱 + 타임아웃
135// ---------------------------------------------------------------------------
136// 게이트웨이 데몬의 심장. 모뎀 채널(fd_a)과 클라우드 소켓(fd_b)을 동시에 감시하고,
137// 아무 일도 없으면 타임아웃으로 깨어나 하트비트/watchdog 을 친다.
138// 규칙 3가지:
139// ① poll 이 -1/EINTR 이면 에러가 아니라 재시도.
140// ② EOF 가 된 fd 는 배열에서 빼야 한다 — 안 그러면 POLLHUP 이 계속 떠서 100% CPU.
141// ③ revents 는 요청 안 한 비트(POLLHUP/POLLERR/POLLNVAL)도 올라온다. 반드시 검사.
142// 여기서는 두 fd 가 모두 EOF 이거나, 한 번 유휴 타임아웃이 나면 0 을 리턴한다.
143// ===========================================================================
144typedef struct {
145 size_t bytes_a; // fd_a 에서 읽은 총 바이트
146 size_t bytes_b; // fd_b 에서 읽은 총 바이트
147 int timeouts; // poll 타임아웃 횟수
148 int eof_a; // fd_a 가 EOF 인가
149 int eof_b; // fd_b 가 EOF 인가
150 int iters; // 루프 회전 수 (스핀 감지용)
151} pump_t;
152 
153int pump_two(int fd_a, int fd_b, int timeout_ms, pump_t *st) {
154 if (!st) { errno = EINVAL; return -1; }
155 memset(st, 0, sizeof *st);
156 
157 while (!(st->eof_a && st->eof_b)) {
158 struct pollfd pfd[2];
159 int slot[2]; // pfd[i] 가 a(0)인지 b(1)인지
160 int nf = 0;
161 if (!st->eof_a) { pfd[nf].fd = fd_a; pfd[nf].events = POLLIN; pfd[nf].revents = 0; slot[nf] = 0; nf++; }
162 if (!st->eof_b) { pfd[nf].fd = fd_b; pfd[nf].events = POLLIN; pfd[nf].revents = 0; slot[nf] = 1; nf++; }
163 
164 st->iters++;
165 int r = poll(pfd, (nfds_t)nf, timeout_ms);
166 if (r < 0) {
167 if (errno == EINTR) continue; // ① 시그널 — 재시도
168 return -1;
169 }
170 if (r == 0) { st->timeouts++; return 0; } // 유휴 타임아웃 → 상위에서 결정
171 
172 for (int i = 0; i < nf; ++i) {
173 // ③ POLLHUP/POLLERR 도 "읽어봐라" 신호. read 가 0 또는 -1 로 알려준다.
174 if (!(pfd[i].revents & (POLLIN | POLLHUP | POLLERR | POLLNVAL))) continue;
175 char buf[256];
176 ssize_t k = read(pfd[i].fd, buf, sizeof buf);
177 if (k < 0) {
178 if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK) continue;
179 return -1;
180 }
181 if (k == 0) { // ② EOF → 배열에서 제외
182 if (slot[i] == 0) st->eof_a = 1; else st->eof_b = 1;
183 continue;
184 }
185 if (slot[i] == 0) st->bytes_a += (size_t)k;
186 else st->bytes_b += (size_t)k;
187 }
188 }
189 return 0;
190}

193// Q42. self-pipe trick — 다른 스레드/시그널 핸들러에서 이벤트 루프 깨우기
194// ---------------------------------------------------------------------------
195// poll() 안에서 자고 있는 루프를 "지금 당장" 깨우는 표준 기법. 파이프를 하나 만들어
196// 읽기단을 항상 poll 집합에 넣어두고, 깨우고 싶은 쪽이 1바이트를 쓴다.
197// · write() 는 async-signal-safe → 시그널 핸들러에서 호출해도 된다.
198// · 두 끝 모두 O_NONBLOCK: 파이프가 차 있어도 notify 가 블록되면 안 된다
199// (이미 깨울 이벤트가 밀려 있다는 뜻이므로 EAGAIN 은 성공으로 취급).
200// · FD_CLOEXEC: fork/exec 하는 데몬에서 fd 가 자식으로 새지 않게.
201// Linux 라면 eventfd(2) 로 fd 1개만 쓰지만, self-pipe 는 어디서나 동작한다.
202// ===========================================================================
203typedef struct { int rd; int wr; } selfpipe_t;
204 
205static int set_cloexec(int fd) {
206 int f = fcntl(fd, F_GETFD, 0);
207 if (f < 0) return -1;
208 return fcntl(fd, F_SETFD, f | FD_CLOEXEC);
209}
210 
211int sp_init(selfpipe_t *sp) {
212 if (!sp) { errno = EINVAL; return -1; }
213 sp->rd = sp->wr = -1;
214 int fds[2];
215 if (pipe(fds) != 0) return -1;
216 sp->rd = fds[0];
217 sp->wr = fds[1];
218 if (set_nonblocking(sp->rd) != 0 || set_nonblocking(sp->wr) != 0 ||
219 set_cloexec(sp->rd) != 0 || set_cloexec(sp->wr) != 0) {
220 int save = errno;
221 close(sp->rd); close(sp->wr);
222 sp->rd = sp->wr = -1;
223 errno = save;
224 return -1;
225 }
226 return 0;
227}

229// 시그널 핸들러에서 호출해도 안전(write 만 쓴다). errno 를 보존하는 것도 중요.
230int sp_notify(const selfpipe_t *sp) {
231 if (!sp || sp->wr < 0) { errno = EINVAL; return -1; }
232 const unsigned char byte = 'x';
233 for (;;) {
234 ssize_t k = write(sp->wr, &byte, 1);
235 if (k == 1) return 0;
236 if (k < 0 && errno == EINTR) continue;
237 if (k < 0 && (errno == EAGAIN || errno == EWOULDBLOCK))
238 return 0; // 파이프가 이미 꽉 참 = 이미 깨울 예정 → 성공
239 return -1;
240 }
241}

243// 루프가 깨어난 뒤 반드시 비워야 한다. 안 비우면 poll 이 영원히 즉시 리턴(busy loop).
244int sp_drain(const selfpipe_t *sp) {
245 if (!sp || sp->rd < 0) { errno = EINVAL; return -1; }
246 int total = 0;
247 unsigned char buf[64];
248 for (;;) {
249 ssize_t k = read(sp->rd, buf, sizeof buf);
250 if (k > 0) { total += (int)k; continue; }
251 if (k == 0) break; // 쓰기단이 닫힘
252 if (errno == EINTR) continue;
253 if (errno == EAGAIN || errno == EWOULDBLOCK) break; // 다 비웠다
254 return -1;
255 }
256 return total;
257}

259void sp_close(selfpipe_t *sp) {
260 if (!sp) return;
261 if (sp->rd >= 0) close(sp->rd);
262 if (sp->wr >= 0) close(sp->wr);
263 sp->rd = sp->wr = -1;
264}

267// Q43. 라인 프레이밍 리더 — 스트림에서 완전한 줄만 넘기는 상태 있는 파서
268// ---------------------------------------------------------------------------
269// TCP/UART 는 "메시지"가 아니라 "바이트 스트림"이다. 모뎀에 AT+CSQ 를 보내면 응답이
270// "+CSQ: 22," / "99\r\nOK\r\n" 처럼 두 번에 나뉘어 도착할 수 있다. 그래서 리더는
271// 반드시 상태를 들고 있어야 한다(누적 버퍼 + 길이). 완성된 줄만 콜백으로 넘긴다.
272// 실전 요구사항 3가지: ①CRLF/LF 모두 지원 ②버퍼보다 긴 줄은 버리되 그 다음 줄부터
273// 정상 복구(절대 크래시/오버런 금지) ③빈 줄도 유효한 줄.
274// ===========================================================================
275#define LR_CAP 128
276typedef void (*lr_cb_t)(const char *line, void *user);
277 
278typedef struct {
279 char buf[LR_CAP]; // 누적 버퍼 ('\0' 자리 1칸 포함)
280 size_t len; // 현재 누적 길이
281 int overflow; // 현재 줄이 이미 넘쳤는가 (줄 끝까지 버리는 중)
282 int dropped; // 버린 줄 수 — 조용한 손실 금지, 반드시 센다
283} linereader_t;

288 lr->overflow = 0;

293int lr_feed(linereader_t *lr, const char *data, size_t n, lr_cb_t cb, void *user) {
294 if (!lr || (!data && n)) return 0;
295 int lines = 0;
296 for (size_t i = 0; i < n; ++i) {
297 char c = data[i];
298 if (c == '\n') {
299 if (lr->overflow) { // 넘친 줄은 통째로 버리고 여기서 복구
300 lr->overflow = 0;
301 lr->len = 0;
302 continue;
303 }
304 size_t len = lr->len;
305 if (len > 0 && lr->buf[len - 1] == '\r') len--; // CRLF → CR 떼기
306 lr->buf[len] = '\0';
307 if (cb) cb(lr->buf, user);
308 lr->len = 0;
309 lines++;
310 continue;
311 }
312 if (lr->len + 1 >= LR_CAP) { // '\0' 자리 확보 — 오버런 방지
313 if (!lr->overflow) { lr->overflow = 1; lr->dropped++; }
314 continue; // 줄이 끝날 때까지 계속 버린다
315 }
316 lr->buf[lr->len++] = c;
317 }
318 return lines;
319}

321// ===========================================================================
322// Q44. 지수 백오프 + 지터 (pure function)
323// ---------------------------------------------------------------------------
324// 200만 대의 게이트웨이가 클라우드 장애 후 동시에 재연결하면 thundering herd 로
325// 복구를 방해한다. 그래서 ①지수적으로 늘리고 ②상한(cap)에서 멈추고 ③지터로 흩는다.
326// 난수를 **주입**(rnd 파라미터)하는 게 핵심 설계 포인트 — 순수 함수가 되어
327// 하드웨어/시간 없이 단위 테스트가 가능하다("modular, testable embedded software").
328// attempt : 0,1,2,... delay = base * 2^attempt, cap 에서 클램프
329// jitter_pct : 결과를 ±jitter_pct % 범위로 흔든다
330// rnd : 0..9999 (테스트에서는 고정값, 실전에서는 rand()/dev/urandom)
331// ===========================================================================
332uint32_t backoff_delay_ms(uint32_t attempt, uint32_t base_ms, uint32_t cap_ms,
333 uint32_t jitter_pct, uint32_t rnd) {
334 if (base_ms == 0) return 0;
335 if (cap_ms < base_ms) cap_ms = base_ms;
336 
337 uint64_t d = base_ms;
338 if (attempt >= 32) {
339 d = cap_ms; // 시프트 오버플로 방지
340 } else {
341 for (uint32_t i = 0; i < attempt; ++i) {
342 d <<= 1;
343 if (d >= cap_ms) { d = cap_ms; break; }
344 }
345 }
346 if (d > cap_ms) d = cap_ms;
347 if (jitter_pct == 0) return (uint32_t)d;
348 if (jitter_pct > 100) jitter_pct = 100;
349 
350 uint64_t span = d * jitter_pct / 100; // 편차 반폭
351 if (span == 0) return (uint32_t)d;
352 uint64_t width = 2 * span + 1; // [d-span, d+span]
353 uint64_t off = width * (uint64_t)(rnd % 10000u) / 10000u;
354 return (uint32_t)(d - span + off);
355}

357// ===========================================================================
358// Q45. monotonic deadline 유틸
359// ---------------------------------------------------------------------------
360// 타임아웃에 절대 CLOCK_REALTIME(gettimeofday/time())을 쓰지 말 것. NTP 가 시계를
361// 뒤로 당기면 타임아웃이 몇 시간 늘어난다 — 게이트웨이가 "멈춘 것처럼" 보이는 전형적
362// 필드 버그. CLOCK_MONOTONIC 은 부팅 이후 단조 증가라 안전하다.
363// 또 하나: poll 은 "상대 시간(ms)"을 받으므로, 루프를 돌 때마다 남은 시간을 **다시
364// 계산**해야 한다. 안 그러면 EINTR 로 깰 때마다 타임아웃이 리셋되어 무한정 늘어난다.
365// ===========================================================================
366uint64_t now_ms(void) {
367 struct timespec ts;
368 if (clock_gettime(CLOCK_MONOTONIC, &ts) != 0) return 0;

502// ===========================================================================
503// Q48. 에러 경로 정리 — fd 누수 없는 goto cleanup + close()/EINTR 주의
504// ---------------------------------------------------------------------------
505// 오래 도는 데몬에서 fd 누수는 며칠 뒤 EMFILE 로 터진다 — "왜 3일마다 죽지?"의 범인.
506// C 에는 RAII 가 없으므로 관용구는 **단일 출구 + goto cleanup**:
507// 자원 핸들을 무효값(-1/NULL)으로 먼저 선언 → 실패하면 goto → cleanup 에서
508// 유효한 것만 해제. 조기 return 을 흩뿌리면 반드시 하나를 빠뜨린다.
509// close() 와 EINTR: Linux 에서 close() 가 EINTR 를 리턴해도 **fd 는 이미 반납됐다**.
510// 여기서 재시도하면 그 사이 다른 스레드가 같은 번호로 연 fd 를 닫아버린다(심각한 버그).
511// 따라서 close 는 재시도하지 않고 EINTR 을 성공으로 취급한다.
512// 반대로 데이터 무결성이 중요하면 close 전에 fsync 로 에러를 미리 확인해야 한다.
513// ===========================================================================
514int xclose(int fd) {
515 if (fd < 0) return 0; // 이미 닫힘/미할당 — no-op
516 int r = close(fd);
517 if (r < 0 && errno == EINTR) return 0; // fd 는 이미 반납됨 → 재시도 금지
518 return r;
519}
