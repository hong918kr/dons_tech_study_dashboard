62// Q21. atomic 카운터 (fetch_add, relaxed) vs 비원자 카운터
63// ---------------------------------------------------------------------------
64// `c++`는 원자 연산이 아니다 — load / add / store 세 단계다. 두 스레드가 이
65// 세 단계 사이에서 끼어들면 증가가 통째로 유실된다(lost update).
66// 통계 카운터(드롭 수, 재전송 수)처럼 "값만 맞으면 되고 다른 데이터와의 순서는
67// 상관없는" 경우 memory_order_relaxed가 정답이다 — 배리어 비용을 내지 않는다.
68// GC31-E: LTE 재연결 횟수, SIM failover 횟수, 링버퍼 drop 카운터가 전부 이것.
69// ===========================================================================
70typedef struct {
71 _Atomic unsigned long long n;
72} Counter;
73 
74void ctr_init(Counter *c) {
75 atomic_init(&c->n, 0ull);
76}
77 
78// relaxed: 원자성(값 유실 없음)만 필요하고 다른 메모리와의 순서는 필요 없다.
79// 반환값은 "더하기 직전의 값"(fetch-then-add). 이걸 이용해 슬롯 예약도 한다.
80unsigned long long ctr_add(Counter *c, unsigned long long delta) {
81 return atomic_fetch_add_explicit(&c->n, delta, memory_order_relaxed);
82}
83 
84unsigned long long ctr_get(const Counter *c) {
85 return atomic_load_explicit(&c->n, memory_order_relaxed);
86}

88// 비원자 `++`의 최악 인터리빙을 **결정적으로** 재현한다.
89// T1: load(shared) -> reg1 = start
90// T2: load(shared) -> reg2 = start (T1이 store하기 전)
91// T1: reg1+1 -> store(shared)
92// T2: reg2+1 -> store(shared) (T1의 증가를 덮어씀)
93// 증가를 2번 했는데 결과는 start+1 — 이것이 lost update다.
94unsigned long long racy_interleave_demo(unsigned long long start) {
95 unsigned long long shared = start;
96 unsigned long long r1, r2;
97 r1 = shared; /* T1: load */
98 r2 = shared; /* T2: load (끼어듦) */
99 r1 += 1; shared = r1; /* T1: add + store */
100 r2 += 1; shared = r2; /* T2: add + store */
101 return shared; /* start + 1 (1회 유실) */
102}
103 
104// 같은 시나리오라도 atomic RMW는 load-add-store가 쪼개지지 않으므로 유실이 없다.
105unsigned long long atomic_interleave_demo(unsigned long long start) {
106 _Atomic unsigned long long shared;
107 atomic_init(&shared, start);
108 atomic_fetch_add_explicit(&shared, 1ull, memory_order_relaxed); /* T1 전체 */
109 atomic_fetch_add_explicit(&shared, 1ull, memory_order_relaxed); /* T2 전체 */
110 return atomic_load_explicit(&shared, memory_order_relaxed); /* start + 2 */
111}

113// ===========================================================================
114// Q22. stop 플래그: release store / acquire load (volatile로는 안 되는 이유)
115// ---------------------------------------------------------------------------
116// GW31-E는 "오프라인 30분이면 자가 재부팅"한다. 감시 스레드가 종료 사유를 채우고
117// stop 플래그를 세우면, 워커는 그 플래그를 보는 순간 **사유 문자열까지 완성된
118// 상태로** 봐야 한다.
119//
120// writer: reason/seq 기록 → atomic_store(stop, true, RELEASE)
121// reader: atomic_load(stop, ACQUIRE) == true → reason/seq 읽기
122//
123// release/acquire 짝이 하는 일은 딱 하나: "release store 이전의 모든 쓰기"가
124// "그 값을 acquire load로 본 스레드"에게 **먼저 보이도록** 강제한다(happens-before).
125//
126// ★ volatile은 왜 안 되는가 (면접 단골)
127// volatile이 보장하는 것: 컴파일러가 접근을 캐시/삭제/병합하지 못한다.
128// volatile이 보장하지 **않는** 것:
129// ① 원자성 — volatile long long ++ 는 여전히 load/add/store 3단계다.
130// ② 순서 — 컴파일러는 volatile 접근끼리의 순서만 지킨다. 주변의
131// **비-volatile** 접근(reason[] 쓰기!)은 자유롭게 재배치한다.
132// ③ 가시성/CPU — arm64/A53급 멀티코어는 store buffer + weak memory model이라
133// CPU 자체가 순서를 바꾼다. volatile은 DMB 같은 배리어를 내보내지 않는다.
134// → x86에서는 "우연히" 동작해서 더 위험하다(TSO라 store-store 재배치가 없음).
135// 같은 코드를 Cortex-A로 옮기면 필드에서만 터진다.
136// 아래 VolStop은 그 반례를 코드로 남겨둔 것이고, 스레드로는 절대 돌리지 않는다.
137// ===========================================================================
138typedef struct {
139 _Atomic bool stop; // 동기화 지점 (release/acquire)
140 char reason[32];// stop 이전에 쓰이고, stop을 본 뒤에만 읽힌다
141 unsigned seq; // 함께 publish되는 평범한(비원자) 데이터
142} StopFlag;
143 
144void stop_init(StopFlag *f) {
145 atomic_init(&f->stop, false);
146 f->reason[0] = '\0';
147 f->seq = 0;
148}
149 
150// ① 평범한 데이터 먼저 → ② release store로 publish. 순서를 바꾸면 안 된다.
151void stop_request(StopFlag *f, const char *reason, unsigned seq) {
152 snprintf(f->reason, sizeof f->reason, "%s", reason);
153 f->seq = seq;
154 atomic_store_explicit(&f->stop, true, memory_order_release);
155}
156 
157// acquire load: true를 봤다면 writer가 release 이전에 쓴 것이 전부 보인다.
158bool stop_requested(const StopFlag *f) {
159 return atomic_load_explicit(&f->stop, memory_order_acquire);
160}

179// ===========================================================================
180// Q23. CAS 루프로 최대값 갱신 (compare_exchange_weak)
181// ---------------------------------------------------------------------------
182// "최대 신호세기", "최대 큐 깊이", "최대 업로드 지연" 같은 high-water mark는
183// fetch_max가 C11에 없으므로 **CAS 루프**로 만든다. 이것이 lock-free의 기본형이다.
184//
185// ★ 실패 시 규칙 (면접에서 제일 자주 틀리는 지점)
186// compare_exchange_*가 실패하면 expected에 **현재 값이 자동으로 들어온다**.
187// 그러니 루프 안에서 다시 atomic_load 하면 안 된다 — 불필요한 로드일 뿐 아니라
188// "읽고 나서 또 바뀌는" 창을 스스로 넓히는 셈이다.
189// ★ weak vs strong
190// weak는 값이 같아도 **가짜 실패(spurious failure)** 할 수 있다. 대신 LL/SC
191// (ARM의 LDXR/STXR) 기반 CPU에서 더 싸다. 어차피 루프 안이면 weak가 정석.
192// 루프가 없는 단발 CAS에서만 strong을 쓴다.
193// ===========================================================================
194 
195// candidate가 더 크면 갱신하고 true. 아니면 false.
196bool atomic_max_update(_Atomic unsigned *slot, unsigned candidate) {
197 unsigned cur = atomic_load_explicit(slot, memory_order_relaxed);
198 while (candidate > cur) {
199 // 성공: release (이 값 이전의 내 작업을 함께 publish)
200 // 실패: relaxed (아직 아무것도 publish하지 않았다) + cur에 현재값이 실린다
201 if (atomic_compare_exchange_weak_explicit(slot, &cur, candidate,
202 memory_order_release,
203 memory_order_relaxed))
204 return true;
205 /* 실패 → cur는 이미 갱신됨. 다시 load하지 말 것. while 조건이 재검사한다 */
206 }
207 return false;
208}

210// 같은 패턴의 최소값 버전 (대칭성 확인용)
211bool atomic_min_update(_Atomic unsigned *slot, unsigned candidate) {
212 unsigned cur = atomic_load_explicit(slot, memory_order_relaxed);
213 while (candidate < cur) {
214 if (atomic_compare_exchange_weak_explicit(slot, &cur, candidate,
215 memory_order_release,
216 memory_order_relaxed))
217 return true;
218 }
219 return false;
220}

222// ===========================================================================
223// Q24. CAS 스핀락 (atomic_flag test_and_set / clear)
224// ---------------------------------------------------------------------------
225// atomic_flag는 C11에서 **항상 lock-free가 보장되는 유일한 타입**이다.
226// lock : test_and_set이 false를 돌려줄 때까지 회전 → ACQUIRE
227// unlock: clear → RELEASE
228// 이 acquire/release 짝 덕분에 스핀락 안의 **평범한(비원자) 데이터**가 안전해진다.
229//
230// ★ 스핀락을 쓰면 안 되는 경우 (이게 진짜 질문이다)
231// 1) 임계구역이 길거나 그 안에서 블록/시스템콜/malloc을 한다 → CPU를 태운다.
232// 배터리로 도는 MT81 트레일러에서 스핀은 곧 주행거리다.
233// 2) 단일 코어 + 선점형 스케줄러 → 락 보유자가 선점되면 대기자가 자기 타임슬라이스
234// 전체를 태우고서야 양보한다. 최악은 우선순위 역전(priority inversion)으로
235// 진행 자체가 멈춘다. 이때는 mutex(+ priority inheritance)가 정답.
236// 3) 유저스페이스 Linux에서 경합이 있는 경우 → 대개 pthread_mutex가 더 빠르다.
237// (경합 없을 땐 futex 덕에 이미 유저스페이스에서 끝난다)
238// 쓰는 곳: 멀티코어에서 수십 나노초짜리 임계구역, 인터럽트를 끌 수 없는 짧은 구간.
239// ===========================================================================
240typedef struct {
241 atomic_flag held;
242} SpinLock;
243 
244void spin_init(SpinLock *l) {
245 atomic_flag_clear_explicit(&l->held, memory_order_relaxed);
246}

248void spin_lock(SpinLock *l) {
249 // test_and_set: 이전 값을 돌려주고 true로 만든다. false를 받으면 내가 잡은 것.
250 while (atomic_flag_test_and_set_explicit(&l->held, memory_order_acquire)) {
251 /* 실전에서는 여기서 CPU 힌트를 준다: ARM `wfe`/`yield`, x86 `pause`.
252 * 이식성 때문에 여기서는 빈 루프로 둔다. */
253 }
254}

256bool spin_trylock(SpinLock *l) {
257 return !atomic_flag_test_and_set_explicit(&l->held, memory_order_acquire);
258}

260void spin_unlock(SpinLock *l) {

264// ===========================================================================
265// Q25. SPSC 링버퍼 push/pop (release/acquire 짝, 한 칸 비우기)
266// ---------------------------------------------------------------------------
267// 모뎀 UART RX ISR(생산자 1) → 파서 태스크(소비자 1). ISR 안에서는 mutex 금지.
268// 성립 근거: **생산자만 tail을 쓰고, 소비자만 head를 쓴다.** 상대 인덱스는 읽기만
269// 하므로 write-write 경합이 구조적으로 없다. 남은 문제는 "순서"뿐이고 그것을
270// release/acquire 한 쌍이 해결한다.
271//
272// push: buf[t] = b (①데이터)
273// store(tail, RELEASE) (②publish) → ①이 ②보다 먼저 보인다
274// pop : load(tail, ACQUIRE) (②를 봤다면)
275// b = buf[h] (①도 보인다)
276//
277// 한 칸을 비워두면 head==tail(빈 것)과 가득 참을 인덱스만으로 구분할 수 있다.
278// → 용량은 RING_CAP-1. count 필드를 두면 양쪽이 쓰게 되어 SPSC의 이점이 사라진다.
279// ===========================================================================
280#define RING_CAP 64u /* 반드시 2의 거듭제곱 */
281 
282typedef struct {
283 uint8_t buf[RING_CAP];
284 _Atomic unsigned head; /* 소비자만 store */
285 _Atomic unsigned tail; /* 생산자만 store */
286} Ring;
287 
288void ring_init(Ring *r) {
289 atomic_init(&r->head, 0u);
290 atomic_init(&r->tail, 0u);
291}
292 
293// 생산자(ISR) 전용. 가득 차면 false — 실전에서는 drop 카운터를 올리고 즉시 리턴한다.
294bool ring_push(Ring *r, uint8_t b) {
295 unsigned t = atomic_load_explicit(&r->tail, memory_order_relaxed); /* 내가 쓴 값 */
296 unsigned next = (t + 1u) & (RING_CAP - 1u);
297 /* 소비자가 얼마나 비웠는지 본다 → acquire (소비자의 buf 읽기 완료를 본다) */
298 if (next == atomic_load_explicit(&r->head, memory_order_acquire))
299 return false; /* full */
300 
301 r->buf[t] = b; /* ① 데이터 먼저 */
302 atomic_store_explicit(&r->tail, next, memory_order_release); /* ② 그 다음 publish */
303 return true;
304}
305 
306// 소비자(task) 전용. 비면 false.
307bool ring_pop(Ring *r, uint8_t *out) {
308 unsigned h = atomic_load_explicit(&r->head, memory_order_relaxed); /* 내가 쓴 값 */
309 /* acquire: tail을 본 뒤에 buf를 읽어야 생산자의 데이터가 보인다 */
310 if (h == atomic_load_explicit(&r->tail, memory_order_acquire))
311 return false; /* empty */
312 
313 *out = r->buf[h];
314 atomic_store_explicit(&r->head, (h + 1u) & (RING_CAP - 1u),
315 memory_order_release);
316 return true;
317}

319// ===========================================================================
320// Q26. count / full / empty + power-of-two mask (& vs %)
321// ---------------------------------------------------------------------------
322// 부호 없는 뺄셈 `(tail - head) & (CAP-1)`은 인덱스가 랩되어도 정확하다.
323// `%`는 나눗셈 명령이 없는 MCU에서 수십 사이클짜리 라이브러리 호출이 되지만,
324// 2의 거듭제곱이면 `&`는 1사이클이다. ISR 안에서는 이 차이가 실제로 문제가 된다.
325// 주의: `&` 트릭은 **2의 거듭제곱일 때만** 성립한다 → init에서 반드시 검증.
326// ===========================================================================
327bool is_pow2(unsigned x) {
328 return x != 0u && (x & (x - 1u)) == 0u;
329}
330 
331// cap이 2의 거듭제곱이면 idx % cap과 동일하되 나눗셈이 없다.
332unsigned ring_mask_index(unsigned idx, unsigned cap) {
333 return idx & (cap - 1u);
334}

336// 관찰용(느슨한 스냅샷). 생산자/소비자 각각 자기 쪽에서 부르면 정확하다.
337unsigned ring_count(const Ring *r) {
338 unsigned t = atomic_load_explicit(&r->tail, memory_order_acquire);
339 unsigned h = atomic_load_explicit(&r->head, memory_order_acquire);
340 return (t - h) & (RING_CAP - 1u); /* unsigned 뺄셈 → 랩되어도 정확 */
341}

343bool ring_is_empty(const Ring *r) {
344 return ring_count(r) == 0u;
345}

347bool ring_is_full(const Ring *r) {

351// ===========================================================================
352// Q27. seqlock write (홀수/짝수 시퀀스 + release fence)
353// ---------------------------------------------------------------------------
354// GC31-E 링크 상태(RSRP/RSRQ/uptime/SIM 슬롯)를 1초마다 쓰는 writer 1개와,
355// 수시로 읽는 reader 여러 개. 요구: reader가 **같은 시점의 스냅샷**을 보고,
356// reader 때문에 writer가 절대 지연되지 않을 것(writer 우선).
357//
358// seq 홀수 = 쓰는 중, 짝수 = 안정
359// writer: seq++(홀수) → [release fence] → 데이터 기록 → seq++(짝수, release store)
360//
361// 첫 번째 fence가 하는 일: "seq가 홀수가 된 사실"이 데이터 기록보다 **먼저** 보이게
362// 한다. 이게 없으면 reader가 짝수 seq를 본 채로 반쯤 쓰인 데이터를 복사하고도
363// 검증을 통과해버린다.
364// 마지막 release store: 데이터 기록이 "짝수 seq publish"보다 먼저 보이게 한다.
365// ===========================================================================
366typedef struct {
367 int32_t rsrp_dbm; /* 테스트 불변식: rsrq_db == -rsrp_dbm */
368 int32_t rsrq_db;
369 uint32_t uptime_s;
370 uint8_t sim_slot;
371} LinkStatus;
372 
373typedef struct {
374 _Atomic uint32_t seq; /* 짝수 = 안정, 홀수 = 쓰기 중 */
375 LinkStatus data; /* seq로 보호되는 평범한 구조체 (원자 타입이 아니다) */
376} SeqLock;
377 
378void seq_init(SeqLock *s) {
379 atomic_init(&s->seq, 0u);
380 memset(&s->data, 0, sizeof s->data);
381}
382 
383// writer 1명 전용. writer가 여러 명이면 writer끼리는 mutex로 직렬화해야 한다.
384void seq_write(SeqLock *s, const LinkStatus *in) {
385 uint32_t v = atomic_load_explicit(&s->seq, memory_order_relaxed);
386 atomic_store_explicit(&s->seq, v + 1u, memory_order_relaxed); /* 홀수: 쓰는 중 */
387 atomic_thread_fence(memory_order_release); /* seq 증가가 데이터보다 먼저 보이게 */
388 
389 s->data = *in; /* 평범한 쓰기 (여기가 race 지점) */
390 
391 /* 데이터 기록이 짝수 publish보다 먼저 보이게 */
392 atomic_store_explicit(&s->seq, v + 2u, memory_order_release); /* 짝수: 완료 */
393}
394 
395// 현재 writer가 쓰는 중인가 (진단/테스트용)
396bool seq_writing(const SeqLock *s) {
397 return (atomic_load_explicit(&s->seq, memory_order_relaxed) & 1u) != 0u;
398}

400// ===========================================================================
401// Q28. seqlock read (재시도 루프)
402// ---------------------------------------------------------------------------
403// before = load(seq, ACQUIRE); 홀수면 재시도
404// *out = s->data; ← 찢어진 값일 수 있다(정상)
405// [acquire fence]
406// after = load(seq, RELAXED);
407// before == after 면 스냅샷 유효, 아니면 처음부터 다시
408//
409// ★ 정직하게 말해야 할 한계 (면접 가산점)
410// - `*out = s->data`는 C 표준상 **data race**다(비원자 객체를 writer와 동시에 접근).
411// 그래서 `-fsanitize=thread`로 빌드하면 TSan이 정확히 여기를 지적한다.
412// 실무 해법: ① 필드별 relaxed atomic으로 읽고 쓰기 ② Linux 커널처럼
413// READ_ONCE/WRITE_ONCE(volatile 접근 + 컴파일러 배리어)로 플랫폼 보장에 기대기.
414// - 데이터에 **포인터/핸들이 있으면 쓰면 안 된다** — 찢어진 포인터를 잠시라도
415// 역참조하면 죽는다. POD 스냅샷 전용.
416// - reader가 writer보다 훨씬 느리면 영원히 재시도할 수 있다(starvation) →
417// 재시도 상한을 두고 초과 시 mutex 경로로 fallback.
418// ===========================================================================
419// ★ 루프는 do/while이 아니라 for(;;)로 쓴다.
420// do { ... if (before & 1) continue; ... } while (before != after);
421// 이렇게 쓰면 `continue`가 **while 조건으로 점프**해서, 아직 대입된 적 없는
422// `after`(첫 바퀴에는 쓰레기값)와 비교한다. 운 나쁘게 같으면 *out을 한 번도
423// 채우지 않고 루프를 빠져나온다 → 호출자가 초기화되지 않은 스냅샷을 받는다.
424// 흔한 seqlock 예제 코드에 실제로 박혀 있는 버그다.
425void seq_read(const SeqLock *s, LinkStatus *out) {
426 for (;;) {
427 uint32_t before = atomic_load_explicit(&s->seq, memory_order_acquire);
428 if (before & 1u) continue; /* 쓰는 중 → 처음부터 다시 */
429 
430 *out = s->data; /* 찢어진 값일 수 있다 */
431 
432 atomic_thread_fence(memory_order_acquire); /* 복사가 after 로드보다 먼저 */
433 uint32_t after = atomic_load_explicit(&s->seq, memory_order_relaxed);
434 if (before == after) return; /* 같으면 유효한 스냅샷 */
435 }
436}
437 
438// 재시도 상한이 있는 버전 — 실시간 경로에서는 이쪽이 안전하다.
439// 상한 내에 일관된 스냅샷을 못 얻으면 false (호출자가 mutex 경로로 fallback).
440bool seq_try_read(const SeqLock *s, LinkStatus *out, unsigned max_tries) {
441 for (unsigned i = 0; i < max_tries; i++) {
442 uint32_t before = atomic_load_explicit(&s->seq, memory_order_acquire);
443 if (before & 1u) continue;
444 *out = s->data;
445 atomic_thread_fence(memory_order_acquire);
446 uint32_t after = atomic_load_explicit(&s->seq, memory_order_relaxed);
447 if (before == after) return true;
448 }
449 return false;
450}

452// ===========================================================================
453// Q29. atomic refcount (fetch_sub release + acquire fence 후 해제)
454// ---------------------------------------------------------------------------
455// 클라우드(Command)에서 새 설정이 내려와도 읽는 중인 설정을 free하면 안 된다.
456// 설정을 **불변 객체**로 만들고 refcount로 수명을 관리한다.
457//
458// 증가: fetch_add(+1, RELAXED) — 이미 유효한 참조를 들고 있는 상태에서만 부른다.
459// 순서 요구가 없으므로 relaxed로 충분하다.
460// 감소: fetch_sub(-1, RELEASE) — "내가 이 객체에 한 모든 접근"을 publish한다.
461// 반환값이 1이면 내가 마지막 참조 → acquire fence로 **다른 스레드들의
462// 모든 접근을 본 뒤에** free한다.
463//
464// ★ 왜 release만으로는 부족한가
465// release 감소만 하면 "내 접근이 다른 스레드에게 보인다"는 보장뿐이다. free하는
466// 쪽은 반대로 "남들의 접근이 나에게 보인다"가 필요하다 → acquire가 있어야 한다.
467// fetch_sub(acq_rel)로 매번 acquire를 거는 것도 맞지만, 마지막 한 번만
468// fence로 거는 편이 빈번한 감소 경로에서 더 싸다. (Boost/LLVM 표준 관용구)
469// ★ 왜 증가는 relaxed인데 감소는 release인가
470// 증가는 이미 살아있는 객체에 대한 것이라 아무것도 publish할 게 없다.
471// 감소는 "나는 이제 이 객체를 안 본다"를 알리는 것이므로 내 작업을 먼저 flush해야 한다.
472// ===========================================================================
473typedef struct {
474 _Atomic int refs;
475 uint32_t version;
476 uint32_t upload_period_ms;
477 char apn[24];
478} RcConfig;
479 
480RcConfig *rc_new(uint32_t version, const char *apn) {
481 RcConfig *c = calloc(1, sizeof *c);
482 if (!c) return NULL;
483 atomic_init(&c->refs, 1); /* 생성자가 참조 1개를 갖는다 */
484 c->version = version;
485 c->upload_period_ms = version * 10u; /* 불변식: period == version*10 */
486 snprintf(c->apn, sizeof c->apn, "%s", apn);
487 return c;
488}
489 
490// 이미 유효한 참조를 들고 있을 때만 호출 (그래서 relaxed로 충분)
491void rc_get(RcConfig *c) {
492 atomic_fetch_add_explicit(&c->refs, 1, memory_order_relaxed);
493}

495// 마지막 참조였으면 free하고 true를 돌려준다. freed는 통계용(NULL 허용).
496bool rc_put(RcConfig *c, Counter *freed) {
497 if (atomic_fetch_sub_explicit(&c->refs, 1, memory_order_release) == 1) {
498 /* 내가 마지막 → 다른 스레드의 접근을 모두 본 뒤 해제 */
499 atomic_thread_fence(memory_order_acquire);
500 if (freed) ctr_add(freed, 1ull);
501 free(c);
502 return true;
503 }
504 return false;
505}

507int rc_count(const RcConfig *c) {
508 return atomic_load_explicit(&c->refs, memory_order_relaxed);

511// ===========================================================================
512// Q30. ABA 문제와 tagged pointer
513// ---------------------------------------------------------------------------
514// ★ ABA란
515// T1이 top == A를 읽는다 → 선점됨.
516// T2가 A를 pop, B를 pop, A를 다시 push. 이제 top == A지만 **A->next는 완전히
517// 다른 값**이다(B가 아니라 그 아래 노드).
518// T1이 깨어나 CAS(top, A, A->next_old)를 한다 → 값이 A로 같으니 **성공한다**.
519// 리스트가 깨진다. CAS는 "값이 같은가"만 보지 "그 사이에 바뀌었는가"는 못 본다.
520//
521// ★ 해법
522// 1) tagged pointer(= 버전 카운터) — 포인터와 태그를 한 워드에 묶어 CAS.
523// pop/push마다 태그를 올리면 "같은 A"라도 값이 달라져 stale CAS가 실패한다.
524// 64비트 CAS로 (32bit index | 32bit tag), 또는 DWCAS(x86 cmpxchg16b,
525// ARM CASP)로 (64bit ptr | 64bit tag).
526// 2) hazard pointer / epoch-based reclamation / RCU — 재사용 자체를 지연시킨다.
527// 3) 그냥 mutex를 쓴다. 대부분의 임베디드 코드에서 이게 정답이다.
528//
529// 여기서는 배열 기반 freelist에 **인덱스+태그**를 packing해서 단일 스레드로
530// ABA 시나리오를 **결정적으로 재현·검증**한다(멀티스레드 없이 증명 가능).
531// 인덱스는 1-based로 저장해 0을 "비었음"(NULL)으로 쓴다.
532// ===========================================================================
533#define POOL_N 8u
534 
535#define TAG_PACK(idx1, tag) (((uint64_t)(uint32_t)(tag) << 32) | (uint32_t)(idx1))
536#define TAG_IDX1(v) ((uint32_t)((v) & 0xFFFFFFFFu)) /* 1-based, 0=NULL */
537#define TAG_CNT(v) ((uint32_t)((v) >> 32))
538 
539typedef struct {
540 uint32_t next[POOL_N]; /* freelist 링크 (1-based, 0 = 끝) */
541 _Atomic uint64_t head; /* [63:32]=tag, [31:0]=1-based index */
542} TaggedPool;
543 
544// 0..POOL_N-1 을 전부 freelist에 올린다 (0이 top이 되도록 역순으로 push).
545void pool_init(TaggedPool *p) {
546 atomic_init(&p->head, (uint64_t)0);
547 for (unsigned i = 0; i < POOL_N; i++) p->next[i] = 0u;
548 for (unsigned i = POOL_N; i-- > 0; ) {
549 uint64_t old = atomic_load_explicit(&p->head, memory_order_relaxed);
550 p->next[i] = TAG_IDX1(old);
551 atomic_store_explicit(&p->head, TAG_PACK(i + 1u, TAG_CNT(old) + 1u),
552 memory_order_relaxed);
553 }
554}
555 
556// CAS 루프 pop. 성공 시 *out_idx 에 0-based 인덱스.
557bool pool_pop(TaggedPool *p, uint32_t *out_idx) {
558 uint64_t old = atomic_load_explicit(&p->head, memory_order_acquire);
559 for (;;) {
560 uint32_t i1 = TAG_IDX1(old);
561 if (i1 == 0u) return false; /* 비었음 */
562 uint64_t neu = TAG_PACK(p->next[i1 - 1u], TAG_CNT(old) + 1u); /* 태그 +1 */
563 if (atomic_compare_exchange_weak_explicit(&p->head, &old, neu,
564 memory_order_acq_rel,
565 memory_order_acquire)) {
566 *out_idx = i1 - 1u;
567 return true;
568 }
569 /* 실패 → old에 현재 head가 자동으로 실린다. 다시 load하지 않는다. */
570 }
571}
572 
573// CAS 루프 push. 태그를 항상 올린다 — 이것이 ABA 방어의 전부다.
574void pool_push(TaggedPool *p, uint32_t idx) {
575 uint64_t old = atomic_load_explicit(&p->head, memory_order_relaxed);
576 uint64_t neu;
577 do {
578 p->next[idx] = TAG_IDX1(old);
579 neu = TAG_PACK(idx + 1u, TAG_CNT(old) + 1u);
580 } while (!atomic_compare_exchange_weak_explicit(&p->head, &old, neu,
581 memory_order_release,
582 memory_order_relaxed));
583}
584 
585uint64_t pool_head_raw(const TaggedPool *p) {
586 return atomic_load_explicit(&p->head, memory_order_acquire);
587}

589// "선점되었던 스레드"가 옛 head 값으로 시도하는 stale CAS. 태그 덕에 실패해야 한다.
590bool pool_stale_cas(TaggedPool *p, uint64_t stale_head, uint64_t desired) {
591 uint64_t expected = stale_head;
592 return atomic_compare_exchange_strong_explicit(&p->head, &expected, desired,
593 memory_order_acq_rel,
594 memory_order_acquire);
595}
