33// ===========================================================================
34// 공통 타입 — 프레임과 N-버퍼 풀
35// ---------------------------------------------------------------------------
36// 버퍼 하나는 언제나 셋 중 한 상태다:
37// WRITER 소유(write_idx) / READY(publish됨, ready_idx) / READER 소유(reader_idx)
38// 어느 상태에도 속하지 않으면 FREE. 인덱스 세 개가 곧 소유권 장부다.
39// nbuf==2 면 더블 버퍼링, nbuf==3 이면 트리플 버퍼링 — 코드는 같다.
40// ===========================================================================
41#define FRAME_LEN 128
42#define MAXBUF 3

44typedef struct {
45 unsigned seq; // 프레임 일련번호 (1부터)
46 unsigned char data[FRAME_LEN]; // 전부 (seq & 0xFF) 로 채운다 → tearing 검출용
47} Frame;

49typedef struct {
50 Frame buf[MAXBUF];
51 int nbuf; // 실제로 쓰는 버퍼 수 (2 또는 3)
52 int write_idx; // writer 소유 버퍼, 없으면 -1
53 int ready_idx; // publish된 최신 버퍼, 없으면 -1
54 int reader_idx; // reader 대여 중 버퍼, 없으면 -1
55 unsigned long dropped; // 소비되지 못하고 버려진 프레임 총수
56 unsigned long starved; // write_begin에서 빈 버퍼가 없어 READY를 희생한 횟수
57 pthread_mutex_t m; // '데이터'가 아니라 '소유권 장부'를 보호한다
58} MBuf;

60// ===========================================================================
61// Q31. 더블 버퍼 소유권 모델 — 초기화 + write_begin
62// ---------------------------------------------------------------------------
63// mb_write_begin: writer가 채울 빈(FREE) 버퍼를 확보한다. **항상 성공한다** —
64// 실시간 경로(센서 ISR/DMA 완료 콜백)는 소비자를 기다릴 수 없기 때문이다.
65// 빈 버퍼가 없으면(reader가 하나 + READY가 하나 점유) READY를 희생시키고
66// dropped/starved를 올린다. reader가 든 버퍼는 절대 건드리지 않는다 — 그게
67// 곧 tearing이다. GC31-E 게이트웨이의 링크 통계 스냅샷, MT81 트레일러 카메라
68// 프레임처럼 "최신값만 중요한" 데이터의 표준 패턴.
69// ===========================================================================
70void mb_init(MBuf *mb, int nbuf)
71{
72 if (!mb) return;
73 memset(mb, 0, sizeof *mb);
74 if (nbuf < 2) nbuf = 2; // 2 미만은 소유권 모델이 성립하지 않는다
75 if (nbuf > MAXBUF) nbuf = MAXBUF;
76 mb->nbuf = nbuf;
77 mb->write_idx = -1;
78 mb->ready_idx = -1;
79 mb->reader_idx = -1;
80 pthread_mutex_init(&mb->m, NULL);
81}
82 
83void mb_destroy(MBuf *mb)
84{
85 if (!mb) return;
86 pthread_mutex_destroy(&mb->m);
87}

89Frame *mb_write_begin(MBuf *mb)
90{
91 if (!mb) return NULL;
92 pthread_mutex_lock(&mb->m);
93 int idx = -1;
94 for (int i = 0; i < mb->nbuf; i++) { // 아무도 소유하지 않은 버퍼 찾기
95 if (i != mb->ready_idx && i != mb->reader_idx) { idx = i; break; }
96 }
97 if (idx < 0) {
98 // nbuf>=2 이면 여기 도달했다는 건 reader 1 + READY 1 로 꽉 찬 경우뿐이다.
99 // → READY를 희생한다. (reader 버퍼를 뺏으면 tearing)
100 idx = mb->ready_idx;
101 mb->ready_idx = -1;
102 mb->dropped++;
103 mb->starved++;
104 }
105 mb->write_idx = idx;
106 pthread_mutex_unlock(&mb->m);
107 return &mb->buf[idx]; // 데이터 write는 락 '밖'에서
108}

110// ===========================================================================
111// Q32. write_commit (publish) / read_acquire / read_release — 소유권 이전
112// ---------------------------------------------------------------------------
113// commit: writer 소유 → READY. 이전 READY가 아직 안 읽혔으면 그건 버려진다(dropped++).
114// acquire: READY → reader 소유. 없으면 NULL(논블로킹). reader가 이미 하나를
115// 들고 있으면 반납 전까지 NULL — 한 번에 한 프레임만 대여한다.
116// release: reader 소유 → FREE.
117// 세 함수 모두 임계구역 안에서 하는 일은 int 대입 몇 줄뿐이다.
118// ===========================================================================
119void mb_write_commit(MBuf *mb)
120{
121 if (!mb) return;
122 pthread_mutex_lock(&mb->m);
123 if (mb->write_idx >= 0) {
124 if (mb->ready_idx >= 0)
125 mb->dropped++; // 소비 전에 더 새 프레임이 나옴
126 mb->ready_idx = mb->write_idx; // publish
127 mb->write_idx = -1;
128 }
129 pthread_mutex_unlock(&mb->m);
130}
131 
132Frame *mb_read_acquire(MBuf *mb)
133{
134 if (!mb) return NULL;
135 Frame *f = NULL;
136 pthread_mutex_lock(&mb->m);
137 if (mb->reader_idx < 0 && mb->ready_idx >= 0) {
138 mb->reader_idx = mb->ready_idx; // READY를 reader 소유로 이전
139 mb->ready_idx = -1;
140 f = &mb->buf[mb->reader_idx];
141 }
142 pthread_mutex_unlock(&mb->m);
143 return f;
144}
145 
146void mb_read_release(MBuf *mb)
147{
148 if (!mb) return;
149 pthread_mutex_lock(&mb->m);
150 mb->reader_idx = -1; // 다시 FREE
151 pthread_mutex_unlock(&mb->m);
152}

154// ===========================================================================
155// Q33. tearing 검출 + 회계 불변식
156// ---------------------------------------------------------------------------
157// tearing = 소비자가 '절반만 갱신된' 프레임을 보는 것. 검출하려면 프레임 전체를
158// seq에서 유도한 같은 패턴으로 채우고, 읽을 때 전 바이트가 그 패턴인지 본다.
159// 한 바이트라도 다르면 두 세대의 데이터가 섞인 것 = torn.
160//
161// 회계 불변식(면접에서 반드시 말할 문장):
162// consumed + dropped + leftover(READY) + inflight(WRITER) == produced
163// 즉 생산된 모든 프레임은 "읽혔거나, 버려졌거나, 아직 남아있다" 셋 중 하나다.
164// 조용한 손실(silent loss)이 실시간 시스템에서 가장 디버깅하기 어렵다.
165// ===========================================================================
166void frame_fill(Frame *f, unsigned seq)
167{
168 if (!f) return;
169 f->seq = seq;
170 memset(f->data, (int)(seq & 0xFFu), FRAME_LEN); // 전 바이트를 같은 패턴으로
171}
172 
173bool frame_is_torn(const Frame *f)
174{
175 if (!f) return true; // NULL은 안전하게 "찢어짐" 취급
176 unsigned char expect = (unsigned char)(f->seq & 0xFFu);
177 for (int i = 0; i < FRAME_LEN; i++)
178 if (f->data[i] != expect) return true;
179 return false;
180}

182// 스레드 join 후(정지 상태)에만 호출한다 — 락 없이 장부를 읽는다.
183bool mb_accounting_ok(const MBuf *mb, unsigned long produced, unsigned long consumed)
184{
185 if (!mb) return false;
186 unsigned long leftover = (mb->ready_idx >= 0) ? 1UL : 0UL;
187 unsigned long inflight = (mb->write_idx >= 0) ? 1UL : 0UL;
188 return consumed + mb->dropped + leftover + inflight == produced;
189}

191// ---- 생산자/소비자 스레드 (Q33/Q34 공용) -----------------------------------
192typedef struct {
193 MBuf *mb;
194 unsigned nframes;
195 _Atomic bool done;
196 unsigned long read_ok; // 소비자만 씀
197 unsigned long torn; // 소비자만 씀
198 unsigned long out_of_order; // 소비자만 씀
199} PingArgs;

201static void *pp_producer(void *arg)
202{
203 PingArgs *a = (PingArgs *)arg;
204 for (unsigned s = 1; s <= a->nframes; s++) {
205 Frame *f = mb_write_begin(a->mb);
206 frame_fill(f, s); // 락 밖에서 채운다
207 mb_write_commit(a->mb);
208 // 실제 센서는 프레임 사이에 간격이 있다. 여기서는 yield로 소비자에게
209 // 스케줄 기회를 준다 — 단정은 전부 '불변식'이라 결과는 여전히 결정적이다.
210 if ((s & 0x7u) == 0u) sched_yield();
211 }
212 atomic_store_explicit(&a->done, true, memory_order_release);
213 return NULL;
214}
215 
216static void *pp_consumer(void *arg)
217{
218 PingArgs *a = (PingArgs *)arg;
219 unsigned last = 0;
220 for (;;) {
221 // done을 '먼저' 읽는다: done==true를 본 뒤의 acquire가 NULL이면
222 // 남은 프레임이 정말 없다는 뜻 → 무한 스핀 없이 안전하게 종료.
223 bool done = atomic_load_explicit(&a->done, memory_order_acquire);
224 Frame *f = mb_read_acquire(a->mb);
225 if (!f) {
226 if (done) break;
227 continue; // 실전이라면 condvar로 대기
228 }
229 if (frame_is_torn(f)) a->torn++;
230 if (f->seq < last) a->out_of_order++; // 항상 최신만 오므로 단조 증가
231 last = f->seq;
232 a->read_ok++;
233 mb_read_release(a->mb);
234 }
235 return NULL;
236}

238// 스레드 테스트 1회 실행. 반환: 회계 불변식 통과 여부.
239static bool pp_run(MBuf *mb, unsigned nframes, PingArgs *out)
240{
241 pthread_t p, c;
242 PingArgs a;
243 memset(&a, 0, sizeof a);
244 a.mb = mb;
245 a.nframes = nframes;
246 atomic_store(&a.done, false);
247 pthread_create(&c, NULL, pp_consumer, &a);
248 pthread_create(&p, NULL, pp_producer, &a);
249 pthread_join(p, NULL);
250 pthread_join(c, NULL);
251 if (out) *out = a;
252 return mb_accounting_ok(mb, nframes, a.read_ok);
253}

257// ---------------------------------------------------------------------------
258// 더블 버퍼의 약점: reader가 하나를 들고 있고 다른 하나가 READY인 순간
259// writer는 쓸 곳이 없어 READY를 빼앗는다(starved++). 버퍼를 하나 더 두면
260// "reader 1 + READY 1 + writer 1"이 동시에 성립하므로 **writer는 절대 굶지
261// 않는다**. 대가는 메모리 +50% (1080p RGB 프레임이면 ~6MB → ~9MB).
262// 주의: 트리플이어도 commit 시 '아직 안 읽힌 READY를 덮어쓰는' drop은 남는다.
263// 그건 writer가 막힌 게 아니라 소비자가 느려 프레임이 낡은 것 — 다른 현상이다.

267// ===========================================================================
268// Q35. latest-value 슬롯 — mutex 버전 vs seqlock 버전
269// ---------------------------------------------------------------------------
270// "최신값 하나"만 필요한 작은 구조체(링크 통계, 배터리 상태, GPS fix)는 버퍼
271// 풀까지 갈 필요가 없다. 두 가지 선택지:
272// (A) mutex: 단순·정확. writer가 reader를 잠깐 막는다(우선순위 역전 가능).
273// (B) seqlock: writer가 **절대 막히지 않는다**. reader는 찢어진 읽기를
274// 감지하고 재시도한다. 읽기 多 · 쓰기 少 · 페이로드 小 일 때 최적.
275// seqlock은 writer가 하나뿐이어야 하고, 페이로드에 포인터가 있으면 위험하다
276// (찢어진 포인터를 역참조하게 된다). 그래서 POD 스냅샷 전용.
277// ===========================================================================
278typedef struct {
279 uint64_t ts_us;
280 int32_t rssi_dbm;
281 int32_t check; // ts/rssi에서 유도 — 찢어진 읽기 검출용
282} LinkSample;
283 
284static int32_t sample_check(uint64_t ts, int32_t rssi)
285{
286 return (int32_t)(ts & 0x7FFFFFFFu) ^ rssi;
287}
288 
289bool sample_consistent(const LinkSample *s)
290{
291 if (!s) return false;
292 return s->check == sample_check(s->ts_us, s->rssi_dbm);
293}

335// ---- (B) seqlock 버전 ------------------------------------------------------
336// seq 홀수 = 쓰기 진행 중. 짝수 = 안정. reader는 앞뒤 seq가 같고 짝수일 때만 채택.
337// 페이로드를 _Atomic + relaxed 로 두는 이유: 알고리즘상 "읽고 버리는" 경합이
338// 정상 동작이지만, 일반 변수로 두면 그것이 C 표준상 data race(UB)이고
339// ThreadSanitizer도 경고한다. relaxed atomic + fence면 의미는 같고 UB가 아니다.
340#define SL_MAX_RETRY 1000
341 
342typedef struct {
343 _Atomic unsigned seq; // 짝수=안정, 홀수=쓰기 중
344 _Atomic uint64_t ts_us;
345 _Atomic int32_t rssi_dbm;
346 _Atomic int32_t check;
347} SeqSlot;
348 
349void sl_init(SeqSlot *sl)
350{
351 if (!sl) return;
352 atomic_init(&sl->seq, 0u);
353 atomic_init(&sl->ts_us, (uint64_t)0);
354 atomic_init(&sl->rssi_dbm, (int32_t)0);
355 atomic_init(&sl->check, sample_check(0, 0));
356}

358void sl_store(SeqSlot *sl, LinkSample s)
359{
360 if (!sl) return;
361 unsigned s0 = atomic_load_explicit(&sl->seq, memory_order_relaxed);
362 atomic_store_explicit(&sl->seq, s0 + 1u, memory_order_relaxed); // 홀수 진입
363 atomic_thread_fence(memory_order_release);
364 atomic_store_explicit(&sl->ts_us, s.ts_us, memory_order_relaxed);
365 atomic_store_explicit(&sl->rssi_dbm, s.rssi_dbm, memory_order_relaxed);
366 atomic_store_explicit(&sl->check, s.check, memory_order_relaxed);
367 atomic_thread_fence(memory_order_release);
368 atomic_store_explicit(&sl->seq, s0 + 2u, memory_order_release); // 짝수 복귀
369}

371// 재시도 상한이 있다 — 미구현/폭주 상태에서도 절대 무한 루프에 빠지지 않는다.
372bool sl_load(const SeqSlot *sl, LinkSample *out)
373{
374 if (!sl || !out) return false;
375 for (int try_i = 0; try_i < SL_MAX_RETRY; try_i++) {
376 unsigned s1 = atomic_load_explicit(&sl->seq, memory_order_acquire);
377 if (s1 & 1u) continue; // writer가 쓰는 중
378 LinkSample tmp;
379 tmp.ts_us = atomic_load_explicit(&sl->ts_us, memory_order_relaxed);
380 tmp.rssi_dbm = atomic_load_explicit(&sl->rssi_dbm, memory_order_relaxed);
381 tmp.check = atomic_load_explicit(&sl->check, memory_order_relaxed);
382 atomic_thread_fence(memory_order_acquire);
383 unsigned s2 = atomic_load_explicit(&sl->seq, memory_order_relaxed);
384 if (s1 == s2) { *out = tmp; return true; } // 읽는 동안 안 바뀜 → 채택
385 }
386 return false; // 재시도 상한 초과
387}

396#define SL_WRITES 20000u
397#define SL_ATTEMPTS 20000
398 
399typedef struct {
400 SeqSlot *sl;
401 unsigned long succ, torn, retry_exhausted;
402} SlArgs;
403 
404static void *sl_writer(void *arg)
405{
406 SlArgs *a = (SlArgs *)arg;
407 for (unsigned i = 1; i <= SL_WRITES; i++) {
408 LinkSample s;
409 s.ts_us = (uint64_t)i * 1000u;
410 s.rssi_dbm = -30 - (int32_t)(i % 40u);
411 s.check = sample_check(s.ts_us, s.rssi_dbm);
412 sl_store(a->sl, s);
413 }
414 return NULL;
415}
416 
417static void *sl_reader(void *arg)
418{
419 SlArgs *a = (SlArgs *)arg;
420 for (int i = 0; i < SL_ATTEMPTS; i++) { // 고정 횟수 → 스케줄링 독립
421 LinkSample s;
422 if (sl_load(a->sl, &s)) {
423 a->succ++;
424 if (!sample_consistent(&s)) a->torn++; // 찢어진 값이 새어나왔는가
425 } else {
426 a->retry_exhausted++;
427 }
428 }
429 return NULL;
430}

433// Q36. 레이트 디커플링 — 400Hz 생산 → 100Hz 소비
434// ---------------------------------------------------------------------------
435// 4:1 데시메이션. 두 정책 모두 순수 함수로 분리해서 하드웨어 없이 테스트한다.
436// RD_LATEST: 블록의 마지막 샘플 채택. 지연 최소, 계단/이벤트성 신호에 적합
437// (링크 up/down, 도어 상태, 최신 GPS fix).
438// RD_MEAN : 블록 평균. 노이즈 억제, 연속 아날로그 신호에 적합
439// (RSSI, 온도, 전류, 공기질 PM2.5).
440// 판단 기준: "이 신호의 순간값이 의미가 있나, 아니면 추세가 의미가 있나?"
441// 평균은 에일리어싱을 줄이는 대신 계단 엣지를 뭉갠다(지연 ~블록 절반).
442// ===========================================================================
443typedef enum { RD_LATEST = 0, RD_MEAN = 1 } RatePolicy;
444 
445// 반올림 나눗셈(0에서 먼 쪽으로). den > 0 가정.
446static int32_t round_div(int64_t num, int64_t den)
447{
448 if (den <= 0) return 0;
449 if (num >= 0) return (int32_t)((num + den / 2) / den);
450 return (int32_t)((num - den / 2) / den);
451}
452 
453int32_t rd_latest(const int32_t *blk, int n)
454{
455 if (!blk || n <= 0) return 0;
456 return blk[n - 1];
457}
458 
459int32_t rd_mean(const int32_t *blk, int n)
460{
461 if (!blk || n <= 0) return 0;
462 int64_t sum = 0;
463 for (int i = 0; i < n; i++) sum += blk[i]; // int64 누적 → 오버플로 방지
464 return round_div(sum, n);
465}
466 
467// in[n_in]을 factor:1로 줄여 out에 쓴다. 꽉 찬 블록만 처리(꼬리는 다음 호출로 미룸).
468// 반환: 실제로 쓴 출력 개수.
469int rd_decimate(const int32_t *in, int n_in, int factor,
470 RatePolicy pol, int32_t *out, int out_cap)
471{
472 if (!in || !out || n_in <= 0 || factor <= 0 || out_cap <= 0) return 0;
473 int nblocks = n_in / factor;
474 if (nblocks > out_cap) nblocks = out_cap;
475 for (int b = 0; b < nblocks; b++) {
476 const int32_t *blk = in + (size_t)b * (size_t)factor;
477 out[b] = (pol == RD_MEAN) ? rd_mean(blk, factor) : rd_latest(blk, factor);
478 }
479 return nblocks;
480}

482// ===========================================================================
483// Q37. DMA ping-pong 시뮬레이션
484// ---------------------------------------------------------------------------
485// 하드웨어 DMA가 buf[0]을 채우는 동안 CPU는 buf[1]을 처리한다. 전송 완료
486// 인터럽트에서 하는 일은 **인덱스 교환뿐**이다 — 데이터 복사도, 파싱도,
487// 락 대기도 하지 않는다(ISR은 수 마이크로초 안에 끝나야 하고, 블로킹 API는
488// 애초에 호출 불가).
489//
490// ⚠ 캐시 주의(실제 하드웨어): DMA는 캐시를 우회해 DRAM에 직접 쓴다.
491// - DMA가 채운 버퍼를 CPU가 읽기 전 → invalidate (오래된 캐시 라인 폐기)
492// - CPU가 채운 버퍼를 DMA가 읽기 전 → clean/flush (쓰기버퍼→DRAM 반영)
493// 버퍼는 캐시 라인 정렬 + 라인 크기 배수여야 한다. 아니면 인접 변수까지
494// 같이 무효화되어 "가끔 깨지는" 최악의 버그가 된다. Linux에서는 보통
495// dma_map_single()/dma_unmap_single()(스트리밍) 또는 dma_alloc_coherent()
496// (논캐시 메모리)가 이걸 대신해준다.
497//
498// 버퍼가 2개뿐이면 소비자의 데드라인은 **전송 1주기**다. 그 안에 못 가져가면
499// overrun이고, 가져간 뒤에도 다음 완료에서 DMA가 그 버퍼를 다시 쓰기 시작한다
500// → 처리시간이 주기를 넘으면 트리플 버퍼(Q34)로 가야 한다.
501// ===========================================================================
502#define DMA_LEN 64
503 
504typedef struct {
505 unsigned char buf[2][DMA_LEN];
506 int active; // DMA가 지금 채우는 버퍼 (ISR 컨텍스트 전용)
507 _Atomic int full; // 앱이 가져갈 수 있는 버퍼, 없으면 -1
508 _Atomic unsigned long overrun; // 앱이 못 가져간 채 덮인 횟수
509 _Atomic unsigned long completes; // 전송 완료 인터럽트 횟수
510} DmaPP;
511 
512void dma_pp_init(DmaPP *d)
513{
514 if (!d) return;
515 memset(d, 0, sizeof *d);
516 d->active = 0;
517 atomic_init(&d->full, -1);
518 atomic_init(&d->overrun, 0UL);
519 atomic_init(&d->completes, 0UL);
520}
521 
522// ISR 컨텍스트. 인덱스 교환만 — 복사/처리 금지.
523void dma_pp_isr_complete(DmaPP *d)
524{
525 if (!d) return;
526 int done = d->active;
527 d->active = done ^ 1; // 다음 전송은 반대쪽 버퍼로
528 int prev = atomic_exchange_explicit(&d->full, done, memory_order_acq_rel);
529 if (prev >= 0) // 직전 것을 앱이 못 가져갔다
530 atomic_fetch_add_explicit(&d->overrun, 1UL, memory_order_relaxed);
531 atomic_fetch_add_explicit(&d->completes, 1UL, memory_order_relaxed);
532}
533 
534// 앱 컨텍스트. full 버퍼를 '가져간다'(복사하지 않고 포인터만 넘긴다).
535unsigned char *dma_pp_take(DmaPP *d, int *idx_out)
536{
537 if (!d) return NULL;
538 int idx = atomic_exchange_explicit(&d->full, -1, memory_order_acq_rel);
539 if (idx < 0) return NULL;
540 if (idx_out) *idx_out = idx;
541 return d->buf[idx];
542}
543 
544int dma_pp_active(const DmaPP *d)
545{
546 return d ? d->active : -1;
547}
548 
549unsigned long dma_pp_overrun(const DmaPP *d)
550{
551 return d ? atomic_load_explicit(&d->overrun, memory_order_relaxed) : 0UL;
552}

554// ===========================================================================
555// Q38. 프레임 지터/누락 측정
556// ---------------------------------------------------------------------------
557// 타임스탬프 배열 하나만 받는 **순수 함수**로 만든다. 그래야 카메라도 커널도
558// 없이 단위 테스트가 가능하다(모듈성·테스트 가능성 항목에 그대로 해당).
559// 놓친 프레임 수는 간격을 nominal로 '반올림' 나눗셈해 추정한다 — 그래야
560// 정상 범위의 지터(±40%)를 누락으로 오판하지 않는다.
561// ===========================================================================
562typedef struct {
563 int n_gaps; // 간격 개수 (= n-1)
564 unsigned long missed; // 추정 누락 프레임 수
565 unsigned long max_gap_us; // 최대 간격
566 unsigned long min_gap_us; // 최소 간격
567 unsigned long max_jitter_us; // max |gap - nominal|
568 unsigned long nonmonotonic; // 타임스탬프가 거꾸로 간 횟수
569} JitterStats;
570 
571JitterStats jitter_analyze(const unsigned long *ts_us, int n, unsigned long nominal_us)
572{
573 JitterStats st;
574 memset(&st, 0, sizeof st);
575 if (!ts_us || n < 2 || nominal_us == 0) return st; // 안전한 0 결과
576 
577 unsigned long min_gap = ULONG_MAX;
578 for (int i = 1; i < n; i++) {
579 unsigned long gap;
580 if (ts_us[i] < ts_us[i - 1]) { // 시계 역행 / 잘못된 입력
581 st.nonmonotonic++;
582 gap = 0;
583 } else {
584 gap = ts_us[i] - ts_us[i - 1];
585 }
586 st.n_gaps++;
587 if (gap > st.max_gap_us) st.max_gap_us = gap;
588 if (gap < min_gap) min_gap = gap;
589 
590 unsigned long jit = (gap > nominal_us) ? gap - nominal_us : nominal_us - gap;
591 if (jit > st.max_jitter_us) st.max_jitter_us = jit;
592 
593 // 간격을 nominal의 몇 배로 볼 것인가 — 반올림
594 unsigned long k = (gap + nominal_us / 2) / nominal_us;
595 if (k > 1) st.missed += k - 1; // k개 주기 = k-1개 누락
596 }
597 st.min_gap_us = (st.n_gaps > 0) ? min_gap : 0UL;
598 return st;
599}
