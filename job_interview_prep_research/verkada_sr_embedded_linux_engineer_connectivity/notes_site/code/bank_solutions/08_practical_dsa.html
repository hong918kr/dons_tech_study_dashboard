38// ===========================================================================
39// Q69. LRU 캐시 (해시 + 이중 연결 리스트, 고정 용량, 노드 풀)
40// ---------------------------------------------------------------------------
41// Command 게이트웨이가 디바이스 메타데이터(마지막 상태/펌웨어 버전)를 캐시한다.
42// 용량이 고정이고 heap fragmentation 을 피해야 하므로 malloc 을 반복하지 않는다.
43// 노드는 정적 풀에서 free list 로 꺼내 쓰고, 링크는 포인터가 아니라 배열 인덱스다.
44// 해시는 버킷 체이닝을 노드 안(hnext)에 심어 별도 할당이 전혀 없다.
45// get/put 모두 평균 O(1): 해시 조회 O(1) + 리스트 재연결 O(1).
46// ===========================================================================
47#define LRU_MAX_CAP 64
48#define LRU_BUCKETS 128 /* 2의 거듭제곱 → & 마스킹으로 모듈로 대체 */
49#define LRU_NIL (-1)
50 
51typedef struct {
52 int key, val;
53 int prev, next; /* LRU 이중 연결 리스트 (free list 일 때는 next 만 사용) */
54 int hnext; /* 해시 버킷 체인 */
55 bool used;
56} LruNode;
57 
58typedef struct {
59 LruNode pool[LRU_MAX_CAP];
60 int bucket[LRU_BUCKETS];
61 int head, tail; /* head = MRU(가장 최근), tail = LRU(가장 오래됨) */
62 int free_head; /* 노드 풀의 free list */
63 int cap, count;
64 unsigned evictions; /* 조용한 손실 금지 — 항상 센다 */
65} LruCache;

67static uint32_t lru_mix(int key) { /* Knuth 곱셈 해시 (음수 키 안전) */
68 uint32_t x = (uint32_t)key * 2654435761u;
69 return x ^ (x >> 16);
70}

72void lru_init(LruCache *c, int cap) {
73 if (!c) return;
74 if (cap < 1) cap = 1;
75 if (cap > LRU_MAX_CAP) cap = LRU_MAX_CAP;
76 memset(c, 0, sizeof *c);
77 c->cap = cap;
78 c->head = c->tail = LRU_NIL;
79 for (int i = 0; i < LRU_BUCKETS; i++) c->bucket[i] = LRU_NIL;
80 for (int i = 0; i < cap; i++) { /* free list 를 미리 엮어둔다 */
81 c->pool[i].next = (i + 1 < cap) ? i + 1 : LRU_NIL;
82 c->pool[i].used = false;
83 }
84 c->free_head = 0;
85}

87/* --- 노드 풀 --------------------------------------------------------------- */
88static int lru_alloc(LruCache *c) {
89 int i = c->free_head;
90 if (i == LRU_NIL) return LRU_NIL;
91 c->free_head = c->pool[i].next;
92 c->pool[i].used = true;
93 return i;
94}
95static void lru_release(LruCache *c, int i) {
96 c->pool[i].used = false;
97 c->pool[i].next = c->free_head;
98 c->free_head = i;
99}

101/* --- 이중 연결 리스트 ------------------------------------------------------- */
102static void lru_unlink(LruCache *c, int i) {
103 int p = c->pool[i].prev, n = c->pool[i].next;
104 if (p != LRU_NIL) c->pool[p].next = n; else c->head = n;
105 if (n != LRU_NIL) c->pool[n].prev = p; else c->tail = p;
106 c->pool[i].prev = c->pool[i].next = LRU_NIL;
107}
108static void lru_push_front(LruCache *c, int i) {
109 c->pool[i].prev = LRU_NIL;
110 c->pool[i].next = c->head;
111 if (c->head != LRU_NIL) c->pool[c->head].prev = i;
112 c->head = i;
113 if (c->tail == LRU_NIL) c->tail = i;
114}

116/* --- 해시 버킷 체인 --------------------------------------------------------- */
117static int lru_find(const LruCache *c, int key) {
118 int b = (int)(lru_mix(key) & (LRU_BUCKETS - 1));
119 for (int i = c->bucket[b]; i != LRU_NIL; i = c->pool[i].hnext)
120 if (c->pool[i].key == key) return i;
121 return LRU_NIL;
122}
123static void lru_hash_insert(LruCache *c, int i) {
124 int b = (int)(lru_mix(c->pool[i].key) & (LRU_BUCKETS - 1));
125 c->pool[i].hnext = c->bucket[b];
126 c->bucket[b] = i;
127}
128static void lru_hash_remove(LruCache *c, int i) {
129 int b = (int)(lru_mix(c->pool[i].key) & (LRU_BUCKETS - 1));
130 int *link = &c->bucket[b];
131 while (*link != LRU_NIL) {
132 if (*link == i) { *link = c->pool[i].hnext; return; }
133 link = &c->pool[*link].hnext;
134 }
135}

137/* 있으면 *out 에 값을 쓰고 MRU 로 올린 뒤 true. 없으면 false. 평균 O(1). */
138bool lru_get(LruCache *c, int key, int *out) {
139 if (!c || c->cap == 0) return false;
140 int i = lru_find(c, key);
141 if (i == LRU_NIL) return false;
142 lru_unlink(c, i);
143 lru_push_front(c, i);
144 if (out) *out = c->pool[i].val;
145 return true;
146}
147 
148/* 삽입/갱신. 가득 차면 tail(LRU)을 축출한다. 평균 O(1). */
149void lru_put(LruCache *c, int key, int val) {
150 if (!c || c->cap == 0) return;
151 int i = lru_find(c, key);
152 if (i != LRU_NIL) { /* 갱신도 "사용"이므로 MRU 로 */
153 c->pool[i].val = val;
154 lru_unlink(c, i);
155 lru_push_front(c, i);
156 return;
157 }
158 if (c->count == c->cap) { /* 축출: tail 부터 */
159 int victim = c->tail;
160 lru_hash_remove(c, victim);
161 lru_unlink(c, victim);
162 lru_release(c, victim);
163 c->count--;
164 c->evictions++;
165 }
166 i = lru_alloc(c);
167 if (i == LRU_NIL) return; /* 여기 도달하면 불변식 위반 */
168 c->pool[i].key = key;
169 c->pool[i].val = val;
170 lru_hash_insert(c, i);
171 lru_push_front(c, i);
172 c->count++;
173}

175/* 디버그/테스트용: MRU→LRU 순서로 키를 훑는다. O(count). */
176size_t lru_keys_mru_first(const LruCache *c, int *out, size_t out_cap) {
177 size_t n = 0;
178 if (!c || !out) return 0;
179 for (int i = c->head; i != LRU_NIL && n < out_cap; i = c->pool[i].next)
180 out[n++] = c->pool[i].key;
181 return n;
182}

185// Q70. 인터벌 병합 — 디바이스 오프라인 구간 합치기
186// ---------------------------------------------------------------------------
187// GW31-E 는 오프라인 30분이면 자가 재부팅한다. 여러 소스(heartbeat 누락, 모뎀 로그,
188// Command ping)가 겹치는 오프라인 구간을 보고하므로 하나로 합쳐야 총 다운타임이 나온다.
189// 경계가 맞닿는 구간([100,200],[200,260])도 끊긴 적이 없으므로 병합한다.
190// 시작 시각으로 정렬 후 한 번 훑는다: O(n log n) time, O(1) 추가 공간(in-place).
191// ===========================================================================
192typedef struct { long start, end; } Interval; /* [start, end] 닫힌 구간(초) */
193 
194static int iv_cmp_start(const void *pa, const void *pb) {
195 const Interval *a = (const Interval *)pa, *b = (const Interval *)pb;
196 if (a->start != b->start) return (a->start < b->start) ? -1 : 1;
197 if (a->end != b->end) return (a->end < b->end) ? -1 : 1;
198 return 0;
199}

201/* iv 를 in-place 로 정렬·병합하고 병합 후 개수를 반환한다. */
202size_t merge_intervals(Interval *iv, size_t n) {
203 if (!iv || n == 0) return 0;
204 qsort(iv, n, sizeof *iv, iv_cmp_start);
205 size_t w = 0; /* 쓰기 커서 = 현재 병합 중인 구간 */
206 for (size_t r = 1; r < n; r++) {
207 if (iv[r].start <= iv[w].end) { /* 겹치거나 맞닿음 → 흡수 */
208 if (iv[r].end > iv[w].end) iv[w].end = iv[r].end;
209 } else {
210 iv[++w] = iv[r]; /* 새 구간 시작 */
211 }
212 }
213 return w + 1;
214}

216/* 병합된 구간들의 총 길이(초). 닫힌 구간이지만 다운타임은 end-start 로 센다. */
217long merged_total_seconds(const Interval *iv, size_t n) {
218 long sum = 0;
219 for (size_t i = 0; i < n; i++) sum += iv[i].end - iv[i].start;
220 return sum;
221}

223// ===========================================================================
224// Q71. 겹치지 않게 만들기 — 제거해야 할 최소 인터벌 수
225// ---------------------------------------------------------------------------
226// 게이트웨이 유지보수 창(펌웨어 OTA, 채널 스캔)은 동시에 하나만 실행할 수 있다.
227// 요청된 창 목록에서 최소 개수만 취소해 겹침을 없애야 한다.
228// 그리디: "끝나는 시각"이 빠른 것부터 유지하면 남은 시간이 최대가 된다.
229// 여기서는 맞닿는 구간([1,2],[2,3])은 겹치지 않는 것으로 본다(Q70 과 반대 규약).
230// O(n log n) time, O(1) 추가 공간.
231// ===========================================================================
232static int iv_cmp_end(const void *pa, const void *pb) {
233 const Interval *a = (const Interval *)pa, *b = (const Interval *)pb;
234 if (a->end != b->end) return (a->end < b->end) ? -1 : 1;
235 if (a->start != b->start) return (a->start < b->start) ? -1 : 1;
236 return 0;
237}

239/* 겹침이 사라지도록 제거해야 하는 최소 인터벌 개수. iv 는 정렬되어 바뀐다. */
240size_t erase_overlap_intervals(Interval *iv, size_t n) {
241 if (!iv || n <= 1) return 0;
242 qsort(iv, n, sizeof *iv, iv_cmp_end);
243 size_t removed = 0;
244 long last_end = iv[0].end; /* 가장 빨리 끝나는 창을 무조건 유지 */
245 for (size_t i = 1; i < n; i++) {
246 if (iv[i].start < last_end) removed++; /* 겹침 → 끝이 더 늦은 이쪽을 버린다 */
247 else last_end = iv[i].end;
248 }
249 return removed;
250}

252// ===========================================================================
253// Q72. top-K 이벤트 — 크기 K 의 min-heap 으로 상위 K개만 유지
254// ---------------------------------------------------------------------------
255// fleet 대시보드는 "지금 가장 심각한 카메라 10대"만 보여준다. 이벤트가 수만 건이어도
256// 전체 정렬(O(n log n))은 낭비다. 크기 K 힙이면 O(n log K) time, O(K) space.
257// 힙 저장소로 호출자가 준 out[] 을 그대로 쓰므로 할당이 0이다.
258// 동점 처리 규칙: score 가 같으면 dev_id 가 작은 쪽이 상위(결정적 출력).
259// ===========================================================================
260typedef struct { uint32_t dev_id; int score; } Event;
261 
262/* a 가 b 보다 "상위"인가 (score 내림차순, 동점이면 dev_id 오름차순) */
263static bool ev_better(const Event *a, const Event *b) {
264 if (a->score != b->score) return a->score > b->score;
265 return a->dev_id < b->dev_id;
266}
267/* min-heap 불변식: 부모는 자식보다 상위가 아니다 → 루트가 "가장 낮은" 이벤트 */
268static void ev_sift_up(Event *h, size_t i) {
269 while (i > 0) {
270 size_t p = (i - 1) / 2;
271 if (!ev_better(&h[p], &h[i])) break; /* 부모가 더 낮으면 정상 */
272 Event t = h[i]; h[i] = h[p]; h[p] = t;
273 i = p;
274 }
275}
276static void ev_sift_down(Event *h, size_t n, size_t i) {
277 for (;;) {
278 size_t l = 2 * i + 1, r = 2 * i + 2, low = i;
279 if (l < n && ev_better(&h[low], &h[l])) low = l;
280 if (r < n && ev_better(&h[low], &h[r])) low = r;
281 if (low == i) break;
282 Event t = h[i]; h[i] = h[low]; h[low] = t;
283 i = low;
284 }
285}
286static int ev_cmp_desc(const void *pa, const void *pb) {
287 const Event *a = (const Event *)pa, *b = (const Event *)pb;
288 if (ev_better(a, b)) return -1;
289 if (ev_better(b, a)) return 1;
290 return 0;
291}

293/* 상위 min(n,k) 개를 out 에 내림차순으로 쓰고 개수를 반환. out 은 k 칸 이상. */
294size_t top_k_events(const Event *ev, size_t n, size_t k, Event *out) {
295 if (!ev || !out || k == 0 || n == 0) return 0;
296 size_t cnt = 0;
297 for (size_t i = 0; i < n; i++) {
298 if (cnt < k) { /* 힙이 덜 찼으면 그냥 push */
299 out[cnt] = ev[i];
300 ev_sift_up(out, cnt);
301 cnt++;
302 } else if (ev_better(&ev[i], &out[0])) { /* 루트(최하위)보다 나으면 교체 */
303 out[0] = ev[i];
304 ev_sift_down(out, cnt, 0);
305 }
306 }
307 qsort(out, cnt, sizeof *out, ev_cmp_desc); /* 보고용으로 상위부터 정렬 */
308 return cnt;
309}

335static uint32_t fnv1a(const char *s) {
336 uint32_t h = 2166136261u;
337 while (*s) { h ^= (uint8_t)*s++; h *= 16777619u; }
338 return h;
339}

341/* 키를 찾으면 (found=true) 그 슬롯을, 없으면 (found=false) 삽입할 빈 슬롯을 반환.
342 테이블이 포화면 AGG_SLOTS 를 반환한다. 읽기 전용이라 get/record 가 공유한다. */
343static size_t agg_probe(const AggSlot *slots, const char *dev, bool *found) {
344 size_t i = (size_t)fnv1a(dev) & (AGG_SLOTS - 1);
345 for (size_t p = 0; p < AGG_SLOTS; p++) {
346 size_t j = (i + p) & (AGG_SLOTS - 1);
347 if (!slots[j].used) { *found = false; return j; }
348 if (strcmp(slots[j].dev, dev) == 0) { *found = true; return j; }
349 }
350 *found = false;
351 return AGG_SLOTS; /* 포화 */
352}

354void agg_init(Aggregator *a) { if (a) memset(a, 0, sizeof *a); }

356/* 이벤트 1건 반영. 성공 true, 버리면 false(dropped++). */
357bool agg_record(Aggregator *a, const char *dev, DevState st) {
358 if (!a || !dev) return false;
359 if (st < 0 || st >= DEV_STATE_COUNT) { a->dropped++; return false; }
360 size_t len = strlen(dev);
361 if (len == 0 || len >= AGG_DEV_LEN) { a->dropped++; return false; } /* 잘림 금지 */
362 
363 bool found = false;
364 size_t j = agg_probe(a->slot, dev, &found);
365 if (j == AGG_SLOTS) { a->dropped++; return false; }
366 if (!found) {
367 memcpy(a->slot[j].dev, dev, len + 1);
368 a->slot[j].used = true;
369 a->n_used++;
370 }
371 a->slot[j].counts[st]++;
372 return true;
373}

375/* 디바이스별 카운트 조회. 없으면 false. */
376bool agg_get(const Aggregator *a, const char *dev, uint32_t out[DEV_STATE_COUNT]) {
377 if (!a || !dev) return false;
378 bool found = false;
379 size_t j = agg_probe(a->slot, dev, &found);
380 if (!found || j == AGG_SLOTS) return false;
381 if (out) for (int s = 0; s < DEV_STATE_COUNT; s++) out[s] = a->slot[j].counts[s];
382 return true;
383}

385// ===========================================================================
386// Q74. 토큰 버킷 rate limiter — 업로드/알람 폭주 억제
387// ---------------------------------------------------------------------------
388// 링크가 살아날 때 큐에 쌓인 알람이 한꺼번에 터지면 LTE 업링크를 잡아먹는다.
389// 토큰 버킷은 "평균 R/초, 순간 최대 C개"를 한 줄로 표현한다.
390// 부동소수점 없이 milli-token(1/1000 토큰) 정수로 계산 → MCU 에서도 그대로 쓴다.
391// 시간을 인자로 주입(now_ms)해 테스트가 sleep 없이 결정적이다 — modular/testable 설계.
392// O(1) time, O(1) space.
393// ===========================================================================
394typedef struct {
395 uint64_t tokens_milli; /* 현재 토큰 * 1000 */
396 uint64_t cap_milli; /* 버스트 상한 * 1000 */
397 uint32_t refill_per_sec; /* 초당 충전 토큰 수 */
398 uint64_t last_ms;
399 uint64_t allowed, rejected;
400} TokenBucket;
401 
402void tb_init(TokenBucket *tb, uint32_t capacity, uint32_t refill_per_sec, uint64_t now_ms) {
403 if (!tb) return;
404 memset(tb, 0, sizeof *tb);
405 tb->cap_milli = (uint64_t)capacity * 1000u;
406 tb->tokens_milli = tb->cap_milli; /* 처음엔 가득 = 즉시 버스트 허용 */
407 tb->refill_per_sec = refill_per_sec;
408 tb->last_ms = now_ms;
409}
410 
411/* 경과 시간만큼 충전. 1ms 당 refill_per_sec milli-token (= rate/1000 토큰). */
412static void tb_refill(TokenBucket *tb, uint64_t now_ms) {
413 if (now_ms <= tb->last_ms) { tb->last_ms = now_ms; return; } /* 시계 역행 → 재동기화만 */
414 uint64_t elapsed = now_ms - tb->last_ms;
415 tb->tokens_milli += elapsed * (uint64_t)tb->refill_per_sec;
416 if (tb->tokens_milli > tb->cap_milli) tb->tokens_milli = tb->cap_milli;
417 tb->last_ms = now_ms;
418}
419 
420/* cost 개의 토큰을 요구한다. 있으면 소비하고 true, 없으면 false(rejected++). */
421bool tb_allow(TokenBucket *tb, uint64_t now_ms, uint32_t cost) {
422 if (!tb) return false;
423 tb_refill(tb, now_ms);
424 uint64_t need = (uint64_t)cost * 1000u;
425 if (tb->tokens_milli < need) { tb->rejected++; return false; }
426 tb->tokens_milli -= need;
427 tb->allowed++;
428 return true;
429}
430 
431/* 현재 사용 가능한 (내림) 토큰 수 — 테스트/텔레메트리용. */
432uint32_t tb_available(const TokenBucket *tb) {
433 return tb ? (uint32_t)(tb->tokens_milli / 1000u) : 0u;
434}

438// ---------------------------------------------------------------------------
439// 게이트웨이 로컬 로그와 카메라 업로드 로그를 시간순 하나의 뷰로 합친다("Camera Logs API").
440// 둘 다 이미 정렬되어 있으므로 다시 정렬하지 말고 two-pointer 로 한 번에 훑는다.
441// 안정성(stable): 타임스탬프가 같으면 스트림 A 를 먼저 — 재실행해도 같은 순서가 나온다.
442// out_cap 을 넘으면 잘라내고 쓴 개수만 반환(고정 버퍼 안전).
443// O(na + nb) time, O(1) 추가 공간.
444// ===========================================================================
445typedef struct { uint64_t ts; uint32_t dev; int code; } LogRec;

447size_t merge_logs(const LogRec *a, size_t na, const LogRec *b, size_t nb,
448 LogRec *out, size_t out_cap) {
449 if (!out || out_cap == 0) return 0;
450 if (!a) na = 0;
451 if (!b) nb = 0;
452 size_t i = 0, j = 0, w = 0;
453 while (i < na && j < nb && w < out_cap) {
454 if (b[j].ts < a[i].ts) out[w++] = b[j++]; /* 동점(==)이면 a 우선 = stable */
455 else out[w++] = a[i++];
456 }
457 while (i < na && w < out_cap) out[w++] = a[i++];
458 while (j < nb && w < out_cap) out[w++] = b[j++];
459 return w;
460}

462// ===========================================================================
463// Q76. 최근 N개 기반 중복 제거 — 같은 알람이 반복될 때 억제
464// ---------------------------------------------------------------------------
465// 링크 플래핑 중이면 "device offline" 알람이 초당 수십 번 올라온다. 최근 N개의
466// 서로 다른 알람 키만 기억해 그 안에 있으면 억제한다(N 개를 지나면 다시 통과).
467// 순환 버퍼(FIFO 창) + 해시(멤버십 O(1))의 조합. 링 슬롯 자체가 해시 체인 노드라
468// 할당이 0이고, 창에서 밀려난 항목은 해시에서도 정확히 제거된다.
469// 평균 O(1) time, O(N) space.
470// ===========================================================================
471#define DEDUP_WINDOW 8 /* 최근 몇 개의 서로 다른 알람을 기억하나 */
472#define DEDUP_SLOTS 32 /* 2의 거듭제곱, 여유롭게 */
473#define DEDUP_NIL (-1)
474 
475typedef struct { uint32_t key; int hnext; bool used; } DedupEntry;
476typedef struct {
477 DedupEntry ring[DEDUP_WINDOW];
478 int bucket[DEDUP_SLOTS];
479 size_t pos; /* 다음에 덮어쓸 링 슬롯 */
480 uint64_t emitted, suppressed;
481} Dedup;
482 
483static uint32_t dd_mix(uint32_t x) { x *= 2654435761u; return x ^ (x >> 16); }
484 
485static void dd_bucket_remove(Dedup *d, int idx) {
486 int b = (int)(dd_mix(d->ring[idx].key) & (DEDUP_SLOTS - 1));
487 int *link = &d->bucket[b];
488 while (*link != DEDUP_NIL) {
489 if (*link == idx) { *link = d->ring[idx].hnext; return; }
490 link = &d->ring[*link].hnext;
491 }
492}
493 
494void dedup_init(Dedup *d) {
495 if (!d) return;
496 memset(d, 0, sizeof *d);
497 for (int i = 0; i < DEDUP_SLOTS; i++) d->bucket[i] = DEDUP_NIL;
498 for (int i = 0; i < DEDUP_WINDOW; i++) { d->ring[i].used = false; d->ring[i].hnext = DEDUP_NIL; }
499}
500 
501/* 최근 창에 없으면 등록하고 true(=내보낸다). 있으면 false(=억제). */
502bool dedup_should_emit(Dedup *d, uint32_t key) {
503 if (!d) return false;
504 int b = (int)(dd_mix(key) & (DEDUP_SLOTS - 1));
505 for (int i = d->bucket[b]; i != DEDUP_NIL; i = d->ring[i].hnext) {
506 if (d->ring[i].key == key) { d->suppressed++; return false; }
507 }
508 int slot = (int)d->pos;
509 if (d->ring[slot].used) dd_bucket_remove(d, slot); /* 창 밖으로 밀려난 키 제거 */
510 d->ring[slot].key = key;
511 d->ring[slot].used = true;
512 d->ring[slot].hnext = d->bucket[b];
513 d->bucket[b] = slot;
514 d->pos = (d->pos + 1) % DEDUP_WINDOW;
515 d->emitted++;
516 return true;
517}

519// ===========================================================================
520// Q77. 시간 범위 쿼리 — 정렬된 타임스탬프 배열에서 구간 추출
521// ---------------------------------------------------------------------------
522// "지난 10분의 로그를 줘" — 로그는 append-only 라 이미 시간순이다. 선형 스캔 대신
523// lower_bound(첫 >= lo) / upper_bound(첫 > hi) 로 O(log n) 에 구간을 잘라낸다.
524// 중복 타임스탬프가 있어도 두 경계가 구간 전체를 정확히 감싼다.
525// O(log n) time, O(1) space.
526// ===========================================================================
527 
528/* ts[0..n) 중 key 이상인 첫 인덱스 (없으면 n) */
529size_t ts_lower_bound(const uint64_t *ts, size_t n, uint64_t key) {
530 size_t lo = 0, hi = n; /* 답은 항상 [lo, hi] 안에 있다 */
531 while (lo < hi) {
532 size_t mid = lo + (hi - lo) / 2; /* 오버플로 없는 중점 */
533 if (ts[mid] < key) lo = mid + 1;
534 else hi = mid;
535 }
536 return lo;
537}
538 
539/* ts[0..n) 중 key 초과인 첫 인덱스 (없으면 n) */
540size_t ts_upper_bound(const uint64_t *ts, size_t n, uint64_t key) {
541 size_t lo = 0, hi = n;
542 while (lo < hi) {
543 size_t mid = lo + (hi - lo) / 2;
544 if (ts[mid] <= key) lo = mid + 1;
545 else hi = mid;
546 }
547 return lo;
548}
549 
550/* [lo, hi] 닫힌 구간에 드는 원소 개수. *out_begin 에 시작 인덱스. */
551size_t ts_range(const uint64_t *ts, size_t n, uint64_t lo, uint64_t hi, size_t *out_begin) {
552 if (out_begin) *out_begin = 0;
553 if (!ts || n == 0 || lo > hi) return 0;
554 size_t b = ts_lower_bound(ts, n, lo);
555 size_t e = ts_upper_bound(ts, n, hi);
556 if (out_begin) *out_begin = b;
557 return (e > b) ? (e - b) : 0;
558}

561// Q78. 로그 라인 파싱 — "ts=... dev=... state=... rssi=..." 를 구조체로
562// ---------------------------------------------------------------------------
563// 모뎀/에이전트가 뱉는 key=value 한 줄을 구조체로 바꾼다. 입력은 신뢰할 수 없다 —
564// 필드 누락, 모르는 키, 중복 키, 숫자가 아닌 값, 범위를 벗어난 RSSI, 너무 긴 dev 는
565// 전부 거부한다(부분 파싱 결과를 out 에 남기지 않는다: 임시 구조체에 채운 뒤 커밋).
566// 입력 버퍼를 변형하지 않도록 strtok_r 대신 const 포인터 수동 스캔을 쓴다.
567// O(L) time (L = 줄 길이), O(1) space.
568// ===========================================================================
569#define LOG_DEV_LEN 16
570 
571typedef struct {
572 uint64_t ts;
573 char dev[LOG_DEV_LEN];
574 DevState state;
575 int rssi;
576} LogLine;

578static const char *skip_ws(const char *p) {
579 while (*p == ' ' || *p == '\t') p++;
580 return p;
581}
582static bool key_is(const char *s, const char *e, const char *lit) {
583 size_t len = (size_t)(e - s);
584 return len == strlen(lit) && memcmp(s, lit, len) == 0;
585}
586/* [s,e) 를 부호 없는 10진수로. 빈 문자열/비숫자/오버플로는 거부. */
587static bool parse_u64_range(const char *s, const char *e, uint64_t *out) {
588 if (s >= e) return false;
589 uint64_t v = 0;
590 for (const char *p = s; p < e; p++) {
591 if (*p < '0' || *p > '9') return false;
592 uint64_t d = (uint64_t)(*p - '0');
593 if (v > (UINT64_MAX - d) / 10u) return false;
594 v = v * 10u + d;
595 }
596 *out = v;
597 return true;
598}
599/* [s,e) 를 부호 있는 32비트 정수로. */
600static bool parse_i32_range(const char *s, const char *e, int *out) {
601 bool neg = false;
602 if (s < e && (*s == '-' || *s == '+')) { neg = (*s == '-'); s++; }
603 uint64_t mag;
604 if (!parse_u64_range(s, e, &mag)) return false;
605 if (mag > 2147483647u) return false;
606 *out = neg ? -(int)mag : (int)mag;
607 return true;
608}
609static bool parse_state(const char *s, const char *e, DevState *out) {
610 static const struct { const char *name; DevState st; } tbl[] = {
611 { "online", DEV_ONLINE },
612 { "offline", DEV_OFFLINE },
613 { "degraded", DEV_DEGRADED },
614 };
615 for (size_t i = 0; i < sizeof tbl / sizeof tbl[0]; i++) {
616 if (key_is(s, e, tbl[i].name)) { *out = tbl[i].st; return true; }
617 }
618 return false;
619}
620 
621/* 성공하면 *out 을 채우고 true. 한 군데라도 이상하면 out 을 건드리지 않고 false. */
622bool parse_log_line(const char *line, LogLine *out) {
623 if (!line || !out) return false;
624 LogLine tmp;
625 memset(&tmp, 0, sizeof tmp);
626 unsigned seen = 0; /* ts=1 dev=2 state=4 rssi=8 */
627 
628 const char *p = skip_ws(line);
629 while (*p) {
630 const char *ks = p;
631 while (*p && *p != ' ' && *p != '\t' && *p != '=') p++;
632 if (*p != '=') return false; /* '=' 없는 토큰 */
633 const char *ke = p++; /* '=' 를 건너뛴다 */
634 const char *vs = p;
635 while (*p && *p != ' ' && *p != '\t') p++;
636 const char *ve = p;
637 if (vs == ve) return false; /* 값이 비었다 */
638 
639 if (key_is(ks, ke, "ts")) {
640 if (seen & 1u) return false; /* 중복 키 */
641 if (!parse_u64_range(vs, ve, &tmp.ts)) return false;
642 seen |= 1u;
643 } else if (key_is(ks, ke, "dev")) {
644 if (seen & 2u) return false;
645 size_t len = (size_t)(ve - vs);
646 if (len == 0 || len >= LOG_DEV_LEN) return false; /* 잘림 대신 거부 */
647 memcpy(tmp.dev, vs, len);
648 tmp.dev[len] = '\0';
649 seen |= 2u;
650 } else if (key_is(ks, ke, "state")) {
651 if (seen & 4u) return false;
652 if (!parse_state(vs, ve, &tmp.state)) return false;
653 seen |= 4u;
654 } else if (key_is(ks, ke, "rssi")) {
655 if (seen & 8u) return false;
656 if (!parse_i32_range(vs, ve, &tmp.rssi)) return false;
657 if (tmp.rssi > 0 || tmp.rssi < -120) return false; /* dBm 범위 검증 */
658 seen |= 8u;
659 } else {
660 return false; /* 모르는 키 */
661 }
662 p = skip_ws(p);
663 }
664 if (seen != 0xFu) return false; /* 네 필드 모두 필수 */
665 *out = tmp; /* 전부 통과한 뒤에만 커밋 */
666 return true;
667}
