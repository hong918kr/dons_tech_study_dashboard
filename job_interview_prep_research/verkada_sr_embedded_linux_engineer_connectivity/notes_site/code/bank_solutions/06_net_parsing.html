34// ===========================================================================
35// Q49. IPv4 문자열 -> uint32_t (엄격 파서)
36// ---------------------------------------------------------------------------
37// "10.0.0.7" -> 0x0A000007 (host order 정수, 상위 바이트 = 첫 옥텟).
38// 엄격하다는 것은 다음을 전부 거부한다는 뜻이다:
39// 선행 0("01.2.3.4"), 범위 초과("256.1.1.1"), 점 개수 오류("1.2.3" / "1.2.3.4.5"),
40// 빈 옥텟("1..2.3"), 공백/꼬리 쓰레기(" 1.2.3.4" / "1.2.3.4x").
41// 왜 엄격해야 하나: 선행 0을 허용하는 구현(inet_aton 계열)은 8진수로 해석해
42// "010.1.1.1" 을 10.1.1.1 이 아닌 8.1.1.1 로 만든다 — SSRF/ACL 우회의 고전 취약점.
43// ===========================================================================
44#define IPV4_STR_MAX 16 // "255.255.255.255" + '\0'
45 
46bool ipv4_parse(const char *s, uint32_t *out) {
47 if (!s || !out) return false;
48 
49 uint32_t ip = 0;
50 int octets = 0;
51 const char *p = s;
52 
53 for (;;) {
54 if (!is_dig(*p)) return false; // 빈 옥텟 / 비숫자 / 선행 공백 거부
55 if (*p == '0' && is_dig(p[1])) return false; // 선행 0 거부 ("0" 자체는 허용)
56 
57 unsigned v = 0;
58 int digits = 0;
59 while (is_dig(*p)) {
60 v = v * 10u + (unsigned)(*p - '0');
61 if (++digits > 3 || v > 255u) return false; // 범위 초과 즉시 탈출
62 p++;
63 }
64 ip = (ip << 8) | v;
65 if (++octets > 4) return false;
66 
67 if (*p == '.') {
68 if (octets == 4) return false; // "1.2.3.4.5"
69 p++;
70 continue;
71 }
72 break;
73 }
74 if (octets != 4 || *p != '\0') return false; // "1.2.3", "1.2.3.4x"
75 *out = ip;
76 return true;
77}

80// Q50. uint32_t -> IPv4 문자열 (버퍼 크기 안전)
81// ---------------------------------------------------------------------------
82// 성공 시 기록한 문자 수(널 제외)를, 버퍼 부족/NULL 이면 -1 을 반환한다.
83// 핵심: (1) 먼저 로컬 tmp 에 만들고 크기를 확인한 뒤 복사 — 부분 기록 금지.
84// (2) 실패해도 out 은 항상 널 종료시켜 호출자가 쓰레기를 읽지 않게 한다.
85// GC31-E 의 진단 로그·Command 업링크 JSON 이 이 함수를 초당 수십 번 호출한다.
86// ===========================================================================
87static size_t u8_to_dec(uint8_t v, char *dst) { // dst 최소 3바이트, 널 미기록
88 size_t k = 0;
89 if (v >= 100) dst[k++] = (char)('0' + v / 100);
90 if (v >= 10) dst[k++] = (char)('0' + (v / 10) % 10);
91 dst[k++] = (char)('0' + v % 10);
92 return k;
93}

95int ipv4_format(uint32_t ip, char *out, size_t out_sz) {
96 if (!out || out_sz == 0) return -1;
97 
98 char tmp[IPV4_STR_MAX];
99 size_t k = 0;
100 for (int i = 3; i >= 0; i--) {
101 k += u8_to_dec((uint8_t)((ip >> (i * 8)) & 0xFFu), tmp + k);
102 if (i) tmp[k++] = '.';
103 }
104 tmp[k] = '\0';
105 
106 if (k + 1 > out_sz) { out[0] = '\0'; return -1; } // 잘라 쓰지 않는다
107 memcpy(out, tmp, k + 1);
108 return (int)k;
109}

111// ===========================================================================
112// Q51. CIDR 매칭 — "10.0.0.0/8" 안에 특정 IP가 들어가는가
113// ---------------------------------------------------------------------------
114// mask = 상위 prefix 비트만 1. 함정은 prefix==0 — C에서 32비트 값을 32비트
115// 시프트하는 것은 **정의되지 않은 동작(UB)** 이라 반드시 특수 처리해야 한다.
116// Verkada 맥락: 게이트웨이가 LAN(사설 대역)과 WAN(CGNAT 100.64/10, 공인)을
117// 구분하고, 카메라를 어느 서브넷으로 내보낼지 라우팅 판단할 때 그대로 쓰인다.
118// ===========================================================================
119uint32_t cidr_mask(unsigned prefix) {
120 if (prefix == 0) return 0u; // 0xFFFFFFFF << 32 는 UB
121 if (prefix >= 32) return 0xFFFFFFFFu;
122 return 0xFFFFFFFFu << (32u - prefix);
123}

125// "a.b.c.d/p" 를 파싱. net_out 에는 마스크가 적용된 네트워크 주소를 넣는다.
126bool cidr_parse(const char *s, uint32_t *net_out, unsigned *prefix_out) {
127 if (!s || !net_out || !prefix_out) return false;
128 
129 const char *slash = strchr(s, '/');
130 if (!slash) return false; // prefix 는 필수
131 
132 size_t iplen = (size_t)(slash - s);
133 if (iplen == 0 || iplen >= IPV4_STR_MAX) return false;
134 
135 char ipbuf[IPV4_STR_MAX];
136 memcpy(ipbuf, s, iplen);
137 ipbuf[iplen] = '\0';
138 
139 uint32_t ip;
140 if (!ipv4_parse(ipbuf, &ip)) return false; // 옥텟 규칙은 Q49 를 재사용
141 
142 const char *q = slash + 1;
143 if (!is_dig(*q)) return false; // "10.0.0.0/" 거부
144 if (*q == '0' && is_dig(q[1])) return false; // "/08" 거부
145 unsigned pfx = 0;
146 int digits = 0;
147 while (is_dig(*q)) {
148 pfx = pfx * 10u + (unsigned)(*q - '0');
149 if (++digits > 2 || pfx > 32u) return false;
150 q++;
151 }
152 if (*q != '\0') return false; // 꼬리 쓰레기 거부
153 
154 *prefix_out = pfx;
155 *net_out = ip & cidr_mask(pfx);
156 return true;
157}

159bool cidr_contains(const char *cidr, uint32_t ip) {
160 uint32_t net;
161 unsigned pfx;
162 if (!cidr_parse(cidr, &net, &pfx)) return false;
163 uint32_t m = cidr_mask(pfx);
164 return (ip & m) == net;
165}

168// Q52. 로그 한 줄에서 IPv4 주소 모두 추출 ★ 실제 보고된 면접 문제 유형
169// ---------------------------------------------------------------------------
170// 반환값 = 찾은 총 개수(호출자가 잘림 여부를 알 수 있게). 실제로 out 에 쓰는 것은
171// 최대 max_out 개. 경계 규칙:
172// - [0-9.] 덩어리를 통째로 잘라 Q49 의 엄격 파서에 넘긴다 -> "1.2.3.4.5" 자동 탈락.
173// - 덩어리 양옆이 영숫자면 거부 -> "v1.2.3.4", "1.2.3.4a" 같은 버전 문자열 오탐 차단.
174// - 뒤가 ':' '/' ',' 공백 등 구분자면 허용 -> "192.168.0.1:8080", "10.0.0.0/8" OK.
175// ===========================================================================
176size_t ipv4_extract_all(const char *line, uint32_t *out, size_t max_out) {
177 if (!line) return 0;
178 
179 size_t found = 0;
180 size_t i = 0;
181 while (line[i] != '\0') {
182 if (!is_dig(line[i])) { i++; continue; }
183 
184 size_t j = i;
185 while (line[j] != '\0' && (is_dig(line[j]) || line[j] == '.')) j++;
186 
187 bool left_ok = (i == 0) || !is_alnum_c(line[i - 1]);
188 bool right_ok = (line[j] == '\0') || !is_alnum_c(line[j]);
189 size_t len = j - i;
190 
191 if (left_ok && right_ok && len < IPV4_STR_MAX) {
192 char tok[IPV4_STR_MAX];
193 memcpy(tok, line + i, len);
194 tok[len] = '\0';
195 uint32_t ip;
196 if (ipv4_parse(tok, &ip)) {
197 if (out && found < max_out) out[found] = ip;
198 found++;
199 }
200 }
201 i = j; // 덩어리 전체를 소비 — 안쪽에서 다시 시작하지 않는다
202 }
203 return found;
204}

206// ===========================================================================
207// Q53. 인터넷 체크섬 (RFC 1071) — 16비트 1의 보수 합의 1의 보수
208// ---------------------------------------------------------------------------
209// 절차: 버퍼를 빅엔디안 16비트 워드로 더하고, 넘친 캐리를 하위로 접어 넣고(fold),
210// 마지막에 비트 반전. 홀수 길이면 마지막 바이트를 상위 바이트로 패딩.
211// 두 가지 성질을 반드시 말할 수 있어야 한다:
212// (1) 엔디안 독립 — 바이트를 직접 (p[0]<<8)|p[1] 로 조립하므로 호스트 엔디안 무관.
213// (2) 검증 성질 — 체크섬 필드에 결과를 넣고 다시 계산하면 0 이 나온다.
214// IP/UDP/ICMP 헤더가 이 알고리즘을 쓴다. 약점: 자리바꿈(transposition)을 못 잡는다.
215// ===========================================================================
216uint16_t inet_checksum(const void *data, size_t len) {
217 const uint8_t *p = (const uint8_t *)data;
218 uint32_t sum = 0;
219 
220 while (len > 1) {
221 sum += ((uint32_t)p[0] << 8) | (uint32_t)p[1];
222 p += 2;
223 len -= 2;
224 }
225 if (len == 1) sum += (uint32_t)p[0] << 8; // 홀수 바이트는 상위로 패딩
226 
227 while (sum >> 16) sum = (sum & 0xFFFFu) + (sum >> 16); // 캐리 fold
228 return (uint16_t)(~sum & 0xFFFFu);
229}

232static void be16_put(uint8_t *dst, uint16_t v) {
233 dst[0] = (uint8_t)(v >> 8);
234 dst[1] = (uint8_t)(v & 0xFFu);
235}

238// Q54. hex dump 유틸 — "offset | hex | ascii" 한 줄
239// ---------------------------------------------------------------------------
240// 형식(= hexdump -C 호환):
241// "00000010 00 01 02 03 04 05 06 07 08 09 0a 0b 0c 0d 0e 0f |................|"
242// ^8자리 오프셋 ^2칸 ^8바이트 ^2칸 ^8바이트 ^2칸 ^ASCII
243// len<16 이면 hex 칸은 공백으로 패딩하고 ASCII 는 있는 만큼만 찍는다.
244// 필드 디버깅 단골 — 모뎀이 뱉은 이상한 바이트를 눈으로 보는 유일한 방법이다.
245// ===========================================================================
246#define HEXDUMP_BYTES_PER_LINE 16
247#define HEXDUMP_LINE_MAX 80 // 60(고정폭) + 1 + 16 + 1 + 1(널)
248 
249int hexdump_line(char *out, size_t out_sz,
250 size_t offset, const uint8_t *data, size_t len) {
251 if (!out || out_sz == 0) return -1;
252 if (len > HEXDUMP_BYTES_PER_LINE) { out[0] = '\0'; return -1; }
253 if (len > 0 && !data) { out[0] = '\0'; return -1; }
254 
255 static const char HEX[] = "0123456789abcdef";
256 char tmp[HEXDUMP_LINE_MAX];
257 size_t k = 0;
258 
259 for (int sh = 28; sh >= 0; sh -= 4) // 오프셋 8자리 (하위 32비트)
260 tmp[k++] = HEX[(offset >> sh) & 0xFu];
261 tmp[k++] = ' ';
262 tmp[k++] = ' ';
263 
264 for (size_t c = 0; c < HEXDUMP_BYTES_PER_LINE; c++) {
265 if (c < len) {
266 tmp[k++] = HEX[(data[c] >> 4) & 0xFu];
267 tmp[k++] = HEX[data[c] & 0xFu];
268 } else {
269 tmp[k++] = ' ';
270 tmp[k++] = ' ';
271 }
272 tmp[k++] = ' ';
273 if (c == 7) tmp[k++] = ' '; // 8바이트마다 한 칸 더
274 }
275 tmp[k++] = ' '; // hex 영역과 ASCII 영역 구분
276 
277 tmp[k++] = '|';
278 for (size_t c = 0; c < len; c++) {
279 uint8_t b = data[c];
280 tmp[k++] = (b >= 0x20 && b <= 0x7E) ? (char)b : '.';
281 }
282 tmp[k++] = '|';
283 tmp[k] = '\0';
284 
285 if (k + 1 > out_sz) { out[0] = '\0'; return -1; }
286 memcpy(out, tmp, k + 1);
287 return (int)k;
288}

291// Q55. AT 커맨드 응답 파서 FSM — 스트림 재조립 + 분류
292// ---------------------------------------------------------------------------
293// 모뎀은 UART로 바이트를 흘려보낼 뿐, "줄" 단위로 주지 않는다. read()가
294// "+CSQ: 2" / "3,99\r\nOK\r" / "\n" 처럼 아무데서나 잘려 들어와도 동작해야 한다.
295// 그래서 파서는 **호출 사이에 상태를 들고 있는 FSM**이어야 한다.
296// - '\r' 또는 '\n' 이 줄 종결자. 빈 줄(연속 CRLF)은 버린다.
297// - 줄이 버퍼보다 길면 AT_KIND_OVERFLOW 로 보고하고 **다음 종결자까지 버린 뒤 복구**.
298// - 분류: "OK" / "ERROR"·"+CME ERROR:"·"+CMS ERROR:" / '+'로 시작 = URC / 그 외 = INFO
299// 반환값 = 이번 feed 로 완성된 줄의 총 개수(out 에는 최대 max_out 개만 기록).
300// ===========================================================================
301#define AT_LINE_MAX 64

303typedef enum {
304 AT_KIND_OK, // 최종 응답 성공
305 AT_KIND_ERROR, // 최종 응답 실패 (ERROR / +CME ERROR / +CMS ERROR)
306 AT_KIND_URC, // '+' 로 시작하는 정보/비동기 통지 (+CSQ, +CREG, ...)
307 AT_KIND_INFO, // 에코나 그 밖의 텍스트

342// 완성된 줄 하나를 방출. 기록 여부와 무관하게 개수는 센다.
343static void at_emit(at_parser_t *p, at_line_t *out, size_t max_out, size_t *count) {
344 if (p->len == 0 && !p->overflow) return; // 빈 줄은 무시 (CRLF 연속)
345 
346 p->buf[p->len] = '\0';
347 if (out && *count < max_out) {
348 at_line_t *L = &out[*count];
349 L->kind = p->overflow ? AT_KIND_OVERFLOW : at_classify(p->buf);
350 memcpy(L->text, p->buf, p->len + 1);
351 }
352 (*count)++;
353 p->len = 0;
354 p->overflow = false; // 종결자를 봤으니 복구
355}

357size_t at_feed(at_parser_t *p, const char *chunk, size_t n,
358 at_line_t *out, size_t max_out) {
359 if (!p || (!chunk && n > 0)) return 0;
360 
361 size_t count = 0;
362 for (size_t i = 0; i < n; i++) {
363 char c = chunk[i];
364 if (c == '\r' || c == '\n') {
365 at_emit(p, out, max_out, &count);
366 continue;
367 }
368 if (p->len + 1 >= AT_LINE_MAX) { // 널 자리 확보
369 p->overflow = true; // 이후 바이트는 버리고 종결자만 기다림
370 continue;
371 }
372 if (!p->overflow) p->buf[p->len++] = c;
373 }
374 return count;
375}

394int sig_clamp(int v, int lo, int hi) {
395 if (v < lo) return lo;
396 if (v > hi) return hi;
397 return v;

400// "+CSQ: 23,99" -> rssi=23, ber=99. 공백 허용, 형식이 틀리면 false.
401bool at_parse_csq(const char *line, int *rssi_out, int *ber_out) {
402 if (!line || !rssi_out || !ber_out) return false;
403 if (!starts_with(line, "+CSQ:")) return false;
404 
405 const char *p = line + 5;
406 int vals[2];
407 for (int f = 0; f < 2; f++) {
408 while (*p == ' ') p++;
409 if (!is_dig(*p)) return false;
410 int v = 0, digits = 0;
411 while (is_dig(*p)) {
412 v = v * 10 + (*p - '0');
413 if (++digits > 3) return false; // 3자리 초과는 스펙 위반
414 p++;
415 }
416 vals[f] = v;
417 if (f == 0) {
418 while (*p == ' ') p++;
419 if (*p != ',') return false;
420 p++;
421 }
422 }
423 while (*p == ' ') p++;
424 if (*p != '\0') return false; // 꼬리 쓰레기 거부
425 
426 *rssi_out = vals[0];
427 *ber_out = vals[1];
428 return true;
429}

435 *dbm_out = -113 + 2 * csq;
436 return true;
437}

466// ===========================================================================
467// Q57. 길이-접두(TLV) 프레이밍 디코더 — 부분 버퍼 / 잘린 헤더 / 악의적 길이
468// ---------------------------------------------------------------------------
469// 프레임: [type:1][len:2 big-endian][payload:len]
470// 스트림 디코더의 3대 계약:
471// (1) 데이터가 모자라면 TLV_NEED_MORE 를 주고 **오프셋을 전진시키지 않는다**
472// (호출자가 남은 바이트를 보관했다가 다음 read 와 이어붙일 수 있어야 하므로).
473// (2) len 이 정책 상한(max_len)을 넘으면 TLV_BAD — 공격자가 0xFFFF 를 보내
474// 수십 KB 할당/대기를 유발하는 것을 막는다.
475// (3) off + 3 + len 계산에서 **절대 오버플로가 나면 안 된다** — 덧셈 대신 뺄셈으로 비교.
476// ===========================================================================
477typedef enum {
478 TLV_OK, // 프레임 하나를 온전히 꺼냄, *off 전진
479 TLV_NEED_MORE, // 바이트 부족 — *off 그대로, 더 받아서 재시도
480 TLV_BAD // 프로토콜 위반 — 재동기화/연결 리셋 필요
481} tlv_status_t;
482 
483typedef struct {
484 uint8_t type;
485 uint16_t len;
486 const uint8_t *val; // buf 안을 가리키는 포인터 (복사 없음, zero-copy)
487} tlv_t;
488 
489#define TLV_HDR_LEN 3
490 
491tlv_status_t tlv_next(const uint8_t *buf, size_t n, size_t *off,
492 tlv_t *out, uint16_t max_len) {
493 if (!buf || !off || !out) return TLV_BAD;
494 if (*off > n) return TLV_BAD;
495 
496 size_t avail = n - *off; // 뺄셈 — 오버플로 없음
497 if (avail < TLV_HDR_LEN) return TLV_NEED_MORE; // 헤더조차 안 왔다
498 
499 const uint8_t *h = buf + *off;
500 uint8_t type = h[0];
501 uint16_t len = (uint16_t)(((uint16_t)h[1] << 8) | (uint16_t)h[2]); // BE
502 
503 if (len > max_len) return TLV_BAD; // 악의적/손상된 길이 방어
504 
505 if (avail - TLV_HDR_LEN < (size_t)len) return TLV_NEED_MORE; // 페이로드 미도착
506 
507 out->type = type;
508 out->len = len;
509 out->val = (len > 0) ? (h + TLV_HDR_LEN) : NULL;
510 *off += TLV_HDR_LEN + (size_t)len;
511 return TLV_OK;
512}

514// ===========================================================================
515// Q58. SLIP 바이트 스터핑 (RFC 1055) 인코드 / 디코드
516// ---------------------------------------------------------------------------
517// END = 0xC0 프레임 경계
518// ESC = 0xDB 이스케이프 도입
519// ESC_END = 0xDC "원래 0xC0 이었음"
520// ESC_ESC = 0xDD "원래 0xDB 이었음"
521// 인코더는 앞뒤로 END 를 붙인다(앞쪽 END = 회선 노이즈로 생긴 쓰레기를 빈 프레임으로
522// 흘려보내 재동기화시키는 장치). 최악의 경우 출력 크기는 2*n + 2.
523// 길이 접두와 달리 **길이를 미리 알 필요가 없다** — 대신 크기가 최대 2배로 늘 수 있다.
524// 반환값: 기록한 바이트 수, 실패 시 -1.
525// ===========================================================================
526#define SLIP_END 0xC0
527#define SLIP_ESC 0xDB
528#define SLIP_ESC_END 0xDC
529#define SLIP_ESC_ESC 0xDD
530 
531long slip_encode(const uint8_t *in, size_t n, uint8_t *out, size_t out_sz) {
532 if ((!in && n > 0) || !out) return -1;
533 
534 size_t k = 0;
535 #define PUT(b) do { if (k >= out_sz) return -1; out[k++] = (uint8_t)(b); } while (0)
536 PUT(SLIP_END);
537 for (size_t i = 0; i < n; i++) {
538 if (in[i] == SLIP_END) { PUT(SLIP_ESC); PUT(SLIP_ESC_END); }
539 else if (in[i] == SLIP_ESC) { PUT(SLIP_ESC); PUT(SLIP_ESC_ESC); }
540 else { PUT(in[i]); }
541 }
542 PUT(SLIP_END);
543 #undef PUT
544 return (long)k;
545}
546 
547// in 은 END 로 감싸진 프레임 하나. 선행 END 들은 건너뛰고, 종결 END 에서 멈춘다.
548// 잘못된 이스케이프 시퀀스(ESC 뒤에 ESC_END/ESC_ESC 가 아님, 또는 ESC 로 끝남) -> -1.
549// 종결 END 가 없으면(잘린 프레임) -> -1.
550long slip_decode(const uint8_t *in, size_t n, uint8_t *out, size_t out_sz) {
551 if ((!in && n > 0) || !out) return -1;
552 
553 size_t i = 0;
554 while (i < n && in[i] == SLIP_END) i++; // 선행 END / 빈 프레임 스킵
555 
556 size_t k = 0;
557 while (i < n) {
558 uint8_t b = in[i++];
559 if (b == SLIP_END) {
560 return (long)k; // 정상 종료
561 }
562 if (b == SLIP_ESC) {
563 if (i >= n) return -1; // ESC 로 끝남 = 잘린 프레임
564 uint8_t e = in[i++];
565 if (e == SLIP_ESC_END) b = SLIP_END;
566 else if (e == SLIP_ESC_ESC) b = SLIP_ESC;
567 else return -1; // 프로토콜 위반
568 }
569 if (k >= out_sz) return -1; // 목적지 버퍼 부족
570 out[k++] = b;
571 }
572 return -1; // 종결 END 미도착
573}
