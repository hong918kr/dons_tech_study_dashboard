56// ===========================================================================
57// Q1. 스레드 생성/조인 + 보호 없는 공유 카운터 (race 재현)
58// ---------------------------------------------------------------------------
59// pthread_create / pthread_join 의 기본기와, 보호 없는 `counter = counter + 1`
60// 이 왜 깨지는지를 동시에 본다. `++` 는 load-modify-store 3단계라 원자적이지
61// 않다. 스레드별 슬롯(local_done)은 경합이 없으므로 "몇 번 시도했는가"는
62// 정확히 셀 수 있고, 공유 카운터는 그보다 작아질 수 있다.
63// 테스트는 결정적으로: attempts == N*iters, 1 <= observed <= attempts.
64// ===========================================================================
65static volatile long g_unsafe_counter = 0; // volatile은 최적화 억제일 뿐, 원자성 없음
66 
67typedef struct {
68 long iters;
69 long local_done; // 스레드마다 다른 슬롯 → 여기에는 race 없음
70} RaceArg;
71 
72static void *race_worker(void *p) {
73 RaceArg *a = (RaceArg *)p;
74 for (long i = 0; i < a->iters; ++i) {
75 g_unsafe_counter = g_unsafe_counter + 1; // ★ 보호 없음 — 의도된 race
76 a->local_done++;
77 }
78 return NULL;
79}

81// 반환: 관측된 최종 공유 카운터. *out_attempts: 스레드별로 센 총 증가 시도 횟수.
82// 실패 시 -1.
83long race_demo(int nthreads, long iters, long *out_attempts) {
84 pthread_t th[MAX_WORKERS];
85 RaceArg args[MAX_WORKERS];
86 if (nthreads < 1 || nthreads > MAX_WORKERS) return -1;
87 
88 g_unsafe_counter = 0;
89 for (int i = 0; i < nthreads; ++i) { args[i].iters = iters; args[i].local_done = 0; }
90 
91 int created = 0;
92 for (int i = 0; i < nthreads; ++i) {
93 if (pthread_create(&th[i], NULL, race_worker, &args[i]) != 0) break;
94 created++;
95#if !VK_RACE_CONCURRENT
96 pthread_join(th[i], NULL); // TSan 빌드: 직렬 실행 → happens-before 성립
97#endif
98 }
99#if VK_RACE_CONCURRENT
100 for (int i = 0; i < created; ++i) pthread_join(th[i], NULL);
101#endif
102 if (created != nthreads) return -1;
103 
104 long attempts = 0;
105 for (int i = 0; i < nthreads; ++i) attempts += args[i].local_done;
106 if (out_attempts) *out_attempts = attempts;
107 return g_unsafe_counter;
108}

132void sc_inc(SafeCounter *c) {
133 if (!c) return;
134 pthread_mutex_lock(&c->m);
135 c->value++; // 임계구역: 딱 이 한 줄
136 pthread_mutex_unlock(&c->m);

139long sc_get(SafeCounter *c) {
140 if (!c) return -1;
141 pthread_mutex_lock(&c->m); // 읽기도 보호해야 한다 (torn/stale 방지)
142 long v = c->value;
143 pthread_mutex_unlock(&c->m);
144 return v;
145}

155// nthreads개 스레드가 각각 iters번 증가시킨 뒤 최종값을 반환. 실패 시 -1.
156long safe_counter_run(int nthreads, long iters) {
157 pthread_t th[MAX_WORKERS];
158 SafeArg args[MAX_WORKERS];
159 SafeCounter c;
160 if (nthreads < 1 || nthreads > MAX_WORKERS) return -1;
161 
162 sc_init(&c);
163 for (int i = 0; i < nthreads; ++i) { args[i].c = &c; args[i].iters = iters; }
164 for (int i = 0; i < nthreads; ++i) pthread_create(&th[i], NULL, safe_worker, &args[i]);
165 for (int i = 0; i < nthreads; ++i) pthread_join(th[i], NULL);
166 
167 long v = sc_get(&c);
168 sc_destroy(&c);
169 return v;
170}

174// ---------------------------------------------------------------------------
175// GC31-E의 WAN 링크 RTT 통계(count/sum/min/max). 필드 4개가 "같은 순간의
176// 값"이어야 한다는 것이 불변식이다. 필드마다 락을 따로 잡거나 snapshot을
177// 여러 번 나눠 읽으면 count는 새 값, min은 옛 값인 찢어진(torn) 상태가 나온다.
178// 규칙: 한 번의 lock 안에서 관련 필드를 전부 갱신 / 전부 복사한다.
179// ===========================================================================
180typedef struct {
181 long count;
182 long sum_ms;
183 int min_ms;
184 int max_ms;
185 pthread_mutex_t m;
186} LinkStats;

189 long count;
190 long sum_ms;

193} LinkStatsView;

195void ls_init(LinkStats *s) {
196 if (!s) return;
197 s->count = 0;
198 s->sum_ms = 0;
199 s->min_ms = INT_MAX;
200 s->max_ms = INT_MIN;
201 pthread_mutex_init(&s->m, NULL);
202}

204void ls_destroy(LinkStats *s) {
205 if (!s) return;
206 pthread_mutex_destroy(&s->m);
207}

209void ls_add(LinkStats *s, int rtt_ms) {
210 if (!s) return;
211 pthread_mutex_lock(&s->m);
212 s->count++;
213 s->sum_ms += rtt_ms;
214 if (rtt_ms < s->min_ms) s->min_ms = rtt_ms;
215 if (rtt_ms > s->max_ms) s->max_ms = rtt_ms;
216 pthread_mutex_unlock(&s->m); // 네 필드가 한 임계구역에서 함께 갱신됨
217}

219// 한 번의 lock으로 4개 필드를 통째로 복사한다. 표본이 없으면 min/max = 0.
220void ls_snapshot(LinkStats *s, LinkStatsView *out) {
221 if (!s || !out) return;
222 pthread_mutex_lock(&s->m);
223 out->count = s->count;
224 out->sum_ms = s->sum_ms;
225 out->min_ms = (s->count == 0) ? 0 : s->min_ms;
226 out->max_ms = (s->count == 0) ? 0 : s->max_ms;
227 pthread_mutex_unlock(&s->m);
228}

238static _Atomic long g_ls_bad_snapshots = 0;
239static _Atomic bool g_ls_reader_stop = false;
240 
241static void *ls_reader(void *p) {
242 LinkStats *s = (LinkStats *)p;
243 while (!atomic_load(&g_ls_reader_stop)) {
244 LinkStatsView v = {0, 0, 0, 0};
245 ls_snapshot(s, &v);
246 // 불변식: min<=max, 그리고 sum은 count*min ~ count*max 사이에 있어야 한다.
247 if (v.count > 0) {
248 if (v.min_ms > v.max_ms ||
249 v.sum_ms < (long)v.min_ms * v.count ||
250 v.sum_ms > (long)v.max_ms * v.count) {
251 atomic_fetch_add(&g_ls_bad_snapshots, 1L);
252 }
253 }
254 }
255 return NULL;
256}

259// Q4. 임계구역 최소화 — 느린 I/O를 락 밖으로
260// ---------------------------------------------------------------------------
261// "락 안에서 I/O도 콜백도 부르지 않는다"는 임베디드 동시성의 1번 규칙.
262// 락 안에서는 상태만 바꾸고 필요한 데이터를 지역 변수로 복사한 뒤, 락을 풀고
263// 업로드한다. 락 안에서 blocking write를 하면 모뎀이 느려질 때 전체 파이프라인이
264// 멈추고, 콜백을 부르면 그 콜백이 다시 락을 잡아 데드락이 난다.
265// 검증 트릭: 가짜 I/O 안에서 같은 mutex를 trylock 해본다. 성공하면 "락을 안 쥔
266// 상태"(정답), EBUSY면 "락을 쥔 채 I/O"(오답). 단일 스레드에서만 유효한 측정이라
267// g_lockcheck_on 플래그로 측정 구간을 한정한다.
268// ===========================================================================
269typedef struct {
270 char last[64];
271 long queued;
272 _Atomic long io_calls;
273 pthread_mutex_t m;
274} LogSink;

276static bool g_lockcheck_on = false;
277static _Atomic long g_io_inside_lock = 0;

282 s->queued = 0;

284 pthread_mutex_init(&s->m, NULL);
285}

292// 느린 업로드 흉내 (모뎀으로 blocking write 한다고 치자). 건드리지 말 것.
293static void slow_upload(LogSink *s, const char *msg) {
294 (void)msg;
295 if (g_lockcheck_on) {
296 if (pthread_mutex_trylock(&s->m) == 0) pthread_mutex_unlock(&s->m); // 락 안 쥠 = 정답
297 else atomic_fetch_add(&g_io_inside_lock, 1L);
298 }
299 for (volatile int i = 0; i < 200; ++i) { } // 지연 흉내
300 atomic_fetch_add(&s->io_calls, 1L);
301}

303// ★ 나쁜 예 (참고용, 이미 구현됨): 락을 쥔 채 업로드한다.
304bool sink_write_bad(LogSink *s, const char *msg) {
305 if (!s || !msg) return false;
306 pthread_mutex_lock(&s->m);
307 snprintf(s->last, sizeof s->last, "%s", msg);
308 s->queued++;
309 slow_upload(s, s->last); // ← 임계구역 안에서 I/O. 모뎀이 막히면 전원이 막힌다.
310 pthread_mutex_unlock(&s->m);
311 return true;
312}

314// ★ 좋은 예: 락 안에서는 상태 갱신 + 지역 복사만, I/O는 락 밖에서.
315bool sink_write(LogSink *s, const char *msg) {
316 if (!s || !msg) return false;
317 char copy[64];
318 pthread_mutex_lock(&s->m);
319 snprintf(s->last, sizeof s->last, "%s", msg);
320 s->queued++;
321 memcpy(copy, s->last, sizeof copy); // 락 안에서 스냅샷
322 pthread_mutex_unlock(&s->m);
323 
324 slow_upload(s, copy); // 락 밖에서 느린 작업
325 return true;
326}

345// Q5. 중첩 락 방지 — `_locked` 접미사 규약
346// ---------------------------------------------------------------------------
347// 공개 함수는 락을 잡고, 내부 헬퍼는 "호출자가 이미 락을 쥐고 있다"를 전제로
348// `_locked` 접미사를 붙여 락 없는 버전으로 둔다. 그러면 여러 연산을 한 임계
349// 구역으로 묶을 때 공개 함수를 재호출하지 않아도 된다.
350// 재귀 뮤텍스(PTHREAD_MUTEX_RECURSIVE)를 쓰지 않는 이유: 잠긴 채 함수를 빠져나가도
351// 컴파일이 되고, "이 함수가 불변식을 깬 중간 상태에서 불릴 수 있다"는 사실이
352// 감춰지며, condvar와 섞이면 wait 시 카운트가 1만 내려가 버그가 난다.
353// 여기서는 진단용으로 ERRORCHECK 뮤텍스를 써서 중첩 락이 EDEADLK 로 즉시 잡히게 한다.
354// ===========================================================================
355#define POE_PORTS 2

357typedef struct {
358 int port_mw[POE_PORTS];
359 int total_mw; // 파생값 — 항상 port_mw 합과 같아야 한다 (불변식)
360 pthread_mutex_t m; // PTHREAD_MUTEX_ERRORCHECK
361} PoeTable;

363typedef struct { int port_mw[POE_PORTS]; int total_mw; } PoeView;

365void poe_init(PoeTable *t) {
366 if (!t) return;
367 for (int i = 0; i < POE_PORTS; ++i) t->port_mw[i] = 0;
368 t->total_mw = 0;
369 pthread_mutexattr_t attr;
370 pthread_mutexattr_init(&attr);
371 pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_ERRORCHECK);
372 pthread_mutex_init(&t->m, &attr);
373 pthread_mutexattr_destroy(&attr);
374}

381// --- 락 없는 내부 헬퍼들: 호출자가 t->m 을 쥐고 있어야 한다 ---
382static void poe_recompute_locked(PoeTable *t) {
383 int sum = 0;
384 for (int i = 0; i < POE_PORTS; ++i) sum += t->port_mw[i];
385 t->total_mw = sum;
386}

402// 두 포트를 한 임계구역에서 원자적으로 바꾼다.
403// ★ 여기서 poe_set_port() 를 부르면 중첩 락 → 데드락(또는 EDEADLK).

409 poe_set_port_locked(t, 1, b_mw);
410 pthread_mutex_unlock(&t->m);
411 return true;
412}

445static void *poe_checker(void *p) {
446 PoeTable *t = (PoeTable *)p;
447 while (!atomic_load(&g_poe_stop)) {
448 PoeView v = {{0, 0}, 0};
449 poe_snapshot(t, &v);
450 // 불변식: total == 합, 그리고 (a,b)는 writer가 쓰는 쌍 중 하나(또는 초기값)

460// ===========================================================================
461// Q6. TOCTOU — PoE 전력 예산 예약을 원자적으로
462// ---------------------------------------------------------------------------
463// GC31-E는 802.3bt 출력 2포트에 총 60W 예산을 나눠 준다. "남았나 확인 → 예약"이
464// 두 개의 임계구역으로 쪼개지면 Time-Of-Check-To-Time-Of-Use 버그가 나서 예산을
465// 초과 배정(over-commit)한다. 검사와 행동은 반드시 같은 lock 안에 있어야 한다.
466// ===========================================================================
467typedef struct {
468 int total_mw;
469 int used_mw;
470 pthread_mutex_t m;
471} PoeBudget;

473void budget_init(PoeBudget *b, int total_mw) {
474 if (!b) return;
475 b->total_mw = total_mw;
476 b->used_mw = 0;
477 pthread_mutex_init(&b->m, NULL);
478}

480void budget_destroy(PoeBudget *b) {
481 if (!b) return;
482 pthread_mutex_destroy(&b->m);
483}

485// ★ 좋은 예: 검사와 행동이 한 임계구역
486bool budget_reserve(PoeBudget *b, int mw) {
487 if (!b || mw <= 0) return false;
488 bool ok = false;
489 pthread_mutex_lock(&b->m);
490 if (b->used_mw + mw <= b->total_mw) { // check
491 b->used_mw += mw; // ...and act — 같은 락 안
492 ok = true;
493 }
494 pthread_mutex_unlock(&b->m);
495 return ok;
496}
497 
498// ★ 나쁜 예 (참고용, 이미 구현됨): 검사와 행동 사이에 락이 풀린다 → over-commit
499bool budget_reserve_toctou(PoeBudget *b, int mw) {
500 if (!b || mw <= 0) return false;
501 pthread_mutex_lock(&b->m);
502 bool fits = (b->used_mw + mw <= b->total_mw); // check ...
503 pthread_mutex_unlock(&b->m); // ← 여기서 창이 열린다
504 
505 if (!fits) return false;
506 sched_yield(); // 창을 벌려 버그를 재현
507 
508 pthread_mutex_lock(&b->m);
509 b->used_mw += mw; // ... act (조건 재확인 없음)
510 pthread_mutex_unlock(&b->m);
511 return true;
512}
513 
514void budget_release(PoeBudget *b, int mw) {
515 if (!b || mw <= 0) return;
516 pthread_mutex_lock(&b->m);
517 b->used_mw -= mw;
518 if (b->used_mw < 0) b->used_mw = 0;
519 pthread_mutex_unlock(&b->m);
520}
521 
522int budget_used(PoeBudget *b) {
523 if (!b) return -1;
524 pthread_mutex_lock(&b->m);
525 int u = b->used_mw;
526 pthread_mutex_unlock(&b->m);
527 return u;
528}

532static void *reserve_worker(void *p) {
533 ResArg *a = (ResArg *)p;
534 a->granted = a->use_toctou ? budget_reserve_toctou(a->b, a->mw)

540static int run_reservations(PoeBudget *b, int nthreads, int mw, bool use_toctou) {
541 pthread_t th[MAX_WORKERS];
542 ResArg args[MAX_WORKERS];
543 if (nthreads < 1 || nthreads > MAX_WORKERS) return -1;

594// ===========================================================================
595// Q8. 스레드별 로컬 집계 후 병합 — 경합 줄이기
596// ---------------------------------------------------------------------------
597// Q2는 증가 1회마다 락을 잡는다(= nthreads*iters 번). 각 스레드가 지역 변수에
598// 모았다가 끝에 한 번만 병합하면 락 획득이 nthreads 번으로 줄고, 캐시 라인
599// 핑퐁도 사라진다. 통계/카운터처럼 "중간값이 필요 없는" 집계의 정석.
600// 테스트는 결과값뿐 아니라 '락 획득 횟수'까지 결정적으로 검사한다.
601// ===========================================================================
602static long g_merged_total = 0;
603static long g_merge_locks = 0;
604static pthread_mutex_t g_merge_m = PTHREAD_MUTEX_INITIALIZER;
605 
606typedef struct { long iters; } ShardArg;
607 
608static void *shard_worker(void *p) {
609 ShardArg *a = (ShardArg *)p;
610 long local = 0;
611 for (long i = 0; i < a->iters; ++i) local++; // 공유 상태 접근 0회
612 
613 pthread_mutex_lock(&g_merge_m); // 스레드당 락 1회
614 g_merged_total += local;
615 g_merge_locks++;
616 pthread_mutex_unlock(&g_merge_m);
617 return NULL;
618}
619 
620// 반환: 병합된 총합. *out_locks: 총 락 획득 횟수(= nthreads 여야 한다).
621long sharded_sum_run(int nthreads, long iters, long *out_locks) {
622 pthread_t th[MAX_WORKERS];
623 ShardArg args[MAX_WORKERS];
624 if (nthreads < 1 || nthreads > MAX_WORKERS) return -1;
625 
626 pthread_mutex_lock(&g_merge_m);
627 g_merged_total = 0;
628 g_merge_locks = 0;
629 pthread_mutex_unlock(&g_merge_m);
630 
631 for (int i = 0; i < nthreads; ++i) args[i].iters = iters;
632 for (int i = 0; i < nthreads; ++i) pthread_create(&th[i], NULL, shard_worker, &args[i]);
633 for (int i = 0; i < nthreads; ++i) pthread_join(th[i], NULL);
634 
635 pthread_mutex_lock(&g_merge_m);
636 long total = g_merged_total;
637 if (out_locks) *out_locks = g_merge_locks;
638 pthread_mutex_unlock(&g_merge_m);
639 return total;
640}

643// Q9. pthread_rwlock — 읽기 다수 / 쓰기 소수 설정 테이블
644// ---------------------------------------------------------------------------

717// ===========================================================================
718// Q10. 취소 가능한 워커 — stop 플래그 + join 으로 깔끔한 종료
719// ---------------------------------------------------------------------------
720// pthread_cancel 은 취소 지점이 어디인지 불분명하고 락/자원 누수를 남긴다.
721// 임베디드에서는 항상 "협조적 종료": mutex로 보호되는 stop 플래그를 세우고
722// (블로킹 중이면 condvar broadcast나 self-pipe로 깨우고) join 해서
723// 스레드가 실제로 끝난 것을 확인한 뒤 자원을 해제한다.
724// stop_and_join 은 두 번 불러도 안전해야 한다(idempotent).
725// ===========================================================================
726typedef struct {
727 pthread_t th;
728 bool running; // m으로 보호
729 bool stop; // m으로 보호
730 _Atomic long ticks; // 외부에서 락 없이 관찰
731 pthread_mutex_t m;
732} Worker;
733 
734void worker_init(Worker *w) {
735 if (!w) return;
736 w->running = false;
737 w->stop = false;
738 atomic_store(&w->ticks, 0L);
739 pthread_mutex_init(&w->m, NULL);
740}
741 
742void worker_destroy(Worker *w) {
743 if (!w) return;
744 pthread_mutex_destroy(&w->m);
745}
746 
747bool worker_should_stop(Worker *w) {
748 if (!w) return true;
749 pthread_mutex_lock(&w->m);
750 bool s = w->stop;
751 pthread_mutex_unlock(&w->m);
752 return s;
753}
754 
755long worker_ticks(Worker *w) { return w ? atomic_load(&w->ticks) : -1; }
756 
757static void *worker_main(void *p) {
758 Worker *w = (Worker *)p;
759 while (!worker_should_stop(w)) {
760 atomic_fetch_add(&w->ticks, 1L); // 실제로는 한 프레임 처리
761 }
762 return NULL;
763}
764 
765bool worker_start(Worker *w) {
766 if (!w) return false;
767 pthread_mutex_lock(&w->m);
768 if (w->running) { pthread_mutex_unlock(&w->m); return false; }
769 w->stop = false;
770 w->running = true;
771 pthread_mutex_unlock(&w->m);
772 
773 if (pthread_create(&w->th, NULL, worker_main, w) != 0) {
774 pthread_mutex_lock(&w->m);
775 w->running = false;
776 pthread_mutex_unlock(&w->m);
777 return false;
778 }
779 return true;
780}
781 
782// stop 플래그를 세우고 join. 이미 멈췄으면 아무것도 하지 않는다. 총 ticks 반환.
783long worker_stop_and_join(Worker *w) {
784 if (!w) return -1;
785 pthread_mutex_lock(&w->m);
786 bool need_join = w->running;
787 w->stop = true;
788 w->running = false;
789 pthread_mutex_unlock(&w->m);
790 
791 if (need_join) pthread_join(w->th, NULL); // ★ join 해야 자원 해제가 안전
792 return atomic_load(&w->ticks);
793}

795// 테스트 헬퍼: ticks가 target 이상이 될 때까지 유한 횟수만 스핀한다.
796// (미구현 stub 에서도 절대 멈추지 않도록 상한을 둔다)
797static long spin_until_ticks(Worker *w, long target, long max_spins) {
798 for (long i = 0; i < max_spins; ++i) {
799 long t = worker_ticks(w);
800 if (t >= target) return t;
801 if ((i & 0x3FF) == 0) sched_yield();
802 }
803 return worker_ticks(w);
804}
