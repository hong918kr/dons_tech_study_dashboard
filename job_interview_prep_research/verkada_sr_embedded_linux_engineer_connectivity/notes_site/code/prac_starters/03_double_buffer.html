1/* 03. Double buffering (ping-pong) — 리크루터 메일에 명시된 주제
2 * ------------------------------------------------------------------
3 * 센서/DMA가 한 버퍼를 채우는 동안 처리 스레드는 다른 버퍼를 읽는다.
4 * 요구: (1) 읽는 쪽이 절반만 갱신된 프레임(tearing)을 절대 보지 않는다
5 * (2) 생산자는 소비자를 기다리지 않는다(실시간 — 블로킹 금지)
6 * (3) 소비자가 느리면 "가장 최근 프레임"만 보면 된다(중간 프레임은 버림)
7 *
8 * 핵심 아이디어: 락으로 '데이터'를 보호하지 말고 '버퍼 소유권'을 보호한다.
9 * 각 버퍼는 항상 셋 중 하나의 상태: WRITER 소유 / READY(publish됨) / READER 소유
10 * 임계구역은 인덱스 몇 개 바꾸는 것뿐 → 수 나노초. 데이터 복사 없음.
11 *
12 * 버퍼가 2개뿐이면: reader가 하나를 들고 있고 다른 하나가 READY인 순간
13 * writer는 쓸 곳이 없다 → READY를 버린다(drop). 이것이 triple buffering을
14 * 쓰는 이유다(문제 04 참고).
15 * 빌드: cc -std=c11 -Wall -Wextra -pthread 03_double_buffer.c
16 */
17#include <assert.h>
18#include <pthread.h>
19#include <stdbool.h>
20#include <stdio.h>
21#include <string.h>
22 
23#define FRAME_LEN 512
24#define NBUF 2 /* 3으로 바꾸면 그대로 triple buffering이 된다 */
25 
26typedef struct {
27 unsigned seq;
28 unsigned char data[FRAME_LEN];
29} Frame;
30 
31typedef struct {
32 Frame buf[NBUF];
33 int write_idx; /* writer 소유 버퍼, 없으면 -1 */
34 int ready_idx; /* publish된 최신 버퍼, 없으면 -1 */
35 int reader_idx; /* reader 대여 중 버퍼, 없으면 -1 */
36 unsigned long dropped; /* 소비되지 못하고 버려진 프레임 수 */
37 pthread_mutex_t m;
38} DoubleBuf;
39 
40void db_init(DoubleBuf *db);
41void db_destroy(DoubleBuf *db);
42Frame *db_write_begin(DoubleBuf *db); /* writer: 채울 버퍼 확보(항상 성공) */
43void db_write_commit(DoubleBuf *db); /* writer: publish */
44Frame *db_read_acquire(DoubleBuf *db); /* reader: 최신 프레임 대여, 없으면 NULL */
45void db_read_release(DoubleBuf *db); /* reader: 반납 */
46 
47/* ------------------------------------------------------------------
48 * 여기부터 직접 구현한다. 위의 선언부와 아래 self-test는 그대로 두고,
49 * 아래 함수 목록을 채운 뒤 `make run N=03` 로 검증한다.
50 *
51 * 구현할 함수:
52 * - db_init()
53 * - db_destroy()
54 * - db_write_begin()
55 * - db_write_commit()
56 * - db_read_acquire()
57 * - db_read_release()
58 *
59 * 막히면 ../solutions/03_double_buffer.c 를 열되, 먼저 15분은 스스로 해볼 것.
60 * ------------------------------------------------------------------ */
61 
62/* ------------------------------- self test ------------------------------- */
63#define NFRAMES 20000
64 
65static DoubleBuf g_db;
66static _Atomic bool g_done;
67static unsigned long g_read_ok, g_torn, g_out_of_order;
68 
69static void *producer(void *arg) /* 센서 / DMA 완료 콜백 역할 */
70{
71 (void)arg;
72 for (unsigned s = 1; s <= NFRAMES; s++) {
73 Frame *f = db_write_begin(&g_db);
74 f->seq = s;
75 /* 프레임 전체를 같은 패턴으로 채운다 → tearing이 나면 값이 섞인다 */
76 memset(f->data, (int)(s & 0xFF), FRAME_LEN);
77 db_write_commit(&g_db);
78 }
79 g_done = true;
80 return NULL;
81}
82 
83static void *consumer(void *arg)
84{
85 (void)arg;
86 unsigned last = 0;
87 for (;;) {
88 Frame *f = db_read_acquire(&g_db);
89 if (!f) {
90 if (g_done) break; /* 생산 종료 + 남은 프레임 없음 */
91 continue; /* 실전이라면 condvar로 대기 */
92 }
93 unsigned char expect = (unsigned char)(f->seq & 0xFF);
94 for (int i = 0; i < FRAME_LEN; i++) {
95 if (f->data[i] != expect) { g_torn++; break; }
96 }
97 if (f->seq < last) g_out_of_order++; /* 항상 최신만 오므로 단조 증가여야 */
98 last = f->seq;
99 g_read_ok++;
100 db_read_release(&g_db);
101 }
102 return NULL;
103}
104 
105int main(void)
106{
107 pthread_t p, c;
108 db_init(&g_db);
109 pthread_create(&c, NULL, consumer, NULL);
110 pthread_create(&p, NULL, producer, NULL);
111 pthread_join(p, NULL);
112 pthread_join(c, NULL);
113 
114 /* 마지막 프레임이 publish된 직후 소비자가 종료했을 수 있다 → 1개는 남을 수 있음 */
115 unsigned long leftover = (g_db.ready_idx >= 0) ? 1u : 0u;
116 printf("frames=%d read=%lu dropped=%lu leftover=%lu torn=%lu ooo=%lu\n",
117 NFRAMES, g_read_ok, g_db.dropped, leftover, g_torn, g_out_of_order);
118 assert(g_torn == 0); /* tearing 없어야 통과 */
119 assert(g_out_of_order == 0);
120 assert(g_read_ok > 0);
121 /* 모든 프레임은 소비되었거나, 버려졌거나, 마지막으로 남아 있다 */
122 assert(g_read_ok + g_db.dropped + leftover == NFRAMES);
123 db_destroy(&g_db);
124 puts("03 PASS");
125 return 0;
