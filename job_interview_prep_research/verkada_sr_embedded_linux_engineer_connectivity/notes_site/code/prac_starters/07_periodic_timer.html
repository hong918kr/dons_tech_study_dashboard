1/* 07. 주기 실행 + 취소 가능한 대기 (timed wait)
2 * ------------------------------------------------------------------
3 * "센서를 100ms마다 샘플링하되, 종료 신호가 오면 즉시 깨어나라."
4 * sleep(100ms) 루프의 문제:
5 * (1) 작업 시간이 누적돼 주기가 밀린다(drift)
6 * (2) 종료 신호를 받아도 최대 100ms 늦게 반응한다
7 * 해결: 절대 시각(deadline) 기준으로 cond_timedwait → drift 없음 + 즉시 취소.
8 *
9 * 이식성 함정(인터뷰에서 가산점):
10 * pthread_cond_timedwait의 기본 시계는 CLOCK_REALTIME이다. NTP가 시각을
11 * 되돌리면 대기가 늘어진다 → Linux에서는 condattr로 CLOCK_MONOTONIC 지정.
12 * macOS는 setclock을 지원하지 않아 *_relative_np를 쓴다.
13 * 빌드: cc -std=c11 -Wall -Wextra -pthread 07_periodic_timer.c
14 */
15#include <assert.h>
16#include <errno.h>
17#include <pthread.h>
18#include <stdatomic.h>
19#include <stdbool.h>
20#include <stdint.h>
21#include <stdio.h>
22#include <time.h>
23 
24#define NS_PER_MS 1000000LL
25#define NS_PER_S 1000000000LL
26 
27typedef struct {
28 pthread_mutex_t m;
29 pthread_cond_t cv;
30 bool stop;
31 unsigned long ticks, late_ticks;
32 int64_t period_ns;
33} Periodic;
34 
35static int64_t now_mono_ns(void);
36void periodic_init(Periodic *p, int64_t period_ns);
37void periodic_destroy(Periodic *p);
38/* deadline까지 대기. 중간에 stop이 오면 즉시 true 반환 */
39bool periodic_wait_until(Periodic *p, int64_t deadline_ns);
40void periodic_stop(Periodic *p);
41 
42/* ------------------------------------------------------------------
43 * 여기부터 직접 구현한다. 위의 선언부와 아래 self-test는 그대로 두고,
44 * 아래 함수 목록을 채운 뒤 `make run N=07` 로 검증한다.
45 *
46 * 구현할 함수:
47 * - now_mono_ns()
48 * - periodic_init()
49 * - periodic_destroy()
50 * - periodic_wait_until()
51 * - periodic_stop()
52 *
53 * 막히면 ../solutions/07_periodic_timer.c 를 열되, 먼저 15분은 스스로 해볼 것.
54 * ------------------------------------------------------------------ */
55 
56/* ------------------------------- self test ------------------------------- */
57static Periodic g_p;
58static _Atomic int64_t g_first_ns, g_last_ns;
59 
60static void *sampler(void *arg)
61{
62 (void)arg;
63 int64_t next = now_mono_ns() + g_p.period_ns;
64 g_first_ns = now_mono_ns();
65 for (;;) {
66 if (periodic_wait_until(&g_p, next)) break; /* stop 신호 */
67 g_p.ticks++;
68 g_last_ns = now_mono_ns();
69 /* 작업 시간 흉내: 주기의 일부를 소모 */
70 struct timespec work = { 0, 2 * NS_PER_MS };
71 nanosleep(&work, NULL);
72 /* drift 방지: '지금 + period'가 아니라 '이전 deadline + period' */
73 next += g_p.period_ns;
74 if (next < now_mono_ns()) { /* 주기를 놓쳤다 */
75 g_p.late_ticks++;
76 next = now_mono_ns() + g_p.period_ns; /* 따라잡기 포기(catch-up 방지) */
77 }
78 }
79 return NULL;
80}
81 
82int main(void)
83{
84 pthread_t t;
85 periodic_init(&g_p, 10 * NS_PER_MS); /* 10ms 주기 */
86 pthread_create(&t, NULL, sampler, NULL);
87 
88 struct timespec run = { 0, 200 * NS_PER_MS }; /* 200ms 동안 → 약 20틱 */
89 nanosleep(&run, NULL);
90 
91 int64_t t0 = now_mono_ns();
92 periodic_stop(&g_p);
93 pthread_join(t, NULL);
94 int64_t stop_latency_us = (now_mono_ns() - t0) / 1000;
95 
96 double elapsed_ms = (double)(g_last_ns - g_first_ns) / 1e6;
97 printf("ticks=%lu late=%lu elapsed=%.1fms avg_period=%.2fms stop_latency=%lldus\n",
98 g_p.ticks, g_p.late_ticks, elapsed_ms,
99 g_p.ticks ? elapsed_ms / (double)g_p.ticks : 0.0, (long long)stop_latency_us);
100 
101 assert(g_p.ticks >= 15 && g_p.ticks <= 25); /* 10ms × 200ms ≈ 20틱 */
102 assert(stop_latency_us < 5000); /* 즉시 취소: 5ms 이내 */
103 periodic_destroy(&g_p);
104 puts("07 PASS");
105 return 0;
106}
