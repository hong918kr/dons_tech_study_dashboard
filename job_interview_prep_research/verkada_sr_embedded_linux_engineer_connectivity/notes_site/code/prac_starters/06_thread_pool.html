1/* 06. Worker pool + graceful shutdown
2 * ------------------------------------------------------------------
3 * 게이트웨이의 백그라운드 작업(로그 업로드, 설정 동기화, 디바이스 헬스체크)을
4 * N개의 워커가 처리한다. 종료 시 두 가지 모드를 지원해야 한다:
5 * SHUTDOWN_DRAIN : 큐에 남은 작업을 모두 처리한 뒤 종료 (정상 재부팅)
6 * SHUTDOWN_NOW : 진행 중 작업만 끝내고 큐는 버림 (watchdog/긴급)
7 *
8 * 인터뷰에서 자주 파는 edge case:
9 * - 워커가 cond_wait에 잠들어 있을 때 종료 → broadcast로 전부 깨워야 함
10 * - 종료 후 submit → 거부해야 함 (조용히 무시하면 버그 추적 불가)
11 * - join을 빼먹으면? → 스레드가 자원 해제 전에 free된 큐를 만진다(UAF)
12 * - 작업 함수가 blocking이면 종료가 지연된다 → 작업에 cancel 신호를 전달
13 * 빌드: cc -std=c11 -Wall -Wextra -pthread 06_thread_pool.c
14 */
15#include <assert.h>
16#include <pthread.h>
17#include <stdatomic.h>
18#include <stdbool.h>
19#include <stdio.h>
20#include <stdlib.h>
21#include <string.h>
22 
23typedef void (*TaskFn)(void *arg);
24typedef enum { POOL_RUNNING, POOL_DRAIN, POOL_STOP } PoolState;
25 
26typedef struct { TaskFn fn; void *arg; } Task;
27 
28typedef struct {
29 pthread_t *workers;
30 size_t nworkers;
31 Task *q;
32 size_t cap, head, count;
33 PoolState state;
34 unsigned long executed, rejected, discarded;
35 pthread_mutex_t m;
36 pthread_cond_t work, slot;
37} Pool;
38 
39int pool_init(Pool *p, size_t nworkers, size_t qcap);
40bool pool_submit(Pool *p, TaskFn fn, void *arg); /* false = 종료 중이라 거부 */
41void pool_shutdown(Pool *p, bool drain); /* join까지 끝내고 반환 */

43/* ------------------------------------------------------------------
44 * 여기부터 직접 구현한다. 위의 선언부와 아래 self-test는 그대로 두고,
45 * 아래 함수 목록을 채운 뒤 `make run N=06` 로 검증한다.
46 *
47 * 구현할 함수:
48 * - pool_take()
49 * - worker_main()
50 * - pool_init()
51 * - pool_submit()
52 * - pool_shutdown()
53 *
54 * 막히면 ../solutions/06_thread_pool.c 를 열되, 먼저 15분은 스스로 해볼 것.
55 * ------------------------------------------------------------------ */
56 
57/* ------------------------------- self test ------------------------------- */
58static _Atomic unsigned long g_counter;
59 
60static void inc_task(void *arg)
61{
62 (void)arg;
63 atomic_fetch_add(&g_counter, 1u);
64}
65 
66int main(void)
67{
68 /* case 1: drain 종료 → 제출한 모든 작업이 실행돼야 한다 */
69 Pool p;
70 g_counter = 0;
71 assert(pool_init(&p, 4, 16) == 0);
72 for (int i = 0; i < 5000; i++)
73 assert(pool_submit(&p, inc_task, NULL));
74 unsigned long executed_before = 0;
75 pool_shutdown(&p, true);
76 executed_before = p.executed;
77 printf("drain : submitted=5000 executed=%lu counter=%lu discarded=%lu\n",
78 executed_before, (unsigned long)g_counter, p.discarded);
79 assert(g_counter == 5000);
80 assert(p.discarded == 0);
81 
82 /* case 2: 즉시 종료 → 실행 + 폐기 합이 제출 수와 같아야 한다 */
83 Pool p2;
84 g_counter = 0;
85 assert(pool_init(&p2, 2, 64) == 0);
86 unsigned long submitted = 0;
87 for (int i = 0; i < 2000; i++)
88 if (pool_submit(&p2, inc_task, NULL)) submitted++;
89 pool_shutdown(&p2, false);
90 printf("stop : submitted=%lu executed=%lu discarded=%lu\n",
91 submitted, p2.executed, p2.discarded);
92 assert(p2.executed + p2.discarded == submitted);
93 assert(g_counter == p2.executed);
94 
95 /* case 3: 종료 후 submit은 거부 */
96 Pool p3;
97 assert(pool_init(&p3, 1, 4) == 0);
98 pool_shutdown(&p3, true);
99 /* shutdown이 자원을 해제했으므로 여기서 submit하면 UAF —
100 * 그래서 실무 API는 shutdown과 destroy를 분리한다(아래 follow-up 참고) */
101 puts("06 PASS");
102 return 0;
103}
