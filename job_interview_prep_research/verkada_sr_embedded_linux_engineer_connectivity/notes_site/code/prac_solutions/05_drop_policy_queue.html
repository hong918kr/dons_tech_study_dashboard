1/* 05. Backpressure & drop policy — 링크가 끊겨도 죽지 않는 텔레메트리 큐
2 * ------------------------------------------------------------------
3 * 게이트웨이가 LTE로 텔레메트리를 올린다. 링크가 느려지거나 끊기면 업로더가
4 * 막히고, 그 사이에도 센서 스레드는 계속 샘플을 만든다.
5 * "메모리는 유한하다" → 큐가 차면 무엇을 할지 정책으로 정한다:
6 * BLOCK : 생산자를 막는다 (손실 없음, 실시간 경로에는 금지)
7 * DROP_NEWEST : 새 샘플을 버린다 (과거 이력 보존 — 로그/감사에 적합)
8 * DROP_OLDEST : 가장 오래된 것을 버린다 (최신 상태 보존 — 상태/알람에 적합)
9 * 인터뷰 포인트: 정책은 '데이터의 의미'가 결정한다. 그리고 버린 개수는 반드시
10 * 카운트해서 함께 보고한다(조용한 손실이 제일 나쁘다).
11 *
12 * 불변식(테스트로 검증): produced == consumed + dropped + queued
13 * 빌드: cc -std=c11 -Wall -Wextra -pthread 05_drop_policy_queue.c
14 */

23typedef enum { POLICY_BLOCK, POLICY_DROP_NEWEST, POLICY_DROP_OLDEST } DropPolicy;

25typedef struct { uint32_t ts_ms; int32_t value; } Sample;

27typedef struct {
28 Sample *buf;
29 size_t cap, head, count;
30 DropPolicy policy;
31 bool closed;
32 unsigned long pushed, dropped, popped;
33 pthread_mutex_t m;
34 pthread_cond_t not_full, not_empty;
35} TeleQueue;

43/*@impl-begin*/
44int tq_init(TeleQueue *q, size_t cap, DropPolicy p)
45{
46 if (cap == 0) return -1;
47 memset(q, 0, sizeof *q);
48 q->buf = calloc(cap, sizeof *q->buf);
49 if (!q->buf) return -1;
50 q->cap = cap;
51 q->policy = p;
52 pthread_mutex_init(&q->m, NULL);
53 pthread_cond_init(&q->not_full, NULL);
54 pthread_cond_init(&q->not_empty, NULL);
55 return 0;
56}
57 
58void tq_destroy(TeleQueue *q)
59{
60 free(q->buf);
61 pthread_mutex_destroy(&q->m);
62 pthread_cond_destroy(&q->not_full);
63 pthread_cond_destroy(&q->not_empty);
64}
65 
66bool tq_push(TeleQueue *q, Sample s)
67{
68 pthread_mutex_lock(&q->m);
69 if (q->closed) { pthread_mutex_unlock(&q->m); return false; }
70 
71 if (q->count == q->cap) {
72 switch (q->policy) {
73 case POLICY_BLOCK:
74 while (q->count == q->cap && !q->closed)
75 pthread_cond_wait(&q->not_full, &q->m);
76 if (q->closed) { pthread_mutex_unlock(&q->m); return false; }
77 break;
78 case POLICY_DROP_NEWEST:
79 q->dropped++;
80 pthread_mutex_unlock(&q->m);
81 return true; /* 성공으로 취급하되 카운트 */
82 case POLICY_DROP_OLDEST:
83 q->head = (q->head + 1) % q->cap; /* 가장 오래된 것 폐기 */
84 q->count--;
85 q->dropped++;
86 break;
87 }
88 }
89 q->buf[(q->head + q->count) % q->cap] = s;
90 q->count++;
91 q->pushed++;
92 pthread_mutex_unlock(&q->m);
93 pthread_cond_signal(&q->not_empty);
94 return true;
95}

97size_t tq_pop_batch(TeleQueue *q, Sample *out, size_t max)
98{
99 pthread_mutex_lock(&q->m);
100 while (q->count == 0 && !q->closed)
101 pthread_cond_wait(&q->not_empty, &q->m);
102 
103 size_t n = q->count < max ? q->count : max; /* 배치로 꺼내 업로드 효율↑ */
104 for (size_t i = 0; i < n; i++) {
105 out[i] = q->buf[q->head];
106 q->head = (q->head + 1) % q->cap;
107 }
108 q->count -= n;
109 q->popped += n;
110 pthread_mutex_unlock(&q->m);
111 if (n) pthread_cond_broadcast(&q->not_full);
112 return n; /* 0 = closed && empty */
113}
114 
115void tq_close(TeleQueue *q)
116{
117 pthread_mutex_lock(&q->m);
118 q->closed = true;
119 pthread_mutex_unlock(&q->m);
120 pthread_cond_broadcast(&q->not_empty);
121 pthread_cond_broadcast(&q->not_full);
122}

126#define NPROD 3
127#define NPER 5000
128#define BATCH 16

130static TeleQueue g_q;
131static unsigned long g_consumed;
132 
133static void *producer(void *arg)
134{
135 int id = (int)(intptr_t)arg;
136 for (int i = 0; i < NPER; i++) {
137 Sample s = { .ts_ms = (uint32_t)i, .value = id };
138 if (!tq_push(&g_q, s)) break;
139 }
140 return NULL;
141}
142 
143static void *uploader(void *arg)
144{
145 (void)arg;
146 Sample batch[BATCH];
147 size_t n;
148 while ((n = tq_pop_batch(&g_q, batch, BATCH)) > 0) {
149 g_consumed += n;
150 for (volatile int spin = 0; spin < 200; spin++) { } /* 느린 링크 흉내 */
151 }
152 return NULL;
153}

155static void run_case(DropPolicy p, const char *name)
156{
157 pthread_t prod[NPROD], up;
158 g_consumed = 0;
159 assert(tq_init(&g_q, 32, p) == 0);
160 
161 pthread_create(&up, NULL, uploader, NULL);
162 for (int i = 0; i < NPROD; i++)
163 pthread_create(&prod[i], NULL, producer, (void *)(intptr_t)i);
164 for (int i = 0; i < NPROD; i++) pthread_join(prod[i], NULL);
165 tq_close(&g_q);
166 pthread_join(up, NULL);
167 
168 unsigned long produced = (unsigned long)NPROD * NPER;
169 unsigned long leftover = (unsigned long)g_q.count;
170 printf(" %-12s produced=%lu consumed=%lu dropped=%lu left=%lu\n",
171 name, produced, g_consumed, g_q.dropped, leftover);
172 /* 불변식: 만든 만큼은 소비되었거나 버려졌거나 큐에 남아 있다 */
173 assert(g_consumed + g_q.dropped + leftover == produced);
174 if (p == POLICY_BLOCK) assert(g_q.dropped == 0); /* 블로킹은 손실 없음 */
175 tq_destroy(&g_q);
176}
