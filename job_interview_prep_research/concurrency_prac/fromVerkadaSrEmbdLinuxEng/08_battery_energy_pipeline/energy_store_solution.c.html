1/* energy_store_solution.c — reference solution.
2 *
3 * ./main.sh sol build main.c + this file and run the checks
4 *
5 * ---------------------------------------------------------------------------
6 * WHY THIS SHAPE (the paragraph to say out loud in the interview)
7 *
8 * 1. bms_read() blocks and is not thread-safe, so exactly ONE thread ever
9 * touches it: the sampler (stage 1). Everything a caller sees is a copy of
10 * state the sampler published, so no caller ever waits on the BMS.
11 *
12 * 2. THE SAMPLER MUST NEVER BE BLOCKED BY THE UPLOADER. The BMS reports on its
13 * own irregular schedule; the samples it produces are only available in the
14 * window between one report and the next. If stage 1 ever waited for stage 2
15 * - a full queue, a rate-limit token, a slow network call - the sampler
16 * would not be sitting in bms_read() when the BMS reports, and the reading
17 * is simply gone: unlike a file, a sensor cannot be re-read later. So the
18 * hand-off from stage 1 to stage 2 is a BOUNDED queue with a DROP-OLDEST
19 * policy: pushing is O(1), never waits, and can never fail. Uploads are
20 * best-effort telemetry; the 30-minute local history is the source of truth
21 * for energy, and it is written before the queue push, so a dropped upload
22 * never costs us an energy sample.
23 *
24 * 3. RATE LIMIT WITHOUT BURSTS AND WITHOUT STARVATION: a token bucket driven by
25 * pthread_cond_timedwait(). The bucket holds at most UPLOAD_BURST tokens and
26 * refills continuously at UPLOAD_RATE_PER_SEC, so the number of uploads in
27 * any one-second window is at most burst + rate - that is the anti-burst
28 * half, and it is provable rather than measured. The anti-starvation half is
29 * the TIMED wait: the uploader does not spin and does not sleep for a fixed
30 * slice, it waits for exactly as long as the bucket needs to hold one token
31 * and no longer, and the same condvar is signalled when work arrives or when
32 * we are shutting down. A plain pthread_cond_wait() cannot express "wake me
33 * when either new work arrives OR enough time has passed"; a usleep() loop
34 * cannot be woken early, so deinit would have to wait out a whole interval.
35 *
36 * 4. Locks: three small ones, each held for a few instructions and NEVER across
37 * a blocking vendor call. latest_lock (publish/read one struct), hist_lock
38 * (append/query the history), q_lock (queue + token bucket). Separate locks
39 * mean a throttled uploader sitting in its timed wait on q_lock cannot delay
40 * a reader in energy_joules_between(), and cannot delay the sampler's
41 * publish either. A seqlock or a double buffer would also work for "latest";
42 * a mutex is enough here because the writer is one thread holding it for a
43 * 16-byte copy.
44 *
45 * 5. Part 2 = ring buffer of (timestamp, power) + PREFIX SUMS of trapezoidal
46 * energy. Append is O(1) (one add), an arbitrary range query is O(log n):
47 * binary search for the segment holding t0 and the one holding t1, subtract
48 * two prefix sums for everything in between, and interpolate the two partial
49 * edge slivers exactly. Fixed memory, no allocation on the sampling path.
50 * ------------------------------------------------------------------------- */

83/* Capacity: the BMS can report as often as every ~20 ms, so 30 min needs
84 * 1800 s / 0.020 s = 90 000 entries. Round up to a power of two so the ring
85 * index is a mask instead of a modulo: 2^17 = 131 072 entries * 24 B = 3.1 MiB

125static void cond_wait_us(pthread_cond_t *cv, pthread_mutex_t *m, uint64_t us)
126{
127#ifdef __APPLE__
128 /* macOS has no condattr clock selection, but it does have a relative wait,
129 * which is what we actually want and is immune to wall-clock jumps. */
130 struct timespec rel;
131 rel.tv_sec = (time_t)(us / 1000000ull);
132 rel.tv_nsec = (long)((us % 1000000ull) * 1000ull);
133 pthread_cond_timedwait_relative_np(cv, m, &rel);
134#else
135 /* A default condvar waits on CLOCK_REALTIME, the same clock gettimeofday()
136 * reads, so build the absolute deadline from gettimeofday(). */
137 struct timeval tv;
138 gettimeofday(&tv, NULL);
139 uint64_t deadline = 1000000ull * (uint64_t)tv.tv_sec + (uint64_t)tv.tv_usec + us;
140 struct timespec abs;
141 abs.tv_sec = (time_t)(deadline / 1000000ull);
142 abs.tv_nsec = (long)((deadline % 1000000ull) * 1000ull);
143 pthread_cond_timedwait(cv, m, &abs);
144#endif
145}

147/* ------------------------------------------------------------------ */
148/* history append (called by the sampler only) */
149/* ------------------------------------------------------------------ */

151/* THE TRAPEZOID, derived.
152 *
153 * The BMS gives us power only at the instants it reports: p(t_i) = V_i * I_i.
154 * Energy is the integral of power: E = integral p(t) dt. Between two reports we
155 * have no information, so we assume power moves linearly from p_i to p_{i+1}
156 * (the cheapest assumption that is continuous and exact when the load ramps).
157 * The integral of a straight line over [t_i, t_{i+1}] is the area of a
158 * trapezoid with parallel sides p_i and p_{i+1} and width dt:
159 *
160 * E_i = (p_i + p_{i+1}) / 2 * dt [ watts * seconds = joules ]
161 *
162 * dt must be in SECONDS, so divide the microsecond difference by 1e6.
163 * (Zero-order hold - "power stayed at p_i" - would be E_i = p_i * dt and is
164 * what you want if the sensor reports on change only; the BMS here reports
165 * samples of a continuously varying load, so trapezoid is the better model and
166 * it is second-order accurate instead of first-order.)
167 *
168 * PREFIX SUMS AND PRECISION.
169 * cum_i = sum of E_0..E_{i-1}, i.e. energy since the first sample ever, so any
170 * range's energy is one subtraction: cum_j - cum_i. That subtraction is exactly
171 * where precision bites - cum grows forever while the differences we want stay
172 * small, which is textbook catastrophic cancellation:
173 * - float has a 24-bit mantissa: once cum passes ~1.7e7 J (about 5 kWh, i.e.
174 * a few hours on this trailer) the spacing between representable values is
175 * larger than 1 J and short windows return garbage. Never accumulate energy
176 * in a float.
177 * - double has a 53-bit mantissa: at 1e12 J the spacing is still ~2e-4 J, so
178 * for a trailer that is fine for centuries. That is why cum is a double
179 * even though the samples arrive as floats.
180 * - if that were not enough, the alternatives are (a) Kahan/Neumaier
181 * compensated summation, which carries the rounding error in a second
182 * variable and buys back ~2 digits per add, or (b) rebasing: when the ring
183 * wraps, subtract cum[head] from every stored cum so the running total
184 * never grows without bound. Both are strictly more code; the reason we do
185 * not need them is the 30-minute window, which bounds the interesting range
186 * while cum only has to stay accurate in its own units.
187 */
188static void hist_append(const struct BmsSample *s)
189{
190 double p = (double)s->volts * (double)s->amps;
191 
192 pthread_mutex_lock(&hist_lock);
193 
194 /* The ring must stay sorted or the binary search is meaningless. The BMS
195 * timestamps come from a wall clock, which can step backwards (NTP): a
196 * sample that does not move time forward is dropped rather than inserted
197 * out of order. With a monotonic clock this branch never fires. */
198 if (hist_count > 0 && s->timestamp <= H(hist_count - 1)->ts) {
199 pthread_mutex_unlock(&hist_lock);
200 return;
201 }
202 
203 double cum = 0.0;
204 if (hist_count > 0) {
205 struct hist_entry *prev = H(hist_count - 1);
206 double dt_s = (double)(s->timestamp - prev->ts) / 1e6;
207 cum = prev->cum + 0.5 * (prev->p + p) * dt_s; /* the trapezoid */
208 }
209 
210 if (hist_count == HIST_CAP) { /* full: overwrite the oldest */
211 hist_head = (hist_head + 1u) & HIST_MASK;
212 hist_count--;
213 }
214 struct hist_entry *e = H(hist_count);
215 e->ts = s->timestamp;
216 e->p = p;
217 e->cum = cum;
218 hist_count++;
219 
220 /* Evict everything older than the window, but KEEP ONE sample older than
221 * the cut: the power at the window's edge is interpolated between the last
222 * sample before the edge and the first one after it, so throwing that
223 * older sample away would silently truncate the oldest part of every query.
224 * Condition: drop entry 0 only while entry 1 is still <= the cutoff. */
225 if (s->timestamp > ENERGY_WINDOW_US) {
226 uint64_t cutoff = s->timestamp - ENERGY_WINDOW_US;
227 while (hist_count >= 2 && H(1)->ts <= cutoff) {
228 hist_head = (hist_head + 1u) & HIST_MASK;
229 hist_count--;
230 }
231 }
232 pthread_mutex_unlock(&hist_lock);
233}

239static void publish(const struct BmsSample *s)
240{
241 uint64_t t0 = bms_now_us();
242 
243 /* (a) Part 1: latest wins immediately - this is the freshest thing callers
244 * can see and it must not depend on the uploader at all. */
245 pthread_mutex_lock(&latest_lock);
246 latest = *s;
247 latest_valid = true;
248 pthread_mutex_unlock(&latest_lock);
249 
250 /* (b) Part 2: the local history. Written BEFORE the upload queue, so the
251 * energy integral is complete even when uploads are being dropped. */
252 hist_append(s);
253 
254 /* (c) count the sample BEFORE handing it to stage 2. Order matters: a
255 * reader that sees "dropped" or "uploaded" must never see a smaller
256 * "sampled", or the invariant sampled >= uploaded + dropped breaks. */
257 atomic_fetch_add(&c_sampled, 1);
258 
259 /* (d) stage 2 hand-off: bounded, drop-oldest, never waits. */
260 pthread_mutex_lock(&q_lock);
261 if (q_count == UPLOAD_QUEUE_CAP) {
262 q_head = (q_head + 1u) % UPLOAD_QUEUE_CAP; /* drop the oldest */
263 q_count--;
264 atomic_fetch_add(&c_dropped, 1);
265 }
266 q[(q_head + q_count) % UPLOAD_QUEUE_CAP] = *s;
267 q_count++;
268 pthread_cond_signal(&q_cv); /* wake the uploader if it is idle */
269 pthread_mutex_unlock(&q_lock);
270 
271 /* evidence for the interviewer: how long stage 1 was away from the BMS */
272 uint64_t d = bms_now_us() - t0;
273 if (d > atomic_load(&c_max_publish_us))
274 atomic_store(&c_max_publish_us, d); /* single writer: no CAS needed */
275}

277static void *sampler_main(void *arg)
278{
279 (void)arg;
280 while (atomic_load(&running)) {
281 struct BmsSample s = bms_read(); /* the only caller, ever */
282 /* Publish even if deinit flipped the flag while we were parked in the
283 * vendor call: that reading cost a full BMS interval and it is the
284 * newest thing anyone will ever know about the pack. The loop condition
285 * is what ends the thread, not a mid-loop bail-out that throws data
286 * away. (Nothing downstream minds: the queue push cannot block, and the
287 * uploader is only joined after this thread is.) */
288 if (s.status == BMS_OK) {
289 publish(&s);
290 } else if (s.status == BMS_STALE) {
291 /* nothing changed: volts/amps/timestamp are NOT valid, so touching
292 * them would publish garbage. The previous sample still stands. */
293 atomic_fetch_add(&c_stale, 1);
294 } else {
295 /* BMS_ERROR: every field is garbage and it returned instantly, so
296 * a tight retry loop would spin a core. Back off a little. */
297 atomic_fetch_add(&c_errors, 1);
298 usleep(2000);
299 }
300 }
301 return NULL;
302}

308/* Continuous refill: tokens += elapsed * rate, capped at the bucket size.
309 * Capping is what makes it a *bucket*: an uploader that was idle for a minute
310 * may not then fire sixty uploads back to back. */
311static void bucket_refill_locked(uint64_t now)
312{
313 if (now > tokens_last_us) {
314 tokens += (double)(now - tokens_last_us) * (double)UPLOAD_RATE_PER_SEC / 1e6;
315 if (tokens > (double)UPLOAD_BURST) tokens = (double)UPLOAD_BURST;
316 tokens_last_us = now;
317 }
318}

320static void *uploader_main(void *arg)
321{
322 (void)arg;
323 struct BmsSample batch[UPLOAD_BATCH_MAX];
324 
325 pthread_mutex_lock(&q_lock);
326 while (atomic_load(&running)) {
327 
328 /* (1) nothing to do: plain wait, no timeout - do not burn a token or a
329 * timer when the queue is empty. */
330 if (q_count == 0) {
331 pthread_cond_wait(&q_cv, &q_lock);
332 continue;
333 }
334 
335 /* (2) work is waiting: buy a token. This is the throttle, and it is a
336 * TIMED wait so that (a) we sleep exactly until the bucket has a
337 * token instead of polling, and (b) deinit can wake us instantly
338 * with a broadcast instead of us waiting the interval out. */
339 uint64_t wait_t0 = bms_now_us();
340 for (;;) {
341 bucket_refill_locked(bms_now_us());
342 if (tokens >= 1.0) break;
343 uint64_t need = (uint64_t)((1.0 - tokens) * (double)TOKEN_INTERVAL_US) + 1;
344 cond_wait_us(&q_cv, &q_lock, need);
345 if (!atomic_load(&running)) break;
346 }
347 if (!atomic_load(&running)) break;
348 uint64_t waited = bms_now_us() - wait_t0;
349 tokens -= 1.0;
350 
351 /* (3) take a batch out of the queue while we still hold the lock */
352 int n = 0;
353 while (n < UPLOAD_BATCH_MAX && q_count > 0) {
354 batch[n++] = q[q_head];
355 q_head = (q_head + 1u) % UPLOAD_QUEUE_CAP;
356 q_count--;
357 }
358 pthread_mutex_unlock(&q_lock);
359 
360 /* (4) the network call goes OUTSIDE the lock. bms_upload() blocks for
361 * milliseconds; holding q_lock across it would make the sampler's
362 * push - and therefore the BMS reading - wait on the network. */
363 atomic_fetch_add(&c_wait_us, waited);
364 if (n > 0 && bms_upload(batch, n) == 0) {
365 atomic_fetch_add(&c_uploaded, (uint64_t)n);
366 atomic_fetch_add(&c_batches, 1);
367 }
368 
369 pthread_mutex_lock(&q_lock);
370 }
371 pthread_mutex_unlock(&q_lock);
372 return NULL;
373}

379int bms_pipeline_init(void)
380{
381 pthread_mutex_lock(&q_lock);
382 q_head = q_count = 0;
383 tokens = (double)UPLOAD_BURST; /* start full: one burst is allowed */
384 tokens_last_us = bms_now_us();
385 pthread_mutex_unlock(&q_lock);
386 
387 atomic_store(&running, true);
388 
389 /* The vendor's FIRST call returns immediately, so take it here, on the
390 * caller's thread, before any thread exists. A getter called one
391 * microsecond after init then already has a real value instead of -1, and
392 * we did not have to block to get it. (Two threads never touch bms_read():
393 * this call happens-before pthread_create.) */
394 struct BmsSample s = bms_read();
395 if (s.status == BMS_OK) publish(&s);
396 else if (s.status == BMS_STALE) atomic_fetch_add(&c_stale, 1);
397 else atomic_fetch_add(&c_errors, 1);
398 
399 if (pthread_create(&sampler_th, NULL, sampler_main, NULL) != 0) {
400 atomic_store(&running, false);
401 return -1;
402 }
403 if (pthread_create(&uploader_th, NULL, uploader_main, NULL) != 0) {
404 atomic_store(&running, false);
405 pthread_cond_broadcast(&q_cv);
406 pthread_join(sampler_th, NULL);
407 return -1;
408 }
409 threads_up = true;
410 return 0;
411}

413void bms_pipeline_deinit(void)
414{
415 if (!threads_up) return;
416 threads_up = false;
417 
418 atomic_store(&running, false);
419 
420 /* Wake the uploader wherever it is: idle in pthread_cond_wait(), or parked
421 * in the token bucket's timed wait. Without this broadcast deinit would
422 * block for up to one full token interval. The signal has to be sent with
423 * the lock held, otherwise the uploader can re-check `running` (still true),
424 * decide to wait, and only then miss the wakeup - the classic lost-wakeup
425 * race between "set the flag" and "signal the condvar". */
426 pthread_mutex_lock(&q_lock);
427 pthread_cond_broadcast(&q_cv);
428 pthread_mutex_unlock(&q_lock);
429 
430 /* The sampler may be parked inside bms_read() for up to one BMS interval;
431 * nothing can shorten that, which is exactly why the vendor call lives on
432 * its own thread and not in a getter. */
433 pthread_join(sampler_th, NULL);
434 pthread_join(uploader_th, NULL);
435 
436 /* History and counters are deliberately kept: energy_joules_between() still
437 * answers after shutdown, and the counters still explain what happened. */
438}

445{
446 if (out == NULL) return -1;

448 pthread_mutex_lock(&latest_lock);
449 if (latest_valid) {
450 *out = latest; /* one struct copy: readers never see a mix of

459struct BmsStats bms_stats(void)
460{
461 struct BmsStats s;
462 /* Eight independent atomic loads are not one atomic snapshot, so the READ
463 * ORDER is chosen to keep the invariant that matters true: read the things
464 * that only ever go up *after* `sampled` does first, and read `sampled`
465 * last. Then sampled is at least as new as uploaded+dropped and
466 * "sampled >= uploaded + dropped" can never appear violated. */
467 s.uploaded = atomic_load(&c_uploaded);
468 s.dropped = atomic_load(&c_dropped);
469 s.batches = atomic_load(&c_batches);
470 s.upload_wait_us = atomic_load(&c_wait_us);
471 s.max_publish_us = atomic_load(&c_max_publish_us);
472 s.stale = atomic_load(&c_stale);
473 s.errors = atomic_load(&c_errors);
474 s.sampled = atomic_load(&c_sampled);
475 return s;
476}

478/* ------------------------------------------------------------------ */
479/* Part 2: energy over an arbitrary interval */
480/* ------------------------------------------------------------------ */
481 
482/* largest i with H(i)->ts <= t, assuming H(0)->ts <= t. Caller holds hist_lock.
483 * Plain lower-bound binary search: O(log n) instead of walking 90 000 samples,
484 * which is the whole reason the ring is kept sorted. */
485static unsigned floor_index(uint64_t t)
486{
487 unsigned lo = 0, hi = hist_count - 1; /* invariant: H(lo)->ts <= t */
488 while (lo < hi) {
489 unsigned mid = lo + (hi - lo + 1u) / 2u; /* bias up: lo may equal hi-1 */
490 if (H(mid)->ts <= t) lo = mid;
491 else hi = mid - 1u;
492 }
493 return lo;
494}
495 
496/* power at t inside segment [i, i+1], linearly interpolated. Caller holds the
497 * lock and guarantees i + 1 < hist_count. */
498static double p_at(unsigned i, uint64_t t)
499{
500 struct hist_entry *a = H(i), *b = H(i + 1);
501 if (t <= a->ts) return a->p;
502 if (t >= b->ts) return b->p;
503 return a->p + (b->p - a->p) * ((double)(t - a->ts) / (double)(b->ts - a->ts));
504}

506double energy_joules_between(uint64_t t0, uint64_t t1)
507{
508 if (t1 <= t0) return 0.0; /* empty or inverted range */
509 
510 pthread_mutex_lock(&hist_lock);
511 
512 if (hist_count < 2) { /* one point has no interval */
513 pthread_mutex_unlock(&hist_lock);
514 return 0.0;
515 }
516 
517 uint64_t first = H(0)->ts, last = H(hist_count - 1)->ts;
518 /* Clamp to what we actually measured. Energy before the oldest retained
519 * sample or after the newest one is unknown, and inventing it (e.g. holding
520 * the last power forever) would quietly turn a gap into a number. */
521 if (t0 < first) t0 = first;
522 if (t1 > last) t1 = last;
523 if (t1 <= t0) { /* range was entirely outside */
524 pthread_mutex_unlock(&hist_lock);
525 return 0.0;
526 }
527 
528 /* t0 < t1 <= last, so t0 < last, so i <= hist_count - 2: segment [i, i+1]
529 * exists and contains t0. Same for j with t1. */
530 unsigned i = floor_index(t0);
531 unsigned j = floor_index(t1);
532 
533 double e;
534 if (i == j) {
535 /* BOTH ENDS IN ONE SEGMENT. There is no whole segment to sum and no
536 * prefix sum to subtract: the answer is a single trapezoid whose two
537 * parallel sides are the INTERPOLATED powers at t0 and t1. Getting this
538 * case wrong (e.g. returning cum_j - cum_i = 0) is the classic bug:
539 * short queries would silently return zero. */
540 e = 0.5 * (p_at(i, t0) + p_at(i, t1)) * ((double)(t1 - t0) / 1e6);
541 } else {
542 /* head sliver: from t0 to the next real sample */
543 struct hist_entry *ip1 = H(i + 1);
544 double head = 0.5 * (p_at(i, t0) + ip1->p)
545 * ((double)(ip1->ts - t0) / 1e6);
546 
547 /* everything whole in between, in one subtraction: O(1) */
548 double mid = H(j)->cum - ip1->cum;
549 
550 /* tail sliver: from the last sample at or before t1 up to t1. Zero when
551 * t1 landed exactly on a sample (including t1 == last, where segment
552 * [j, j+1] does not exist - hence the guard). */
553 double tail = 0.0;
554 struct hist_entry *jp = H(j);
555 if (t1 > jp->ts && j + 1 < hist_count)
556 tail = 0.5 * (jp->p + p_at(j, t1)) * ((double)(t1 - jp->ts) / 1e6);
557 
558 e = head + mid + tail;
559 }
560 
561 pthread_mutex_unlock(&hist_lock);
562 return e;
