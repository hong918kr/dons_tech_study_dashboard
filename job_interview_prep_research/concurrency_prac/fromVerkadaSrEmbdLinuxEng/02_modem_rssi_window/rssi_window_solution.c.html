1/* rssi_window_solution.c — reference solution (Part 1 + Part 2).
2 *
3 * ./main.sh sol
4 *
5 * Shape
6 * one sampler thread ── the ONLY caller of modem_poll_rssi() (it blocks up to
7 * 100 ms and keeps static state, so it must be one
8 * thread and it must not be a thread a caller owns).
9 * │ MODEM_OK
10 * ├──► g_pub (Part 1) ONE _Atomic uint64_t = 48-bit timestamp
11 * │ (us since init) | 16-bit signed rssi_dbm.
12 * └──► g_win (Part 2) ring of (ts, rssi) + two monotonic deques,
13 * guarded by a mutex held for amortised O(1).
14 *
15 * WHY PACK VALUE AND TIMESTAMP INTO ONE ATOMIC (Part 1)
16 * The contract is "value and its age must describe the same sample". With two
17 * separate variables (an atomic int and an atomic uint64) a reader can load
18 * the value, get preempted, and load a timestamp belonging to the NEXT sample:
19 * a stale -113 dBm then looks 0 ms old, which is exactly the failure this API
20 * exists to prevent (the LED shows full bars while the radio is dead, or the
21 * failover logic trusts a nine-minute-old number). Publishing both halves in
22 * ONE atomic store makes the pair indivisible: the reader does a single
23 * 64-bit load, so it is wait-free, needs no lock, and cannot tear.
24 * Budget: 16 bits is plenty for dBm (-32768..32767 covers -115..-55 with room
25 * to spare) and 48 bits of microseconds since init is 8.9 years of uptime.
26 * If the payload ever grew past 64 bits (say rssi + sinr + band + ts) the
27 * packing stops working and the next tool is a SEQLOCK — writer bumps an even
28 * sequence to odd, writes the struct, bumps to even; reader retries while the
29 * sequence is odd or changed. Same wait-free-ish reader, no mutex, but the
30 * reader can spin, so it is strictly more expensive. A mutex would also be
31 * correct here and is ~20 ns, but then a reader can be descheduled holding it
32 * and every LED tick waits on it; an atomic load cannot be blocked at all.
33 *
34 * WHY MONOTONIC DEQUES (Part 2)
35 * min/max over a sliding window by scanning is O(n) per query. n is not small:
36 * the modem can produce a value every 10 ms, so a 60 s window is 6000 samples,
37 * and the installer UI asks for min and max several times a second while the
38 * sampler appends — that is millions of comparisons per minute inside a lock
39 * that the radio task also needs. The standard fix is one deque per statistic
40 * holding sample INDICES, kept monotonic: for the minimum, values along the
41 * deque are non-decreasing. A new sample smaller than the back of the deque
42 * deletes it permanently — a larger, OLDER sample can never be the minimum
43 * again, because it expires no later than the new one. The front is therefore
44 * always the answer; expiry pops the front. Each sample is pushed once and
45 * popped at most once, so a query is amortised O(1) and the worst case is a
46 * burst of pops that the appends already paid for.
47 * The average is a running int64 sum / count, also O(1). Because rssi_dbm is
48 * an integer the running sum is EXACT; if the samples were floats (like the
49 * ALS lux problem) an incrementally updated sum would accumulate rounding
50 * error that never washes out — add/subtract of unequal magnitudes is lossy —
51 * and the fixes are periodic recomputation from the ring or Kahan summation.
52 *
53 * Eviction happens both on append AND inside every getter: samples age out even
54 * when the modem has stopped talking, and that is precisely the case the
55 * staleness API exists for. Unlike the ALS "value in effect at time t" problem
56 * there is no keep-one-older rule here: a sample outside the window is simply
57 * not part of the window's statistics.
58 */

75/* Capacity: worst case the modem hands us a new value every 10 ms -> 100 Hz.
76 * 60 s * 100 Hz = 6000 -> 8192 slots. 8192*16 B ring + 2*8192*4 B deques
77 * = 192 KiB of BSS, fine on a gateway, and nothing is ever malloc'd. */
78#ifndef RSSI_CAP
79#define RSSI_CAP 8192u
80#endif
81_Static_assert((RSSI_CAP & (RSSI_CAP - 1u)) == 0u, "RSSI_CAP must be a power of two");
82#define CAP_MASK (RSSI_CAP - 1u)
83 
84#define ERROR_BACKOFF_US 5000u

86/* ------------------------------------------------------------------ */
87/* Part 1 state: the packed publication word */
88/* ------------------------------------------------------------------ */
89 
90#define PUB_NONE 0ull /* rel is >= 1 for real samples, see below */
91#define PUB_RSSI_BITS 16
92#define PUB_RSSI_MASK 0xFFFFull
93 
94static _Atomic uint64_t g_pub = PUB_NONE;
95static _Atomic uint64_t g_t0; /* us; timestamps are published relative to it */
96 
97static void publish(uint64_t ts, int rssi_dbm)
98{
99 uint64_t t0 = atomic_load_explicit(&g_t0, memory_order_relaxed);
100 uint64_t rel = (ts > t0) ? (ts - t0) : 0;
101 rel += 1; /* so a real sample is never 0 */
102 if (rel > (UINT64_MAX >> PUB_RSSI_BITS)) /* 8.9 years of uptime; clamp */
103 rel = UINT64_MAX >> PUB_RSSI_BITS;
104 
105 uint64_t word = (rel << PUB_RSSI_BITS) | (uint64_t)(uint16_t)(int16_t)rssi_dbm;
106 /* release: the ring/deque writes done before this store are visible to any
107 * reader that observes the new word (not strictly needed for Part 1, which
108 * is self-contained, but it makes g_t0 safe to read after a non-NONE load). */
109 atomic_store_explicit(&g_pub, word, memory_order_release);
110}

116typedef struct { uint64_t ts; int rssi; } sample_t;
117 
118static struct {
119 pthread_mutex_t lock;
120 sample_t s[RSSI_CAP];
121 uint32_t head, tail; /* free-running; count = head - tail */
122 uint32_t minq[RSSI_CAP]; /* sample indices, rssi non-decreasing */
123 uint32_t min_f, min_b;
124 uint32_t maxq[RSSI_CAP]; /* sample indices, rssi non-increasing */
125 uint32_t max_f, max_b;
126 int64_t sum; /* exact: rssi_dbm is an integer */
127 uint32_t n_overflow;
128} g_win = { .lock = PTHREAD_MUTEX_INITIALIZER };
129 
130static _Atomic uint32_t g_errors;
131static _Atomic bool g_running;
132static bool g_started; /* only ever touched by init/deinit */
133static pthread_t g_thread;

135/* all of the helpers below run with g_win.lock held */

137static uint32_t win_count(void) { return g_win.head - g_win.tail; }
138 
139static void drop_oldest(void)
140{
141 uint32_t idx = g_win.tail;
142 g_win.sum -= g_win.s[idx & CAP_MASK].rssi;
143 g_win.tail++;
144 /* deque indices are strictly increasing and all >= tail, so only the front
145 * can be the sample we just dropped. */
146 if (g_win.min_f != g_win.min_b && g_win.minq[g_win.min_f & CAP_MASK] == idx)
147 g_win.min_f++;
148 if (g_win.max_f != g_win.max_b && g_win.maxq[g_win.max_f & CAP_MASK] == idx)
149 g_win.max_f++;
150}
151 
152/* Amortised O(1): each sample can be dropped once, ever. */
153static void evict_expired(uint64_t now)
154{
155 while (g_win.head != g_win.tail) {
156 uint64_t ts = g_win.s[g_win.tail & CAP_MASK].ts;
157 if (ts > now || now - ts <= RSSI_WINDOW_US)
158 break;
159 drop_oldest();
160 }
161}

163static void win_append(uint64_t ts, int rssi)
164{
165 uint64_t now = modem_now_us();
166 
167 pthread_mutex_lock(&g_win.lock);
168 
169 /* gettimeofday can step backwards (NTP). The ring must stay sorted by ts or
170 * eviction from the front stops being correct, so clamp. */
171 if (win_count() > 0) {
172 uint64_t last = g_win.s[(g_win.head - 1u) & CAP_MASK].ts;
173 if (ts < last)
174 ts = last;
175 }
176 
177 evict_expired(now);
178 
179 if (win_count() == RSSI_CAP) { /* modem chattier than we sized for: */
180 drop_oldest(); /* prefer recent data over old data */
181 g_win.n_overflow++;
182 }
183 
184 uint32_t idx = g_win.head;
185 g_win.s[idx & CAP_MASK] = (sample_t){ .ts = ts, .rssi = rssi };
186 g_win.head++;
187 g_win.sum += rssi;
188 
189 /* min deque: drop every back entry that this sample makes useless. It is
190 * >= rssi and older, so it expires first -- it can never be the minimum
191 * again while this sample is alive. */
192 while (g_win.min_f != g_win.min_b &&
193 g_win.s[g_win.minq[(g_win.min_b - 1u) & CAP_MASK] & CAP_MASK].rssi >= rssi)
194 g_win.min_b--;
195 g_win.minq[g_win.min_b & CAP_MASK] = idx;
196 g_win.min_b++;
197 
198 /* max deque: mirror image */
199 while (g_win.max_f != g_win.max_b &&
200 g_win.s[g_win.maxq[(g_win.max_b - 1u) & CAP_MASK] & CAP_MASK].rssi <= rssi)
201 g_win.max_b--;
202 g_win.maxq[g_win.max_b & CAP_MASK] = idx;
203 g_win.max_b++;
204 
205 pthread_mutex_unlock(&g_win.lock);
206}

212static void handle(const RssiSample *r)
213{
214 switch (r->status) {
215 case MODEM_OK:
216 publish(r->timestamp, r->rssi_dbm); /* cheapest, and most readers only want this */
217 win_append(r->timestamp, r->rssi_dbm);
218 break;
219 case MODEM_BUSY:
220 break; /* no new value; the published one still holds */
221 default: /* MODEM_ERROR: payload is garbage */
222 atomic_fetch_add_explicit(&g_errors, 1u, memory_order_relaxed);
223 usleep(ERROR_BACKOFF_US); /* it returns instantly -- do not spin the CPU */
224 break;
225 }
226}

228static void *sampler_main(void *arg)
229{
230 (void)arg;
231 while (atomic_load_explicit(&g_running, memory_order_acquire)) {
232 RssiSample r = modem_poll_rssi();
233 handle(&r);
234 }
235 return NULL;
236}

242int rssi_init(void)
243{
244 if (g_started)
245 return 0;
246 
247 atomic_store_explicit(&g_t0, modem_now_us(), memory_order_relaxed);
248 
249 /* The first poll never blocks, so take it here: a getter called the
250 * instant init() returns already has a value. Retry a couple of times in
251 * case it came back ERROR. */
252 for (int tries = 0; tries < 3; tries++) {
253 RssiSample r = modem_poll_rssi();
254 handle(&r);
255 if (r.status == MODEM_OK)
256 break;
257 }
258 
259 atomic_store_explicit(&g_running, true, memory_order_release);
260 if (pthread_create(&g_thread, NULL, sampler_main, NULL) != 0) {
261 atomic_store_explicit(&g_running, false, memory_order_release);
262 return -1;
263 }
264 g_started = true;
265 return 0;
266}
267 
268void rssi_deinit(void)
269{
270 if (!g_started)
271 return;
272 /* No cancellation, no signals: just ask, then wait for the thread to come
273 * back out of the vendor call (<= ~100 ms) and re-test the flag. */
274 atomic_store_explicit(&g_running, false, memory_order_release);
275 pthread_join(g_thread, NULL);
276 g_started = false;
277}

283int rssi_get_latest(int *dbm, uint64_t *age_us)
284{
285 uint64_t word = atomic_load_explicit(&g_pub, memory_order_acquire);
286 if (word == PUB_NONE)
287 return -1; /* nothing valid has ever been read */
288 
289 int value = (int)(int16_t)(uint16_t)(word & PUB_RSSI_MASK);
290 uint64_t ts = atomic_load_explicit(&g_t0, memory_order_relaxed)
291 + (word >> PUB_RSSI_BITS) - 1u;
292 uint64_t now = modem_now_us();
293 
294 if (dbm)
295 *dbm = value;
296 if (age_us)
297 *age_us = (now > ts) ? now - ts : 0; /* clock stepped back -> "just now" */
298 return 0;
299}

301bool rssi_is_healthy(uint64_t max_age_us)
302{
303 uint64_t age;
304 if (rssi_get_latest(NULL, &age) != 0)
305 return false; /* never sampled: not healthy */
306 return age <= max_age_us;
307}

313int rssi_min_in_window(void)
314{
315 uint64_t now = modem_now_us();
316 pthread_mutex_lock(&g_win.lock);
317 evict_expired(now); /* samples age out with no new input */
318 int r = (g_win.min_f == g_win.min_b)
319 ? RSSI_NONE
320 : g_win.s[g_win.minq[g_win.min_f & CAP_MASK] & CAP_MASK].rssi;
321 pthread_mutex_unlock(&g_win.lock);
322 return r;
323}

337double rssi_avg_in_window(void)
338{
339 uint64_t now = modem_now_us();
340 pthread_mutex_lock(&g_win.lock);
341 evict_expired(now);
342 uint32_t n = win_count();
343 double r = n ? (double)g_win.sum / (double)n : NAN;
344 pthread_mutex_unlock(&g_win.lock);
345 return r;
