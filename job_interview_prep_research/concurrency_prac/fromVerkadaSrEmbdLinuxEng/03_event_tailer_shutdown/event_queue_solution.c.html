1/* event_queue_solution.c — reference solution (Part 1 + Part 2).
2 *
3 * ./main.sh sol normal run
4 * ./main.sh fast sol 20 ms buckets / 32 buckets, so eviction really happens
5 *
6 * Shape
7 * one sampler thread ── the ONLY caller of evsrc_read_blocking(). It has to
8 * be exactly one: the vendor call is not thread-safe
9 * and it blocks with no timeout, so it cannot live on
10 * a consumer's thread.
11 * │ event
12 * ├──► g_cnt (Part 2) ring of EVQ_NBUCKETS fixed-width time buckets,
13 * │ per-type counters, advanced lazily. O(1) insert,
14 * │ O(buckets) query, memory independent of rate.
15 * └──► g_q (Part 1) bounded ring of EVQ_CAPACITY events + condvar.
16 * Full => drop the OLDEST and bump a counter.
17 * consumers call evq_try_pop / evq_pop_timed / evq_count_since from any
18 * number of threads and never touch the vendor.
19 *
20 * Why these choices
21 * - mutex + condvar, not lock-free: a consumer must be able to *wait* for an
22 * event (evq_pop_timed), and a condvar is the only cheap way to sleep and
23 * be woken. The critical sections are a struct copy and a bounded loop, so
24 * the worst getter latency stays microseconds against a vendor that blocks
25 * for minutes.
26 * - two independent locks: a counting query must not queue up behind a
27 * consumer that is draining events, and vice versa. They protect
28 * unrelated state, so keeping them separate costs nothing.
29 * - drop-OLDEST on overflow: for a live gateway the newest events are the
30 * ones worth acting on, and dropping oldest keeps a slow consumer from
31 * blocking the sampler (which would block the vendor bridge). Nothing is
32 * silently lost: the drop is counted, and the *history* in Part 2 is
33 * recorded before the queue, so reporting stays correct even while the
34 * queue is overflowing.
35 * - shutdown: the sampler is parked in a call with no timeout, so a flag is
36 * not enough. Order is: stop flag -> evsrc_wake() (the vendor's escape
37 * hatch) -> wake any blocked consumer -> join. See evq_deinit().
38 */

64static struct {
65 pthread_mutex_t lock;
66 pthread_cond_t cv; /* signalled on every push, broadcast on stop */
67 struct Event buf[EVQ_CAPACITY];
68 uint32_t head; /* free-running: next slot to write */
69 uint32_t tail; /* free-running: oldest unread slot */
70 bool stopping; /* true before init and after deinit */
71 uint64_t received, dropped, popped, errors;
72} g_q = {
73 .lock = PTHREAD_MUTEX_INITIALIZER,
74 .cv = PTHREAD_COND_INITIALIZER,
75 .stopping = true, /* a getter before evq_init() must fail, not crash */
76};
77 
78/* head/tail are free-running counters, so the difference is the depth and
79 * uint32 wrap-around is harmless as long as capacity is a power of two. */
80static uint32_t q_depth_locked(void) { return g_q.head - g_q.tail; }
81 
82static void q_push(const struct Event *e)
83{
84 pthread_mutex_lock(&g_q.lock);
85 g_q.received++;
86 if (q_depth_locked() == (uint32_t)EVQ_CAPACITY) {
87 g_q.tail++; /* drop-oldest: the slot is about to be reused */
88 g_q.dropped++;
89 }
90 g_q.buf[g_q.head & Q_MASK] = *e;
91 g_q.head++;
92 /* signal, not broadcast: one event can only satisfy one waiter. */
93 pthread_cond_signal(&g_q.cv);
94 pthread_mutex_unlock(&g_q.lock);
95}
96 
97static bool q_pop_locked(struct Event *out)
98{
99 if (q_depth_locked() == 0u)
100 return false;
101 *out = g_q.buf[g_q.tail & Q_MASK];
102 g_q.tail++;
103 g_q.popped++;
104 return true;
105}

107int evq_try_pop(struct Event *out)
108{
109 if (!out)
110 return -1;
111 pthread_mutex_lock(&g_q.lock);
112 bool got = !g_q.stopping && q_pop_locked(out);
113 pthread_mutex_unlock(&g_q.lock);
114 return got ? 0 : -1;
115}

117/* Wait on cv until an absolute microsecond deadline.
118 *
119 * Portability: pthread_condattr_setclock() is not available on macOS, so the
120 * two platforms are split. The deadline is absolute on purpose — a relative
121 * wait restarted after a spurious wakeup would stretch the timeout. */
122static int cond_wait_until(uint64_t deadline_us)
123{
124 uint64_t now = ev_now_us();
125 if (now >= deadline_us)
126 return ETIMEDOUT;
127 uint64_t left = deadline_us - now;
128 
129#ifdef __APPLE__
130 struct timespec rel;
131 rel.tv_sec = (time_t)(left / 1000000ull);
132 rel.tv_nsec = (long)((left % 1000000ull) * 1000ull);
133 return pthread_cond_timedwait_relative_np(&g_q.cv, &g_q.lock, &rel);
134#else
135 /* ev_now_us() is gettimeofday(), i.e. CLOCK_REALTIME — the same clock the
136 * default condattr uses, so the absolute deadline needs no conversion. */
137 struct timespec abs;
138 abs.tv_sec = (time_t)(deadline_us / 1000000ull);
139 abs.tv_nsec = (long)((deadline_us % 1000000ull) * 1000ull);
140 return pthread_cond_timedwait(&g_q.cv, &g_q.lock, &abs);
141#endif
142}

144int evq_pop_timed(struct Event *out, uint64_t timeout_us)
145{
146 if (!out)
147 return -1;
148 uint64_t deadline = ev_now_us() + timeout_us;
149 
150 pthread_mutex_lock(&g_q.lock);
151 for (;;) {
152 if (g_q.stopping)
153 break; /* deinit released us */
154 if (q_pop_locked(out)) {
155 pthread_mutex_unlock(&g_q.lock);
156 return 0;
157 }
158 if (timeout_us == 0)
159 break; /* degenerate case == evq_try_pop */
160 /* Loop, do not trust a single wakeup: condvars wake spuriously, and
161 * another consumer may have taken the event we were signalled for. */
162 if (cond_wait_until(deadline) == ETIMEDOUT) {
163 if (!g_q.stopping && q_pop_locked(out)) {
164 pthread_mutex_unlock(&g_q.lock);
165 return 0; /* raced in just before the deadline */
166 }
167 break;
168 }
169 }
170 pthread_mutex_unlock(&g_q.lock);
171 return -1;
172}

174void evq_stats(struct EvqStats *out)
175{
176 if (!out)
177 return;
178 /* One lock for all five fields: a caller checking
179 * received == popped + dropped + depth must see a consistent snapshot. */
180 pthread_mutex_lock(&g_q.lock);
181 out->received = g_q.received;
182 out->dropped = g_q.dropped;
183 out->popped = g_q.popped;
184 out->errors = g_q.errors;
185 out->depth = q_depth_locked();
186 pthread_mutex_unlock(&g_q.lock);
187}

189/* ------------------------------------------------------------------ */
190/* Part 2: ring of fixed-width time buckets */
191/* ------------------------------------------------------------------ */
192/*
193 * Why not keep the events themselves?
194 * A per-event list (or the ring from the previous problem) makes
195 * count_since() O(number of events in the window), and the number of events
196 * is exactly the thing we do not control: a tamper storm on 8 doors can
197 * produce thousands of events per second, and a burst is when a monitoring
198 * query is most likely to be running. Memory would also grow with the rate,
199 * and evicting a long backlog under the lock is a latency spike.
200 *
201 * Bucketing flips both costs:
202 * memory = EVQ_NBUCKETS * EV_TYPE_COUNT counters, fixed, whatever the rate
203 * insert = O(1) increment (plus a lazy advance, bounded below)
204 * query = O(EVQ_NBUCKETS) — 300 adds for a 5-minute window, ~1 us
205 * cost is the resolution: the answer is rounded to a whole bucket.
206 *
207 * Lazy advance: buckets are not cleared by a timer. When an event lands in a
208 * newer bucket we zero only the buckets we skipped over — and if we skipped
209 * more than a full lap (a gateway that saw nothing for an hour) we memset the
210 * whole table once instead of looping over a million stale bucket indices.
211 * That keeps the "first event after a quiet night" from stalling the sampler.
212 */
213 
214static struct {
215 pthread_mutex_t lock;
216 uint32_t n[EVQ_NBUCKETS][EV_TYPE_COUNT];
217 uint64_t cur; /* absolute index of the newest bucket held */
218 bool primed; /* false until the first event */
219 uint64_t unknown; /* events with a type we do not know */
220} g_cnt = { .lock = PTHREAD_MUTEX_INITIALIZER };
221 
222static void cnt_add(const struct Event *e)
223{
224 uint64_t b = e->timestamp / (uint64_t)EVQ_BUCKET_US;
225 
226 pthread_mutex_lock(&g_cnt.lock);
227 if (e->type < 0 || e->type >= EV_TYPE_COUNT) {
228 g_cnt.unknown++; /* never index an array with vendor data */
229 pthread_mutex_unlock(&g_cnt.lock);
230 return;
231 }
232 
233 if (!g_cnt.primed) {
234 g_cnt.cur = b;
235 g_cnt.primed = true;
236 } else if (b > g_cnt.cur) {
237 if (b - g_cnt.cur >= (uint64_t)EVQ_NBUCKETS) {
238 memset(g_cnt.n, 0, sizeof g_cnt.n); /* skipped a whole lap */
239 } else {
240 for (uint64_t k = g_cnt.cur + 1; k <= b; k++)
241 memset(g_cnt.n[k % (uint64_t)EVQ_NBUCKETS], 0, sizeof g_cnt.n[0]);
242 }
243 g_cnt.cur = b;
244 }
245 
246 /* A stamp older than the whole window has no bucket to live in any more.
247 * (b < cur - (NBUCKETS-1), written without underflow.) */
248 if (b + (uint64_t)EVQ_NBUCKETS > g_cnt.cur)
249 g_cnt.n[b % (uint64_t)EVQ_NBUCKETS][e->type]++;
250 
251 pthread_mutex_unlock(&g_cnt.lock);
252}

254uint32_t evq_count_since(int type, uint64_t since_us)
255{
256 if (type != EV_TYPE_ANY && (type < 0 || type >= EV_TYPE_COUNT))
257 return 0;
258 
259 uint64_t now = ev_now_us();
260 if (since_us > now)
261 return 0; /* nothing has happened in the future yet */
262 
263 const uint64_t NB = (uint64_t)EVQ_NBUCKETS;
264 uint64_t nowb = now / (uint64_t)EVQ_BUCKET_US;
265 uint64_t lo = since_us / (uint64_t)EVQ_BUCKET_US;
266 
267 /* Clamp to the window: older than EVQ_WINDOW_US is simply "the window". */
268 uint64_t window_lo = (nowb + 1 >= NB) ? nowb - (NB - 1) : 0;
269 if (lo < window_lo)
270 lo = window_lo;
271 
272 uint32_t total = 0;
273 pthread_mutex_lock(&g_cnt.lock);
274 if (g_cnt.primed) {
275 uint64_t hi = g_cnt.cur < nowb ? g_cnt.cur : nowb; /* no future buckets */
276 uint64_t ring_lo = (g_cnt.cur + 1 >= NB) ? g_cnt.cur - (NB - 1) : 0;
277 if (lo < ring_lo)
278 lo = ring_lo; /* anything older was zeroed and reused */
279 /* At most EVQ_NBUCKETS iterations, whatever the event rate was. */
280 for (uint64_t b = lo; b <= hi; b++) {
281 const uint32_t *row = g_cnt.n[b % NB];
282 if (type == EV_TYPE_ANY) {
283 for (int t = 0; t < EV_TYPE_COUNT; t++)
284 total += row[t];
285 } else {
286 total += row[type];
287 }
288 }
289 }
290 pthread_mutex_unlock(&g_cnt.lock);
291 return total;
292}

298static _Atomic bool g_running;
299static bool g_started; /* only touched by init/deinit */
300static pthread_t g_thread;

302static void *sampler_main(void *arg)
303{
304 (void)arg;
305 while (atomic_load_explicit(&g_running, memory_order_acquire)) {
306 struct Event e;
307 int rc = evsrc_read_blocking(&e);
308 
309 if (rc == -2)
310 break; /* evsrc_wake(): shutdown, nothing in *out */
311 if (!atomic_load_explicit(&g_running, memory_order_acquire))
312 break; /* raced with deinit; drop what we have */
313 if (rc != 0) {
314 pthread_mutex_lock(&g_q.lock);
315 g_q.errors++;
316 pthread_mutex_unlock(&g_q.lock);
317 usleep(ERROR_BACKOFF_US);
318 continue;
319 }
320 
321 /* History before the queue: a full queue must not lose a statistic. */
322 cnt_add(&e);
323 q_push(&e);
324 }
325 return NULL;
326}

332int evq_init(void)
333{
334 if (g_started)
335 return 0;
336 
337 /* Note what is NOT here: a synchronous first read. evsrc_read_blocking()
338 * has no timeout, so priming the state in init could hang the caller for
339 * as long as the site stays quiet. Consumers get "empty" instead, which is
340 * the honest answer. */
341 pthread_mutex_lock(&g_q.lock);
342 g_q.head = g_q.tail = 0;
343 g_q.received = g_q.dropped = g_q.popped = g_q.errors = 0;
344 g_q.stopping = false;
345 pthread_mutex_unlock(&g_q.lock);
346 
347 pthread_mutex_lock(&g_cnt.lock);
348 memset(g_cnt.n, 0, sizeof g_cnt.n);
349 g_cnt.cur = 0;
350 g_cnt.primed = false;
351 g_cnt.unknown = 0;
352 pthread_mutex_unlock(&g_cnt.lock);
353 
354 atomic_store_explicit(&g_running, true, memory_order_release);
355 if (pthread_create(&g_thread, NULL, sampler_main, NULL) != 0) {
356 atomic_store_explicit(&g_running, false, memory_order_release);
357 pthread_mutex_lock(&g_q.lock);
358 g_q.stopping = true;
359 pthread_mutex_unlock(&g_q.lock);
360 return -1;
361 }
362 g_started = true;
363 return 0;
364}

366void evq_deinit(void)
367{
368 if (!g_started)
369 return; /* idempotent */
370 
371 /* THE point of this problem. The sampler is sitting inside a vendor call
372 * with no timeout, so clearing a flag it cannot see is useless — the
373 * thread would only notice after the next event, which may never come.
374 *
375 * 1. publish the stop, so the sampler cannot start another read;
376 * 2. evsrc_wake() — the vendor's escape hatch, which makes the pending
377 * read return -2 (and, because it latches, makes any read that starts
378 * in the gap return -2 too: that closes the "woke it a microsecond too
379 * early" race). If the vendor only affected a *pending* call, we would
380 * have to loop wake + timed-join instead;
381 * 3. release consumers blocked in evq_pop_timed, or they would sit out
382 * their whole timeout on a queue that will never fill again;
383 * 4. join — and only now is it safe to say the thread is gone. */
384 atomic_store_explicit(&g_running, false, memory_order_release);
385 evsrc_wake();
386 
387 pthread_mutex_lock(&g_q.lock);
388 g_q.stopping = true;
389 pthread_cond_broadcast(&g_q.cv);
390 pthread_mutex_unlock(&g_q.lock);
391 
392 pthread_join(g_thread, NULL);
393 g_started = false;
