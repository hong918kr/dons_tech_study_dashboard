44{
45 uint32_t x = *s;

117/* Brute force over the injected samples, mirroring the documented
118 * semantics: inclusive range, restricted to minutes the 1-hour window
119 * still covers. Returns the sample count (0 == "expect -1"). */
120static uint64_t bf_stats(uint64_t t0, uint64_t t1, float *mn, float *mx, double *avg)
121{
122 if (t0 > t1 || inj_n == 0) return 0;
123 
124 uint64_t newest_m = inj[inj_n - 1].ts / TEMP_BUCKET_US;
125 uint64_t oldest_m = (newest_m >= TEMP_NBUCKETS - 1u)
126 ? newest_m - (TEMP_NBUCKETS - 1u) : 0;
127 
128 uint64_t n = 0;
129 double sum = 0.0;
130 float lo = 0.0f, hi = 0.0f;
131 for (int i = 0; i < inj_n; i++) {
132 uint64_t ts = inj[i].ts;
133 uint64_t m = ts / TEMP_BUCKET_US;
134 if (ts < t0 || ts > t1) continue;
135 if (m < oldest_m || m > newest_m) continue;
136 float c = inj[i].c;
137 if (n == 0) { lo = c; hi = c; }
138 else { if (c < lo) lo = c; if (c > hi) hi = c; }
139 n++;
140 sum += (double)c;
141 }
142 if (n == 0) return 0;
143 if (mn) *mn = lo;
144 if (mx) *mx = hi;
145 if (avg) *avg = sum / (double)n;
146 return n;
147}

158typedef struct {
159 pthread_mutex_t lk;
160 pthread_cond_t cv;
161 int need, arrived;
162} barrier_t;
163 
164static void bar_init(barrier_t *b, int need)
165{
166 pthread_mutex_init(&b->lk, NULL);
167 pthread_cond_init(&b->cv, NULL);
168 b->need = need;
169 b->arrived = 0;
170}
171 
172static void bar_wait(barrier_t *b)
173{
174 pthread_mutex_lock(&b->lk);
175 if (++b->arrived >= b->need)
176 pthread_cond_broadcast(&b->cv);
177 else
178 while (b->arrived < b->need)
179 pthread_cond_wait(&b->cv, &b->lk);
180 pthread_mutex_unlock(&b->lk);
181}
182 
183static void bar_destroy(barrier_t *b)
184{
185 pthread_mutex_destroy(&b->lk);
186 pthread_cond_destroy(&b->cv);
187}

212/* Runs one wave of WAVE_N simultaneous temp_get(_, 0). */
213static void run_wave(wave_arg_t out[WAVE_N])
214{
215 barrier_t bar;
216 pthread_t th[WAVE_N];
217 bar_init(&bar, WAVE_N);
218 for (int i = 0; i < WAVE_N; i++) {
219 out[i].bar = &bar;
220 out[i].rc = -999;
221 out[i].v = NAN;
222 out[i].lat_us = 0;
223 pthread_create(&th[i], NULL, wave_thread, &out[i]);
224 }
225 for (int i = 0; i < WAVE_N; i++)
226 pthread_join(th[i], NULL);
227 bar_destroy(&bar);
228}

246static void *stress_thread(void *p)
247{
248 stress_stat_t *s = p;
249 int k = 0;
250 while (!atomic_load(&stress_stop)) {
251 uint64_t max_age;
252 switch ((s->id + k) % 3) {
253 case 0: max_age = 0; break; /* force a refresh */
254 case 1: max_age = 5000; break; /* 5 ms: mostly miss */
255 default: max_age = 3600ull * 1000000ull; break; /* always a hit */
256 }
257 k++;
258 
259 float v = NAN;
260 uint64_t t0 = tempdev_now_us();
261 int rc = temp_get(&v, max_age);
262 uint64_t dt = tempdev_now_us() - t0;
263 if (dt > s->max_get_us) s->max_get_us = dt;
264 s->gets++;
265 if (rc == TEMP_OK) s->hits++;
266 else if (rc == TEMP_STALE) s->stales++;
267 else s->nodata++;
268 /* Any value handed out must be one the probe really produced. */
269 if ((rc == TEMP_OK || rc == TEMP_STALE) && !vt_contains(v))
270 s->bad_value++;
271 
272 float mn = NAN, mx = NAN, av = NAN;
273 t0 = tempdev_now_us();
274 int r2 = temp_range_stats(stress_r0, stress_r1, &mn, &mx, &av);
275 dt = tempdev_now_us() - t0;
276 if (dt > s->max_range_us) s->max_range_us = dt;
277 s->ranges++;
278 if (r2 != 0 || mn != stress_min || mx != stress_max ||
279 !close_enough((double)av, stress_avg))
280 s->bad_range++;
281 }
282 return NULL;
283}

691/* Only one thread may be in tempdev_read() at a time, so this needs no

693 * trips the detector below instead of tripping the sanitizer here. */

697uint64_t tempdev_now_us(void)
698{
699 struct timeval tv;
700 gettimeofday(&tv, NULL);
701 return 1000000ull * (uint64_t)tv.tv_sec + (uint64_t)tv.tv_usec;
702}

712int tempdev_read(float *celsius)
713{
714 unsigned now_inside = atomic_fetch_add(&v_inside, 1u) + 1u;
715 if (now_inside > atomic_load(&v_maxconc))
716 atomic_store(&v_maxconc, now_inside);
717 if (now_inside > 1u)
718 atomic_store(&v_conc_seen, true); /* THE bug this exercise is about */
719 
720 uint64_t seq = atomic_fetch_add(&v_calls, 1u) + 1u;
721 
722 usleep(atomic_load(&v_read_us)); /* the slow I2C transaction */
723 
724 bool nack = atomic_load(&v_force_nack);
725 unsigned every = atomic_load(&v_nack_every);
726 if (!nack && every != 0u && (seq % every) == 0u)
727 nack = true;
728 
729 int rc;
730 if (nack) {
731 rc = -1; /* *celsius untouched */
732 } else {
733 pthread_mutex_lock(&v_rng_lk);
734 float c = 15.0f + 20.0f * xs_u01(&v_rng);
735 pthread_mutex_unlock(&v_rng_lk);
736 vt_record(c); /* harness ground truth */
737 *celsius = c;
738 rc = 0;
739 }
740 
741 atomic_fetch_sub(&v_inside, 1u);
742 return rc;
