1/* watchdog_solution.c — reference solution (Part 1 concurrency + Part 2 min-heap).
2 *
3 * ./main.sh sol
4 *
5 * ============================== WHY THIS SHAPE ==============================
6 *
7 * This problem is the mirror image of the usual "wrap a blocking vendor call"
8 * question: there are MANY WRITERS (every worker thread) and exactly ONE READER
9 * (the monitor thread). So the cost has to be pushed onto the reader.
10 *
11 * 1. WHY HEARTBEATS ARE STORES AND NEVER LOCKS
12 * wd_heartbeat() sits on the hot path of the modem manager, the uploader,
13 * the PoE controller... It is called far more often than it is read. If it
14 * took a mutex then (a) every worker would serialise on one cache line and
15 * one futex, so the watchdog itself would become the scalability bottleneck,
16 * and (b) a worker could be *blocked inside the watchdog* — priority
17 * inversion against the very thread whose liveness we are measuring, and if
18 * the lock holder is descheduled the whole fleet stalls. A heartbeat is
19 * therefore ONE relaxed/release atomic store of a 64-bit timestamp into a
20 * slot owned by that worker: wait-free (bounded number of instructions, no
21 * matter what any other thread does), no allocation, no syscall.
22 * The only "call" inside it is the clock read, which is a vDSO/commpage read
23 * on Linux/macOS — a few nanoseconds of user-space work, no kernel entry.
24 *
25 * 2. WHY THE MONITOR IS THE ONLY READER
26 * Deciding "is this worker late?" needs the *set* of workers, the deadline
27 * table and the fault bookkeeping ("report each death exactly once"). That is
28 * mutable shared state, so it lives behind one mutex — and it is only ever
29 * touched off the hot path: by the monitor once per tick, and by
30 * register/unregister/the two query functions. Writers never take that lock,
31 * so a worker can never be delayed by the monitor and vice versa.
32 * Publication is release/acquire: the worker's store is a release, the
33 * monitor's load is an acquire, which is all the ordering we need because the
34 * timestamp is self-describing (nothing else is published with it).
35 *
36 * 3. FALSE SHARING / CACHE-LINE PADDING
37 * 32 naked `_Atomic uint64_t` would pack 8 per 64-byte line. Eight workers
38 * storing every millisecond would then ping one line between cores: every
39 * store invalidates the other seven workers' copies (MESI RFO traffic), and
40 * the "independent" heartbeats cost as much as a shared lock even though
41 * they are logically disjoint. So each worker's timestamp gets its own
42 * 64-byte, 64-byte-aligned slot (wd_hot_t below) and the cold fields (name,
43 * deadline, fault flag) live in a *separate* array that the writers never
44 * touch. The harness measures this: heartbeats from 4 threads versus the same
45 * 4 threads going through one mutex.
46 *
47 * 4. THE "KICK ONLY IF EVERY WORKER IS HEALTHY" RULE
48 * The classic bug is a thread that does `for(;;){ hwwdt_kick(); sleep(1); }`.
49 * That thread keeps the board alive *precisely while the rest of the system
50 * is dead* — the hardware watchdog has been defeated and turned into a
51 * liveness lie. A hardware watchdog is only worth having if the kick is
52 * CONDITIONAL: the monitor kicks only when every registered worker has
53 * checked in inside its own deadline. If anything is overdue the monitor
54 * reports the fault and then simply stops kicking, and the SoC resets the
55 * board HWWDT_TIMEOUT_US later. Corollary: the monitor must itself be the
56 * only kicker (hwwdt_kick is an MMIO write with no synchronisation of its
57 * own), and the monitor's own liveness is covered by the hardware — if the
58 * monitor dies, nobody kicks, the board resets. That is the desired failure
59 * mode, which is why it is the monitor that owns the kick and not a timer.
60 *
61 * 5. PART 2: MIN-HEAP OVER NEXT-DEADLINE, WITH LAZY UPDATES
62 * A monitor that scans all N workers every tick is O(N) per tick and has to
63 * pick a polling period out of thin air. Instead keep a binary min-heap
64 * keyed by "when does this worker's deadline expire", plus g_pos[] so a
65 * worker's heap slot is found in O(1):
66 * register O(log n) insert
67 * unregister O(log n) remove at known position
68 * heartbeat O(1) ... and it does NOT touch the heap at all
69 * next O(log n) amortised: refresh the root, then read it
70 * overdue O(k) k = entries whose key is already in the past
71 * The heartbeat MUST NOT fix the heap: sifting is O(log n) *and* needs the
72 * lock, which would put a lock back on the hot path (see 1). So the key
73 * stored in the heap is allowed to be stale — but only ever in one direction:
74 *
75 * INVARIANT: heap key <= that worker's true deadline
76 * (last_seen only moves forward, so the true deadline only
77 * moves forward; the key is a LOWER BOUND.)
78 *
79 * That one-sidedness is what makes lazy updates correct. To answer "what is
80 * the earliest true deadline?" you only have to repair the ROOT:
81 * while (true_deadline(root) > root.key) { root.key = true; sift_down; }
82 * When it stops, the root's key is exact, and for every other entry i the
83 * heap property gives key_i >= key_root and the invariant gives
84 * true_i >= key_i >= key_root. So the repaired root IS the global minimum —
85 * without looking at anybody else. Each repair pushes one entry down
86 * (O(log n)) and can only happen once per heartbeat, so the amortised cost is
87 * tiny and it is paid by the reader, never by the worker.
88 *
89 * ALTERNATIVE: a timer wheel (hashed buckets of `deadline / tick`, as in the
90 * Linux kernel's classic timer_list or in a network stack's retransmit
91 * timers). It gives O(1) insert/remove/expire instead of O(log n) and is the
92 * better choice when there are thousands of timers, when they are constantly
93 * rearmed (exactly what heartbeats are!) and when a tick of slack is
94 * acceptable. Its costs: memory proportional to the wheel, quantisation to
95 * the tick, cascading between wheel levels, and an awkward answer to "what is
96 * the exact next expiry?" — you have to walk buckets. Here n <= 32, we want
97 * an exact next-deadline so the monitor can sleep precisely, and rearming is
98 * free because of the lazy trick, so the heap wins on simplicity. Third
99 * option, a sorted list: O(1) peek but O(n) insert and horrible under churn.
100 *
101 * 6. CLOCK CAVEAT
102 * hwwdt_now_us() is gettimeofday(), i.e. CLOCK_REALTIME, because that is the
103 * clock a default-initialised condvar waits on and macOS has no
104 * pthread_condattr_setclock(). It is NOT monotonic: an NTP step backwards
105 * makes every deadline look further away (we under-report deaths for the
106 * length of the step) and a step forwards can declare every worker dead at
107 * once and stop the kicks, i.e. reset the board. In production this belongs
108 * on CLOCK_MONOTONIC with a condvar created with that clock (Linux) or with
109 * pthread_cond_timedwait_relative_np() plus a monotonic clock (macOS); the
110 * relative wait below is already immune, only the stored timestamps are not.
111 */

182static void heap_swap(int a, int b)
183{
184 wd_node_t t = g_heap[a];
185 g_heap[a] = g_heap[b];
186 g_heap[b] = t;
187 g_pos[g_heap[a].id] = a;
188 g_pos[g_heap[b].id] = b;
189}

191static void sift_up(int i)
192{
193 while (i > 0) {
194 int p = (i - 1) / 2;
195 if (g_heap[p].key <= g_heap[i].key) break;
196 heap_swap(p, i);
197 i = p;
198 }
199}
200 
201static void sift_down(int i)
202{
203 for (;;) {
204 int l = 2 * i + 1, r = l + 1, m = i;
205 if (l < g_heap_n && g_heap[l].key < g_heap[m].key) m = l;
206 if (r < g_heap_n && g_heap[r].key < g_heap[m].key) m = r;
207 if (m == i) return;
208 heap_swap(m, i);
209 i = m;
210 }
211}

213static void heap_insert(uint32_t id, uint64_t key)
214{
215 int i = g_heap_n++;
216 g_heap[i].id = id;
217 g_heap[i].key = key;
218 g_pos[id] = i;
219 sift_up(i);
220}

222/* O(log n) thanks to g_pos: no search for the victim. */
223static void heap_remove(uint32_t id)
224{
225 int i = g_pos[id];
226 if (i < 0) return;
227 g_pos[id] = -1;
228 int last = --g_heap_n;
229 if (i != last) {
230 g_heap[i] = g_heap[last];
231 g_pos[g_heap[i].id] = i;
232 sift_down(i); /* the replacement may belong lower ... */
233 sift_up(i); /* ... or higher; only one of the two moves it */
234 }
235}

237static uint64_t true_deadline(uint32_t id)
238{
239 /* acquire pairs with the worker's release store in wd_heartbeat() */
240 return atomic_load_explicit(&g_hot[id].last_seen_us, memory_order_acquire)
241 + g_cold[id].deadline_us;
242}

244/* The lazy-update heart of Part 2.
245 *
246 * Repair the root until its key equals its true deadline. Because every key is
247 * a lower bound on its worker's true deadline, an exact root is the exact global
248 * minimum (see note 5 at the top). Returns false only if the heap is empty.
249 *
250 * Side effect on purpose: a worker we find to be healthy again gets its
251 * `faulted` flag cleared here, which is what makes "report each death exactly
252 * once, and report it again if it dies again" work.
253 */
254static bool heap_refresh_root(uint64_t now, uint64_t *out_key, uint32_t *out_id)
255{
256 while (g_heap_n > 0) {
257 uint32_t id = g_heap[0].id;
258 uint64_t truth = true_deadline(id);
259 
260 if (truth > now && g_cold[id].faulted)
261 g_cold[id].faulted = false; /* it came back */
262 
263 if (truth <= g_heap[0].key) { /* key is exact */
264 *out_key = g_heap[0].key;
265 *out_id = id;
266 return true;
267 }
268 g_heap[0].key = truth; /* keys only grow ... */
269 sift_down(0); /* ... so: sift down */
270 }
271 return false;
272}

278static void wd_wait_until(uint64_t abs_us)
279{
280#if defined(__APPLE__)
281 /* macOS has no pthread_condattr_setclock; use the relative form. */
282 uint64_t now = hwwdt_now_us();
283 uint64_t rel = (abs_us > now) ? (abs_us - now) : 0;
284 struct timespec ts;
285 ts.tv_sec = (time_t)(rel / 1000000ull);
286 ts.tv_nsec = (long)((rel % 1000000ull) * 1000ull);
287 (void)pthread_cond_timedwait_relative_np(&g_cv, &g_lock, &ts);
288#else
289 /* hwwdt_now_us() is gettimeofday(), i.e. CLOCK_REALTIME, which is exactly
290 * the clock a default-initialised condvar uses. */
291 struct timespec ts;
292 ts.tv_sec = (time_t)(abs_us / 1000000ull);
293 ts.tv_nsec = (long)((abs_us % 1000000ull) * 1000ull);
294 (void)pthread_cond_timedwait(&g_cv, &g_lock, &ts);
295#endif
296}

302static void *monitor_main(void *arg)
303{
304 (void)arg;
305 uint32_t stash[WD_MAX_WORKERS];
306 
307 pthread_mutex_lock(&g_lock);
308 while (g_running) {
309 uint64_t now = hwwdt_now_us();
310 int n_overdue = 0;
311 uint64_t root_key = 0;
312 bool have_root = false;
313 
314 /* Walk only the entries whose key has already expired. Each one is
315 * either stale (refresh pushes it down, cost O(log n), and we continue)
316 * or genuinely overdue (report it once, then lift it out of the way so
317 * we can look at the next one). Everything still in the heap after the
318 * loop has key > now, hence true deadline > now, hence is healthy. */
319 for (;;) {
320 uint64_t k;
321 uint32_t id;
322 if (!heap_refresh_root(now, &k, &id)) break; /* heap empty */
323 if (k > now) { root_key = k; have_root = true; break; }
324 
325 if (!g_cold[id].faulted) {
326 g_cold[id].faulted = true;
327 /* Cheap vendor call, and the monitor is its only caller, so it
328 * needs no extra serialisation. */
329 hwwdt_report_fault(id);
330 }
331 heap_remove(id);
332 stash[n_overdue++] = id;
333 }
334 /* Put the overdue workers back with their exact (past-dated) keys, so
335 * that next tick we look at them again and notice when they recover. */
336 for (int i = 0; i < n_overdue; i++)
337 heap_insert(stash[i], true_deadline(stash[i]));
338 
339 /* THE RULE: kick only when every registered worker is healthy. Zero
340 * registered workers is vacuously healthy. */
341 if (n_overdue == 0 && now - g_last_kick_us >= WD_KICK_PERIOD_US) {
342 hwwdt_kick();
343 g_last_kick_us = now;
344 }
345 
346 uint64_t wake;
347 if (n_overdue > 0) {
348 wake = now + WD_RECHECK_US; /* not kicking; just re-check */
349 } else {
350 wake = g_last_kick_us + WD_KICK_PERIOD_US;
351 if (have_root && root_key < wake) wake = root_key;
352 if (wake <= now) wake = now + 200; /* never busy-spin */
353 }
354 wd_wait_until(wake); /* register/unregister/deinit
355 * signal g_cv to recompute */
356 }
357 pthread_mutex_unlock(&g_lock);
358 return NULL;
359}

365int wd_init(void)
366{
367 if (g_started) return -1;
368 
369 pthread_mutex_lock(&g_lock);
370 for (int i = 0; i < WD_MAX_WORKERS; i++) {
371 g_pos[i] = -1;
372 g_cold[i].in_use = false;
373 g_cold[i].faulted = false;
374 atomic_store_explicit(&g_hot[i].last_seen_us, 0, memory_order_relaxed);
375 }
376 g_heap_n = 0;
377 g_last_kick_us = 0; /* kick on the very first tick */
378 g_running = true;
379 pthread_mutex_unlock(&g_lock);
380 
381 if (pthread_create(&g_monitor, NULL, monitor_main, NULL) != 0) {
382 g_running = false;
383 return -1;
384 }
385 g_started = true;
386 return 0;
387}

390{

393 g_running = false;

404int wd_register(const char *name, uint64_t deadline_us)
405{
406 if (deadline_us == 0) return -1;
407 
408 pthread_mutex_lock(&g_lock);
409 int id = -1;
410 for (int i = 0; i < WD_MAX_WORKERS; i++) {
411 if (!g_cold[i].in_use) { id = i; break; }
412 }
413 if (id >= 0) {
414 uint64_t now = hwwdt_now_us();
415 g_cold[id].in_use = true;
416 g_cold[id].faulted = false;
417 g_cold[id].deadline_us = deadline_us;
418 snprintf(g_cold[id].name, sizeof g_cold[id].name, "%s",
419 name ? name : "worker");
420 /* Grace period: a brand-new worker counts as having just checked in,
421 * otherwise it would be declared dead before its first heartbeat. */
422 atomic_store_explicit(&g_hot[id].last_seen_us, now, memory_order_release);
423 heap_insert((uint32_t)id, now + deadline_us);
424 pthread_cond_signal(&g_cv); /* the next deadline may have moved in */
425 }
426 pthread_mutex_unlock(&g_lock);
427 return id;
428}

430void wd_unregister(int id)
431{
432 if (id < 0 || id >= WD_MAX_WORKERS) return;
433 pthread_mutex_lock(&g_lock);
434 if (g_cold[id].in_use) {
435 heap_remove((uint32_t)id);
436 g_cold[id].in_use = false;
437 g_cold[id].faulted = false;
438 pthread_cond_signal(&g_cv);
439 }
440 pthread_mutex_unlock(&g_lock);
441}

447void wd_heartbeat(uint32_t worker_id)
448{
449 if (worker_id >= WD_MAX_WORKERS) return;
450 /* One release store into this worker's own cache line. Wait-free: no lock,
451 * no CAS loop, no branch on other threads' state.
452 *
453 * We deliberately do NOT check in_use here — that would need the lock. A
454 * store into a retired slot is ignored (the monitor only walks the heap).
455 * Slot reuse: a retired thread with one heartbeat still in flight can land
456 * it on the worker that just took the slot over. It writes "now", and
457 * wd_register() already gave the new worker a grace period ending at
458 * now + deadline_us, so the only effect is to shift its first deadline by
459 * microseconds. If that were not acceptable the fix is a generation counter
460 * packed into the same 64-bit word (store (gen << 48) | us) so a stale
461 * writer's word is rejected on the read side — still one store. */
462 atomic_store_explicit(&g_hot[worker_id].last_seen_us, hwwdt_now_us(),
463 memory_order_release);
464}

470int wd_next_deadline(uint64_t *when_us)
471{
472 pthread_mutex_lock(&g_lock);
473 uint64_t k = 0;
474 uint32_t id = 0;
475 bool ok = heap_refresh_root(hwwdt_now_us(), &k, &id);
476 pthread_mutex_unlock(&g_lock);
477 
478 if (!ok) return -1; /* empty: nothing to wait for */
479 if (when_us) *when_us = k;
480 return 0;
481}

483size_t wd_overdue(uint32_t *ids_out, size_t max)
484{
485 size_t n = 0;
486 int stack[WD_MAX_WORKERS];
487 int sp = 0;
488 
489 pthread_mutex_lock(&g_lock);
490 uint64_t now = hwwdt_now_us();
491 
492 /* Pruned DFS instead of a full scan: a node with key > now has children
493 * with key >= its key (heap property) and every true deadline is >= its key
494 * (the invariant), so that whole subtree is healthy and can be skipped. */
495 if (g_heap_n > 0) stack[sp++] = 0;
496 while (sp > 0) {
497 int i = stack[--sp];
498 if (g_heap[i].key > now) continue; /* prune subtree */
499 uint32_t id = g_heap[i].id;
500 if (true_deadline(id) <= now) { /* stale keys filtered */
501 if (n < max) ids_out[n] = id;
502 n++;
503 }
504 int l = 2 * i + 1, r = l + 1;
505 if (l < g_heap_n) stack[sp++] = l;
506 if (r < g_heap_n) stack[sp++] = r;
507 }
508 pthread_mutex_unlock(&g_lock);
509 return (n < max) ? n : max;
