// 61. 간단 명령 파서(키워드+인자)
/*
설명:
임베디드 CLI, UART 명령, 디버그 콘솔 등에서 키워드와 인자를 파싱하는 간단 명령 파서가 필요합니다.
공백, 따옴표, 에러 리포트 등 robust한 토크나이저/파서가 요구됩니다.
예시 상황:
- UART로 "set led 1" 명령 수신 및 파싱
- 부트로더/테스트 모드에서 명령어 해석
요구사항:
- 공백/탭/CRLF 구분자 처리
- 따옴표(")로 감싼 인자 지원
- 최대 인자 개수/길이 제한
- 에러 리포트(오버플로, 잘못된 구문)
- 명령/인자 구조체 반환
*/
#ifndef SIMPLE_CMD_PARSER_H
#define SIMPLE_CMD_PARSER_H
#include <stddef.h>
#define CMD_MAX_ARGS 8
#define CMD_MAX_LEN  32
typedef struct {
    char cmd[CMD_MAX_LEN];
    char* argv[CMD_MAX_ARGS];
    int argc;
} simple_cmd_t;
int simple_cmd_parse(const char* line, simple_cmd_t* out);
#endif

// 62. 환경 의존 HAL 인터페이스
/*
설명:
임베디드 소프트웨어에서 하드웨어 추상화 계층(HAL)은 이식성과 테스트 용이성을 위해 필수적입니다.
GPIO, UART, I2C 등 다양한 하드웨어를 추상화하고, 모킹/테스트를 위한 인터페이스 설계가 중요합니다.
예시 상황:
- 보드별 GPIO 핀맵 차이 대응
- 유닛 테스트에서 HAL 모킹
요구사항:
- GPIO/UART/I2C HAL 인터페이스 구조체
- 함수 포인터 기반 추상화
- 에러 코드/상태 반환
- 모킹/테스트 구현 예시
*/
#ifndef HAL_IFACE_H
#define HAL_IFACE_H
#include <stdint.h>
typedef struct {
    int (*gpio_write)(int pin, int val);
    int (*gpio_read)(int pin);
    int (*uart_send)(const uint8_t* data, size_t len);
    int (*uart_recv)(uint8_t* data, size_t len);
    int (*i2c_write)(uint8_t addr, const uint8_t* data, size_t len);
    int (*i2c_read)(uint8_t addr, uint8_t* data, size_t len);
} hal_iface_t;
#endif

// 63. 논블로킹 I/O 버퍼링
/*
설명:
임베디드 네트워크, UART, 파일 I/O 등에서 논블로킹(부분 전송) 처리가 필요합니다.
EAGAIN/WOULDBLOCK, 재시도 정책, 버퍼 관리 등 robust한 설계가 요구됩니다.
예시 상황:
- TCP 소켓 send/recv 반복
- UART DMA 송수신 버퍼 관리
요구사항:
- send/recv API: 부분 전송 반환
- EAGAIN/WOULDBLOCK 처리
- 내부 버퍼 상태 추적
- 재시도/타임아웃 정책
*/
#ifndef NONBLOCK_IO_H
#define NONBLOCK_IO_H
#include <stddef.h>
#include <stdint.h>
typedef struct {
    uint8_t* buf;
    size_t cap;
    size_t head, tail;
} nbio_buf_t;
int nbio_send(nbio_buf_t* b, const uint8_t* data, size_t len);
int nbio_recv(nbio_buf_t* b, uint8_t* data, size_t len);
int nbio_status(const nbio_buf_t* b);
#endif

/*

타임아웃, 최대 조각 개수, 완성 콜백 등 robust한 구현이 요구됩니다.
예시 상황:
- BLE/LoRaWAN 조각 패킷 재조립

- 시퀀스/오프셋 기반 조각 관리
- 최대 조각 개수/버퍼 크기 제한
- 타임아웃/오래된 조각 삭제
- 완성 시 콜백 호출
*/

void pkt_reassembly_destroy(pkt_reassembly_t* r);
int pkt_reassembly_feed(pkt_reassembly_t* r, uint16_t seq, uint16_t offset, const uint8_t* data, size_t len, uint32_t now_tick);
void pkt_reassembly_timeout(pkt_reassembly_t* r, uint32_t now_tick, uint32_t timeout_ticks);
#endif

// 69. 지연 실행 디퍼드 큐
/*
설명:
임베디드에서 인터럽트에서 바로 처리하지 않고, 메인 루프에서 나중에 실행할 작업을 큐에 등록하는 디퍼드 큐가 널리 쓰입니다.
경계/드롭 정책, 우선순위(선택), lock-free 등 robust한 구현이 요구됩니다.
예시 상황:
- 센서 인터럽트에서 데이터 처리 요청 등록
- 타이머 만료 후 작업 예약
요구사항:
- 고정 크기 디퍼드 큐
- 오버플로 시 드롭/에러 반환
- ISR/메인루프 분리
- (선택) 우선순위 지원
*/
#ifndef DEFERRED_QUEUE_H
#define DEFERRED_QUEUE_H
#include <stddef.h>
typedef void (*deferred_cb_t)(void*);
typedef struct {
    deferred_cb_t cb;
    void* user;
} deferred_task_t;
#define DEFERRED_QUEUE_CAP 16
typedef struct {
    deferred_task_t tasks[DEFERRED_QUEUE_CAP];
    volatile size_t head, tail;
} deferred_queue_t;
void deferred_queue_init(deferred_queue_t* q);
int  deferred_queue_push(deferred_queue_t* q, deferred_cb_t cb, void* user); // ISR safe
int  deferred_queue_pop(deferred_queue_t* q, deferred_cb_t* cb, void** user); // Main loop
#endif
