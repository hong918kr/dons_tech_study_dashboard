// 31. 고정소수점 PID 컨트롤러
/*
설명:
저가형 MCU 기반의 모터/온도/압력 제어 등에서 부동소수점 연산 없이 PID 제어기를 구현해야 하는 상황이 많습니다.
Q16.16 또는 Q8.8 등 고정소수점 연산을 사용해, 실시간 제어 루프에서 빠르고 예측 가능한 동작을 보장해야 합니다.
예시 상황:
- DC 모터 속도/위치 제어
- 온도 유지 PID 루프
요구사항:
- Q 포맷(예: Q16.16) 고정소수점 연산
- anti-windup(적분 포화 방지)
- 미분 필터(노이즈 억제)
- 출력 포화(상/하한)
- 파라미터/상태 구조체 분리
- 초기화/리셋 API
*/
#ifndef FIXED_PID_H
#define FIXED_PID_H
#include <stdint.h>
typedef int32_t q16_16_t;
typedef struct {
    q16_16_t kp, ki, kd;
    q16_16_t out_min, out_max;
    q16_16_t integrator;
    q16_16_t prev_error;
    q16_16_t prev_deriv;
} fixed_pid_t;
void fixed_pid_init(fixed_pid_t* pid, q16_16_t kp, q16_16_t ki, q16_16_t kd, q16_16_t out_min, q16_16_t out_max);
void fixed_pid_reset(fixed_pid_t* pid);
q16_16_t fixed_pid_update(fixed_pid_t* pid, q16_16_t setpoint, q16_16_t measurement);
#endif

// 32. 신호 디바운서
/*
설명:
기계식 버튼, 릴레이 등에서 접점 튐(bounce) 현상으로 인해 신호가 불안정하게 여러 번 변하는 것을 방지해야 합니다.
임베디드 시스템에서는 소프트웨어적으로 시간 기반 또는 카운터 기반 디바운싱이 널리 사용됩니다.
예시 상황:
- 사용자 버튼 입력 처리
- 릴레이 상태 감지
요구사항:
- 시간 기반(샘플 주기, 안정 시간)과 카운터 기반(연속 N회) 두 방식 제공
- 상태 변화 감지(pressed/released)
- 초기화/리셋 API
- 디바운스 파라미터 구조체화
*/
#ifndef DEBOUNCE_H
#define DEBOUNCE_H
#include <stdint.h>
typedef struct {
    uint8_t stable_state;
    uint8_t counter;
    uint8_t threshold;
} debounce_counter_t;
void debounce_counter_init(debounce_counter_t* db, uint8_t threshold);
uint8_t debounce_counter_update(debounce_counter_t* db, uint8_t raw_state);

typedef struct {
    uint8_t stable_state;
    uint32_t last_change_tick;
    uint32_t debounce_ticks;
} debounce_time_t;
void debounce_time_init(debounce_time_t* db, uint32_t debounce_ticks);
uint8_t debounce_time_update(debounce_time_t* db, uint8_t raw_state, uint32_t now_tick);
#endif

// 33. 이동 평균 필터(박스/EMA)
/*
설명:
센서 노이즈 제거, 신호 평활화 등에 이동 평균 필터가 널리 쓰입니다.
박스(SMA)와 지수(EMA) 방식 모두 임베디드에서 자주 사용됩니다.
예시 상황:
- 온도/압력/조도 센서 신호 평활화
- ADC 입력 노이즈 저감
요구사항:
- 박스(SMA): 순환 버퍼, 오버플로 방지
- EMA: alpha 파라미터, 정수화 전략
- 초기화/리셋 API
- 최근 평균값 반환
*/
#ifndef MOVING_AVG_H
#define MOVING_AVG_H
#include <stddef.h>
#include <stdint.h>
typedef struct {
    int32_t* buf;
    size_t window, pos, count;
    int64_t sum;
} sma_filter_t;
void sma_init(sma_filter_t* f, int32_t* buf, size_t window);
void sma_reset(sma_filter_t* f);
void sma_push(sma_filter_t* f, int32_t val);
int32_t sma_get(const sma_filter_t* f);

typedef struct {
    int32_t avg;
    uint8_t alpha_q8; // 0~255 (Q0.8)
} ema_filter_t;
void ema_init(ema_filter_t* f, uint8_t alpha_q8);
void ema_reset(ema_filter_t* f);
void ema_push(ema_filter_t* f, int32_t val);
int32_t ema_get(const ema_filter_t* f);
#endif
