// 91. 소형 이벤트 디스패처(구독자)
/*
설명:
임베디드에서 센서/통신/상태 변화 등 다양한 이벤트를 여러 콜백(구독자)에게 분배하는 이벤트 디스패처가 필요합니다.
고정 테이블 기반, 재진입성, 중복 등록 방지, 빠른 분배 등이 중요합니다.
예시 상황:
- 센서 값 변화 시 여러 모듈에 알림
- 통신 수신 이벤트를 여러 핸들러에 분배
요구사항:
- 고정 크기 구독자 테이블(주제별)
- 콜백 등록/해제/호출 API
- 재진입성(동일 이벤트 중첩 호출 안전)
- 중복 등록 방지
- 최대 구독자 초과 시 에러 반환
*/
#ifndef EVENT_DISPATCHER_H
#define EVENT_DISPATCHER_H
#include <stddef.h>
typedef void (*event_cb_t)(int topic, void* data, void* user);
#define MAX_TOPICS 8
#define MAX_SUBSCRIBERS 8
typedef struct {
    event_cb_t cb;
    void* user;
} subscriber_t;
typedef struct {
    subscriber_t subs[MAX_TOPICS][MAX_SUBSCRIBERS];
} event_dispatcher_t;
void event_dispatcher_init(event_dispatcher_t* d);
int  event_subscribe(event_dispatcher_t* d, int topic, event_cb_t cb, void* user);
int  event_unsubscribe(event_dispatcher_t* d, int topic, event_cb_t cb, void* user);
void event_publish(event_dispatcher_t* d, int topic, void* data);
#endif

// 96. 간단 스케줄 테이블 실행기
/*

임베디드에서 시간/주기 기반으로 여러 작업을 스케줄 테이블에 따라 실행해야 할 때가 많습니다.
드리프트 보정, 미스 실행 처리, 상태 추적 등 robust한 설계가 요구됩니다.
예시 상황:
- 주기적 센서 측정/통신 태스크
- 실시간 제어 루프 스케줄링
요구사항:
- 스케줄 테이블 구조체(시간/주기/콜백)
- 드리프트 보정(누적 오차)

    sched_entry_t* entries;
    size_t count;
    uint32_t now_tick;
} sched_table_t;

void sched_table_tick(sched_table_t* t, uint32_t now_tick);
