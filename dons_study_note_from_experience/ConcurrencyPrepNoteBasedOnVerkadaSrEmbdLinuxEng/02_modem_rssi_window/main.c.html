1/* main.c — test harness + fake LTE modem for 02_modem_rssi_window. (given)
2 *
3 * Same idea as the original Verkada screening harness: the fake vendor API
4 * records every value it ever produced ("ground truth"), and every PASS/FAIL is
5 * computed against that, never against a hard-coded number — the RNG sequence
6 * must not have to match between macOS and Linux, so the RNG is a fixed-seed
7 * xorshift instead of rand().
8 *
9 * What is checked, in order:
10 * 0. getters before rssi_init() -> "no data", no crash
11 * 1. right after rssi_init() -> a value exists, age ~0
12 * 2. latest value + age vs ground truth
13 * 3. min / max / avg vs a brute-force O(n) reference, twice, the second time
14 * after samples have aged out of the window
15 * 4. modem dies -> age grows, rssi_is_healthy() flips false, window drains
16 * to RSSI_NONE / NAN; modem recovers -> health comes back
17 * 5. modem_poll_rssi() was never called concurrently (it is not thread-safe)
18 * 6. 4 readers for 2 s: no invented values, worst getter latency << 100 ms
19 * 7. after rssi_deinit() the modem stops being polled (thread really gone)
20 *
21 * Exact comparisons are done with the fake modem paused and at a moment when no
22 * sample sits near the window edge, so that a sample expiring between the
23 * getter and the reference cannot make a correct implementation look wrong.
24 *
25 * Do not modify: the same file is built against rssi_window.c and
26 * rssi_window_solution.c.
27 */

49/* splitmix64, fixed seed: same sequence on macOS and Linux, and unlike a bare
50 * xorshift it has no correlation between consecutive draws (this fake draws in
51 * a fixed 2-5 call pattern per poll, which a weak generator shows up in).
52 * Touched only by the single thread that is allowed to poll the modem. */
53static uint64_t g_rng = 0x9E3779B97F4A7C15ull;
54 
55static uint64_t splitmix64(void)
56{
57 uint64_t z = (g_rng += 0x9E3779B97F4A7C15ull);
58 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ull;
59 z = (z ^ (z >> 27)) * 0x94D049BB133111EBull;
60 return z ^ (z >> 31);
61}

118/* Did the modem ever hand out this rssi value? (catches torn / invented /
119 * garbage-payload values, e.g. publishing on MODEM_BUSY or MODEM_ERROR.) */
120static bool truth_contains(int rssi)
121{
122 bool found = false;
123 pthread_mutex_lock(&truth_lock);
124 for (int i = 0; i < truth_n && !found; i++)
125 found = (truth[i].rssi == rssi);
126 pthread_mutex_unlock(&truth_lock);

130/* Brute-force O(n) reference for the window ending at `now`. */
131typedef struct { int n; int mn, mx; double avg; } wstat_t;
132 
133static wstat_t truth_window(uint64_t now)
134{
135 wstat_t w = {0, RSSI_NONE, RSSI_NONE, NAN};
136 long long sum = 0;
137 pthread_mutex_lock(&truth_lock);
138 for (int i = 0; i < truth_n; i++) {
139 uint64_t ts = truth[i].ts;
140 if (ts > now || now - ts > RSSI_WINDOW_US)
141 continue;
142 int v = truth[i].rssi;
143 if (w.n == 0) { w.mn = v; w.mx = v; }
144 else { if (v < w.mn) w.mn = v; if (v > w.mx) w.mx = v; }
145 sum += v;
146 w.n++;
147 }
148 pthread_mutex_unlock(&truth_lock);
149 if (w.n > 0)
150 w.avg = (double)sum / (double)w.n;
151 return w;
152}

192static void modem_pause(bool paused) { atomic_store(&g_modem_dead, paused); }

194/* Caller has paused the modem. Wait out any in-flight poll, then wait for an
195 * instant where no recorded sample is within EDGE_GUARD_US of the window edge,
196 * so getter and reference must agree. */
197static void settle(void)
198{
199 sleep_us(120 * 1000); /* > a typical in-flight vendor poll */
200 for (int tries = 0; tries < 200; tries++) {
201 uint64_t now = modem_now_us();
202 uint64_t edge = (now > RSSI_WINDOW_US) ? now - RSSI_WINDOW_US : 0;
203 bool close = false;
204 pthread_mutex_lock(&truth_lock);
205 for (int i = 0; i < truth_n && !close; i++) {
206 uint64_t ts = truth[i].ts;
207 uint64_t d = ts > edge ? ts - edge : edge - ts;
208 if (d < EDGE_GUARD_US) close = true;
209 }
210 pthread_mutex_unlock(&truth_lock);
211 if (!close) return;
212 usleep(5 * 1000);
213 }
214}

286static void *stress_reader(void *arg)
287{
288 stress_stat_t *st = arg;
289 while (!atomic_load(&stress_stop)) {
290 int dbm = 0;
291 uint64_t age = 0;
292 
293 uint64_t t0 = modem_now_us();
294 int rc = rssi_get_latest(&dbm, &age);
295 bool healthy = rssi_is_healthy(500 * 1000ull);
296 int mn = rssi_min_in_window();
297 int mx = rssi_max_in_window();
298 double av = rssi_avg_in_window();
299 uint64_t dt = modem_now_us() - t0;
300 
301 if (dt > st->max_us) st->max_us = dt;
302 st->total_us += dt;
303 
304 if (rc == 0) {
305 if (!truth_contains(dbm)) st->bad++; /* value nobody produced */
306 if (age > 30ull * 1000000ull) st->bad++; /* nonsense age */
307 if (healthy && age > 500 * 1000ull) st->bad++;
308 } else if (healthy) {
309 st->bad++; /* healthy with no value */
310 }
311 if (mn != RSSI_NONE && !truth_contains(mn)) st->bad++;
312 if (mx != RSSI_NONE && !truth_contains(mx)) st->bad++;
313 if (mn != RSSI_NONE && mx != RSSI_NONE && mn > mx) st->bad++;
314 if (!isnan(av) && (av < -140.0 || av > -20.0)) st->bad++;
315 st->calls++;
316 }
317 return NULL;
318}

470struct RssiSample modem_poll_rssi(void)
471{
472 static bool first_run = true;
473 static uint32_t g_polls;
474 static int cur_rssi = -85;
475 static int cur_sinr = 12;
476 struct RssiSample r;
477 
478 /* harness only: the vendor API is not thread-safe, so catch two callers */
479 if (atomic_fetch_add(&g_in_call, 1) != 0)
480 atomic_fetch_add(&g_concurrent, 1);
481 
482 /* poison the payload: only MODEM_OK is allowed to be trusted */
483 r.status = MODEM_BUSY;
484 r.rssi_dbm = 9999;
485 r.sinr_db = -9999;
486 r.timestamp = 0xDEADBEEFDEADBEEFull;
487 
488 if (first_run) { /* first call: instant, always valid */
489 first_run = false;
490 r.status = MODEM_OK;
491 r.rssi_dbm = cur_rssi;
492 r.sinr_db = cur_sinr;
493 r.timestamp = modem_now_us();
494 truth_record(r.timestamp, r.rssi_dbm, r.sinr_db);
495 atomic_fetch_add(&g_n_ok, 1);
496 atomic_fetch_sub(&g_in_call, 1);
497 return r;
498 }
499 
500 /* AT command failed: returns instantly with a garbage payload. Every 17th
501 * poll (~6%), deterministic so the ERROR path is always exercised. */
502 if (++g_polls % 17u == 5u) {
503 r.status = MODEM_ERROR;
504 atomic_fetch_add(&g_n_err, 1);
505 atomic_fetch_sub(&g_in_call, 1);
506 return r;
507 }
508 
509 usleep((useconds_t)(rnd01() * 100000.0)); /* blocks 0 .. 100 ms */
510 
511 if (atomic_load(&g_modem_dead) || rnd01() < 0.45) { /* nothing new to report */
512 atomic_fetch_add(&g_n_busy, 1);
513 atomic_fetch_sub(&g_in_call, 1);
514 return r;
515 }
516 
517 cur_rssi += (int)(rnd01() * 13.0) - 6; /* random walk, clamped */
518 if (cur_rssi < -115) cur_rssi = -115;
519 if (cur_rssi > -55) cur_rssi = -55;
520 cur_sinr += (int)(rnd01() * 5.0) - 2;
521 if (cur_sinr < -10) cur_sinr = -10;
522 if (cur_sinr > 30) cur_sinr = 30;
523 
524 r.status = MODEM_OK;
525 r.rssi_dbm = cur_rssi;
526 r.sinr_db = cur_sinr;
527 r.timestamp = modem_now_us();
528 truth_record(r.timestamp, r.rssi_dbm, r.sinr_db);
529 atomic_fetch_add(&g_n_ok, 1);
530 atomic_fetch_sub(&g_in_call, 1);
531 return r;
