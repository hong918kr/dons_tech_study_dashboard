2 *
3 * ./main.sh builds main.c + audit_log.c (your code)
4 * ./main.sh sol builds main.c + audit_log_solution.c (reference)
5 *
6 * Everything is checked against ground truth that the fake vendor records as it
7 * runs — there are no hard-coded expected values, so the result is the same on
8 * macOS and Linux. Randomness is a hand-rolled xorshift with a fixed per-door
9 * seed (never rand(): its sequence differs per platform).
10 *
11 * Phases
12 * 0 empty state right after audit_init()
13 * A load: 4 readers produce faster than the audit store can append, while 3
14 * harness threads hammer the getters (thread safety + getter latency)
15 * B park the readers, wait for the queue to drain
16 * C accounting + per-door ordering of everything that reached the store
17 * D audit_is_duplicate() vs brute force over the store's contents
18 * E audit_recent_for_door() vs the store's contents
19 * F dedupe table: lazy age eviction, then capacity pressure, then latency
20 * G audit_deinit() while all 4 producers are parked in reader_wait_badge()
21 *
22 * Scaled delays (real AC42 numbers in parentheses):

103/* Portable "wait on a condvar for at most us microseconds": macOS has no
104 * pthread_condattr_setclock, so branch on the platform. */
105static int cv_wait_us(pthread_cond_t *cv, pthread_mutex_t *m, uint64_t us)
106{
107#ifdef __APPLE__
108 struct timespec rel;
109 rel.tv_sec = (time_t)(us / 1000000ull);
110 rel.tv_nsec = (long)((us % 1000000ull) * 1000ull);
111 return pthread_cond_timedwait_relative_np(cv, m, &rel);
112#else
113 struct timespec ts;
114 clock_gettime(CLOCK_REALTIME, &ts);
115 ts.tv_sec += (time_t)(us / 1000000ull);
116 ts.tv_nsec += (long)((us % 1000000ull) * 1000ull);
117 if (ts.tv_nsec >= 1000000000L) {
118 ts.tv_nsec -= 1000000000L;
119 ts.tv_sec += 1;
120 }
121 return pthread_cond_timedwait(cv, m, &ts);
122#endif
123}

257static int reader_wait_inner(uint8_t door, struct Badge *out)
258{
259 struct DoorSim *d = &g_door[door];
260 bool parked = atomic_load(&g_reader_park) != 0;
261 uint64_t want = parked ? READER_PARK_US
262 : READER_MIN_US + rng32(&d->rng) % READER_SPAN_US;
263 uint64_t deadline = now_us() + want;
264 
265 pthread_mutex_lock(&d->m);
266 for (;;) {
267 if (d->cancelled)
268 break;
269 uint64_t n = now_us();
270 if (n >= deadline)
271 break;
272 cv_wait_us(&d->cv, &d->m, deadline - n);
273 }
274 bool cancelled = d->cancelled;
275 pthread_mutex_unlock(&d->m);
276 
277 if (cancelled)
278 return -1; /* interrupted by reader_cancel() */
279 if (atomic_load(&g_reader_park) != 0)
280 return -1; /* nobody showed up */
281 if (rng32(&d->rng) % READER_ERR_ONE_IN == 0u) {
282 pthread_mutex_lock(&g_truth_lock);
283 g_read_errors++;
284 pthread_mutex_unlock(&g_truth_lock);
285 return -1; /* transient read error */
286 }
287 
288 struct Badge b;
289 b.badge_id = BADGE_ID_BASE + rng32(&d->rng) % BADGE_ID_POOL;
290 b.door = door;
291 b.timestamp = now_us();
292 truth_record_produced(&b); /* harness only */
293 *out = b;
294 return 0;

373static void *stress_main(void *arg)
374{
375 stress_t *st = arg;
376 struct Badge tmp[AUDIT_RECENT_PER_DOOR];
377 uint32_t rng = 0xdeadbeefu ^ (uint32_t)(uintptr_t)arg;
378 
379 while (!atomic_load(&g_stress_stop)) {
380 uint8_t door = (uint8_t)(rng32(&rng) % (uint32_t)AC42_DOORS);
381 uint32_t id = BADGE_ID_BASE + rng32(&rng) % BADGE_ID_POOL;
382 
383 uint64_t t0 = now_us();
384 int dup = audit_is_duplicate(id, t0, 50000ull);
385 size_t k = audit_recent_for_door(door, tmp, AUDIT_RECENT_PER_DOOR);
386 uint64_t t1 = now_us();
387 uint64_t dt = t1 - t0;
388 
389 if (dt > st->max_us)
390 st->max_us = dt;
391 st->sum_us += dt;
392 st->calls++;
393 (void)dup;
394 
395 /* No torn or invented records: every field in range, the door must be
396 * the one we asked for, newest first. The exact comparison against
397 * ground truth happens later, when the pipeline is quiescent. */
398 if (k > AUDIT_RECENT_PER_DOOR) {
399 st->bad++;
400 continue;
401 }
402 for (size_t i = 0; i < k; i++) {
403 if (tmp[i].door != door)
404 st->bad++;
405 if (tmp[i].badge_id < BADGE_ID_BASE ||
406 tmp[i].badge_id >= BADGE_ID_BASE + BADGE_ID_POOL)
407 st->bad++;
408 if (tmp[i].timestamp < g_t_start || tmp[i].timestamp > t1)
409 st->bad++;
410 if (i + 1 < k && tmp[i].timestamp < tmp[i + 1].timestamp)
411 st->bad++;
412 }
413 }
414 return NULL;
415}

421/* Wait until the store has been idle for 200 ms (queue drained), max 1.5 s. */
422static void drain_wait(void)
423{
424 uint64_t deadline = now_us() + 1500000ull;
425 uint64_t last = sink_total();
426 uint64_t stable_since = now_us();
427 while (now_us() < deadline) {
428 usleep(20000);
429 uint64_t c = sink_total();
430 if (c != last) {
431 last = c;
432 stable_since = now_us();
433 continue;
434 }
435 if (now_us() - stable_since > 200000ull)
436 return;
437 }
438}

443/* Push synthetic badges straight into the pipeline (audit_submit is the
444 * producer API, so the harness may use it too). Returns #accepted and, when
445 * `record` is set, leaves the accepted ids in g_fill_ids in submission order. */
446static size_t fill_submit(uint32_t first_id, size_t count, uint64_t first_ts, bool record)
447{
448 size_t accepted = 0, consecutive_fail = 0;
449 if (record)
450 g_fill_n = 0;
451 for (size_t i = 0; i < count; i++) {
452 struct Badge b;
453 b.badge_id = first_id + (uint32_t)i;
454 b.door = 0;
455 b.timestamp = first_ts + (uint64_t)i;
456 bool ok = false;
457 for (int attempt = 0; attempt < 5 && !ok; attempt++) {
458 if (audit_submit(&b) == 0)
459 ok = true;
460 else
461 usleep(200);
462 }
463 if (ok) {
464 accepted++;
465 consecutive_fail = 0;
466 if (record && g_fill_n < FILL_MAX)
467 g_fill_ids[g_fill_n++] = b.badge_id;
468 } else if (++consecutive_fail >= 20) {
469 break; /* nothing is consuming: give up instead of hanging */
470 }
471 }
472 drain_wait();
473 return accepted;
474}

609 const uint64_t windows[] = {0ull, 1000ull, 20000ull, 100000ull, 500000ull, 2000000ull};
610 size_t mismatch = 0, dup_true = 0, dup_false = 0, shown = 0;
611 for (uint32_t id = BADGE_ID_BASE; id < BADGE_ID_BASE + BADGE_ID_POOL; id++) {
612 uint64_t last = 0;
613 bool seen = sink_last_seen(id, &last);
614 for (size_t w = 0; w < sizeof windows / sizeof windows[0]; w++) {
615 bool want = seen && last <= t_q && (t_q - last) <= windows[w];
616 int got = audit_is_duplicate(id, t_q, windows[w]);
617 if ((got != 0) != want) {
618 mismatch++;
619 if (shown++ < 3)
620 printf(" id %u window %llu us: got %d want %d (last seen: %s)\n",
621 id, (unsigned long long)windows[w], got, (int)want,
622 seen ? "yes" : "never");
623 }
624 if (want)
625 dup_true++;
626 else
627 dup_false++;
628 }
629 }
630 check(mismatch == 0 && dup_true > 0 && dup_false > 0,
631 "%zu queries agree with brute force (%zu duplicates, %zu not)",
632 dup_true + dup_false, dup_true, dup_false);
633 
634 uint32_t ghost = BADGE_ID_BASE + BADGE_ID_POOL + 7u;
635 check(audit_is_duplicate(ghost, t_q, 2000000ull) == 0,
636 "a badge that was never presented is not a duplicate");
637 
638 /* exactly at the window edge, one microsecond inside it, and a now_us
639 * that predates the sighting */
640 uint32_t probe = 0;
641 uint64_t probe_ts = 0;
642 for (uint32_t id = BADGE_ID_BASE; id < BADGE_ID_BASE + BADGE_ID_POOL; id++) {
643 if (sink_last_seen(id, &probe_ts)) {
644 probe = id;
645 break;
646 }
647 }
648 if (probe == 0) {
649 check(false, "window-edge cases: no logged badge to probe with");
650 } else {
651 uint64_t age = t_q - probe_ts;
652 check(audit_is_duplicate(probe, t_q, age) == 1,
653 "exactly at the window edge counts as a duplicate (age %llu us)",
654 (unsigned long long)age);
655 check(age == 0 || audit_is_duplicate(probe, t_q, age - 1) == 0,
656 "one microsecond inside the edge does not");
657 check(audit_is_duplicate(probe, probe_ts, 0) == 1,
658 "window 0 at the exact sighting time is a duplicate");
659 check(probe_ts == 0 || audit_is_duplicate(probe, probe_ts - 1, 1000000ull) == 0,
660 "a now_us before the sighting is not a duplicate");
661 }
662 }
663 
664 printf("\n== Phase E: audit_recent_for_door() ==\n");
665 {
666 size_t bad_doors = 0;
667 for (int d = 0; d < AC42_DOORS; d++) {
668 struct Badge want[AUDIT_RECENT_PER_DOOR];
669 size_t want_n = 0;
670 pthread_mutex_lock(&g_truth_lock);
671 for (size_t i = g_sink_n; i-- > 0 && want_n < AUDIT_RECENT_PER_DOOR;)
672 if (g_sink[i].door == d)
673 want[want_n++] = g_sink[i]; /* newest first */
674 pthread_mutex_unlock(&g_truth_lock);
675 
676 struct Badge got[AUDIT_RECENT_PER_DOOR + 8];
677 memset(got, 0, sizeof got);
678 size_t got_n = audit_recent_for_door((uint8_t)d, got, AUDIT_RECENT_PER_DOOR + 8);
679 bool ok = (got_n == want_n);
680 for (size_t i = 0; ok && i < got_n; i++)
681 ok = got[i].badge_id == want[i].badge_id &&
682 got[i].timestamp == want[i].timestamp && got[i].door == want[i].door;
683 if (!ok) {
684 bad_doors++;
685 printf(" door %d: got %zu records, want %zu\n", d, got_n, want_n);
686 }
687 
688 /* a smaller max must return the NEWEST 3, not the oldest 3 */
689 size_t few = audit_recent_for_door((uint8_t)d, got, 3);
690 if (few != (want_n < 3 ? want_n : 3) ||
691 (few > 0 && (got[0].timestamp != want[0].timestamp ||
692 got[0].badge_id != want[0].badge_id)))
693 bad_doors++;
694 }
695 check(bad_doors == 0, "all %d doors return the newest %u records, newest first",
696 AC42_DOORS, (unsigned)AUDIT_RECENT_PER_DOOR);
697 }
698 
699 printf("\n== Phase F: dedupe table under age and capacity pressure ==\n");
700 atomic_store(&g_store_delay_us, 0); /* fast store so the fill is quick */
701 {
702 uint64_t vt = now_us() + 1000ull;
703 size_t live0 = audit_dedupe_live();
704 
705 /* F1 — lazy age eviction. AGE_BATCH badges, then AGE_BATCH more a full
706 * retention period later: the second batch must recycle the first
707 * batch's aged-out slots instead of forcing live evictions. */
708 uint64_t ev_before = audit_dedupe_evictions();
709 size_t a1 = fill_submit(100000u, AGE_BATCH, vt, false);
710 uint64_t vt2 = vt + AUDIT_DEDUPE_RETENTION_US + 1000000ull;
711 size_t a2 = fill_submit(200000u, AGE_BATCH, vt2, true);
712 uint64_t ev_after = audit_dedupe_evictions();
713 size_t live1 = audit_dedupe_live();
714 printf(" batch1 %zu accepted, batch2 (+%llu s) %zu accepted; live %zu -> %zu,"
715 " stale reuse %llu, forced evictions +%llu\n",
716 a1, (unsigned long long)(AUDIT_DEDUPE_RETENTION_US / 1000000ull + 1), a2,
717 live0, live1, (unsigned long long)audit_dedupe_stale_reuse(),
718 (unsigned long long)(ev_after - ev_before));
719 check(a1 >= AGE_BATCH - 50 && a2 >= AGE_BATCH - 50,
720 "both batches went through (%zu, %zu of %d)", a1, a2, AGE_BATCH);
721 check(live1 <= AUDIT_DEDUPE_CAP, "live slots %zu <= capacity %u", live1,
722 (unsigned)AUDIT_DEDUPE_CAP);
723 check(live1 <= AUDIT_DEDUPE_LOAD_LIMIT,
724 "lazy eviction kept the table inside its %u-slot budget "
725 "(%zu live after %zu distinct badges)",
726 (unsigned)AUDIT_DEDUPE_LOAD_LIMIT, live1, a1 + a2);
727 check(ev_after - ev_before < 50,
728 "aged-out slots were recycled instead of evicting live ones (%llu forced)",
729 (unsigned long long)(ev_after - ev_before));
730 if (g_fill_n > 0)
731 check(audit_is_duplicate(g_fill_ids[g_fill_n - 1], vt2 + AGE_BATCH,
732 1000000ull) == 1,
733 "the newest badge of batch 2 is still known");
734 else
735 check(false, "batch 2 accepted nothing");
736 
737 /* F2 — capacity pressure: far more distinct recent badges than slots. */
738 uint64_t vt3 = vt2 + 2ull * AUDIT_DEDUPE_RETENTION_US;
739 uint64_t ev2_before = audit_dedupe_evictions();
740 size_t a3 = fill_submit(300000u, FILL_MAX, vt3, true);
741 uint64_t vt_now = vt3 + (uint64_t)FILL_MAX + 1000ull;
742 size_t live2 = audit_dedupe_live();
743 printf(" %zu distinct recent badges into %u slots: live %zu, forced evictions %llu\n",
744 a3, (unsigned)AUDIT_DEDUPE_CAP, live2,
745 (unsigned long long)(audit_dedupe_evictions() - ev2_before));
746 check(a3 > FILL_MAX / 2, "the fill went through (%zu of %d)", a3, FILL_MAX);
747 check(live2 <= AUDIT_DEDUPE_CAP, "live slots %zu never exceed capacity %u", live2,
748 (unsigned)AUDIT_DEDUPE_CAP);
749 check(audit_dedupe_evictions() > ev2_before,
750 "capacity pressure is reported as evictions instead of silently corrupting");
751 
752 if (g_fill_n >= 2) {
753 check(audit_is_duplicate(g_fill_ids[g_fill_n - 1], vt_now, 5000000ull) == 1 &&
754 audit_is_duplicate(g_fill_ids[g_fill_n - 2], vt_now, 5000000ull) == 1,
755 "a just-inserted badge is never its own eviction victim");
756 } else {
757 check(false, "fill phase produced nothing to probe");
758 }
759 
760 if (g_fill_n >= AUDIT_DEDUPE_CAP / 2) {
761 size_t n = AUDIT_DEDUPE_CAP / 2, hit = 0;
762 for (size_t i = g_fill_n - n; i < g_fill_n; i++)
763 if (audit_is_duplicate(g_fill_ids[i], vt_now, 5000000ull))
764 hit++;
765 check(hit * 100 >= n * 70,
766 "eviction prefers the oldest: %zu/%zu of the most recent badges kept",
767 hit, n);
768 } else {
769 check(false, "not enough accepted badges to test eviction preference");
770 }
771 
772 size_t fp = 0;
773 for (uint32_t i = 0; i < 256; i++)
774 if (audit_is_duplicate(900000u + i, vt_now, 5000000ull))
775 fp++;
776 check(fp == 0, "no false positives among 256 never-seen badges (%zu)", fp);
777 
778 /* O(1)-ish lookups: an unbounded probe loop or a full scan shows here. */
779 uint32_t rng = 0x12345677u;
780 uint64_t t0 = now_us();
781 int hits = 0;
782 for (int i = 0; i < 20000; i++)
783 hits += audit_is_duplicate(300000u + rng32(&rng) % 8000u, vt_now, 5000000ull);
784 uint64_t elapsed = now_us() - t0;
785 printf(" 20000 random lookups in %llu us (%.3f us each), %d hits\n",
786 (unsigned long long)elapsed, (double)elapsed / 20000.0, hits);
787 check(elapsed < 2000000ull, "20000 lookups stay well under 2 s (%llu us)",
788 (unsigned long long)elapsed);
789 }
790 atomic_store(&g_store_delay_us, STORE_WRITE_US);
791 
792 printf("\n== Phase G: shutdown while all 4 producers are parked ==\n");
793 sleep_us(100000); /* make sure everyone is parked */
794 uint64_t sink_before = sink_total();
795 uint64_t t_deinit = now_us();
796 audit_deinit();
797 uint64_t deinit_us = now_us() - t_deinit;
798 printf(" audit_deinit() returned in %llu us (the readers were parked for up to %llu us)\n",
799 (unsigned long long)deinit_us, (unsigned long long)READER_PARK_US);
800 check(deinit_us < 250000ull,
801 "deinit cancelled the vendor call instead of waiting it out (%llu us)",
802 (unsigned long long)deinit_us);
803 s = snapshot();
804 check(s.cancelled_doors == AC42_DOORS, "reader_cancel() called for all %d doors (%d)",
805 AC42_DOORS, s.cancelled_doors);
806 {
807 struct Badge b = {.badge_id = 1u, .door = 0, .timestamp = now_us()};
808 check(audit_submit(&b) < 0, "audit_submit() after deinit is rejected, not queued");
809 }
810 sleep_us(150000);
811 uint64_t sink_after = sink_total();
812 check(sink_after == sink_before, "no store writes after deinit (%llu -> %llu)",
813 (unsigned long long)sink_before, (unsigned long long)sink_after);
814 s = snapshot();
815 check(s.same_door == 0 && s.store_concurrent == 0 && s.store_foreign == 0,
816 "vendor contracts still intact after shutdown");
