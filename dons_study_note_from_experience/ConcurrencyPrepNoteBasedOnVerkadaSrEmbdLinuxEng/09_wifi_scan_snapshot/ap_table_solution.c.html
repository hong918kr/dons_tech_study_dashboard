1/* ap_table_solution.c — reference solution.
2 *
3 * ./main.sh sol build main.c + this file and run every check
4 * ./main.sh tsan same, under -fsanitize=thread
5 *
6 * Shape
7 * one scanner thread ── the ONLY caller of wifi_scan() (it blocks 100-300 ms
8 * and the vendor code is not thread-safe).
9 * │ on success
10 * ├──► g_snap[] Part 1: THREE preallocated slots, each holding a whole
11 * │ variable-length scan (count + entries). An atomic
12 * │ index says which slot is current; an atomic per-slot
13 * │ refcount ("pin") says which slots a reader is inside.
14 * │ The writer fills a slot nobody is pinning, then
15 * │ release-stores the index. Readers pin, re-check,
16 * │ memcpy, unpin.
17 * └──► g_tab Part 2: open-addressing hash map keyed by BSSID,
18 * mutex-guarded, EWMA + lazy time eviction.
19 *
20 * ------------------------------------------------------------------------
21 * PART 1 — why three slots and a refcount, and not the other two options
22 *
23 * The published value is not a word, it is a COLLECTION whose length changes
24 * every scan (0..40 entries, ~1.6 KB). "Store one atomic and be done" does
25 * not exist here, so the three real candidates are:
26 *
27 * (a) mutex + memcpy under the lock.
28 * Correct and 10 lines. But every reader holds the same lock for a
29 * 1.6 KB copy, so readers serialise against each other AND against the
30 * writer. On a gateway that publishes Wi-Fi state to several consumers
31 * (roaming logic, cloud telemetry, CLI) that is the wrong trade: the
32 * data is written 10x/second and read thousands of times a second.
33 * It also puts an unbounded-priority-inversion hazard on the hot path.
34 *
35 * (b) double/triple buffer + refcount <-- CHOSEN
36 * Writer touches only an idle slot, so it never waits on a reader for
37 * the copy itself; publishing is one release store. Readers never block
38 * each other at all — no mutual exclusion, only a shared counter.
39 * Cost: fixed memory (3 * 1.6 KB) and the retry loop, which is bounded
40 * in practice by the publish rate (one flip per ~100 ms).
41 * THREE slots, not two: with two, the writer's only candidate is the
42 * slot readers just stopped reading, so one slow reader stalls the
43 * writer. With three, the writer has the current slot plus two others
44 * and essentially always finds a free one.
45 *
46 * (c) RCU-ish pointer swap (malloc a new buffer, atomically swap the
47 * pointer, free the old one "later").
48 * Cleanest reader (one pointer load) but the hard part is "later":
49 * without a grace period you free memory a reader is still in. Real
50 * answers are epochs, hazard pointers or call_rcu — that is a kernel
51 * facility, not something to hand-roll in a userspace daemon. It also
52 * mallocs on every scan, which an embedded gateway would rather not do.
53 * The refcount in (b) IS the poor man's grace period, with the malloc
54 * replaced by a fixed slot pool.
55 *
56 * A seqlock is the fourth option and is worth mentioning: writer bumps an
57 * odd/even sequence around the copy, reader reads seq, copies, re-reads seq
58 * and retries if it changed. Wait-free for the writer, but the reader can
59 * be starved and it copies out of a buffer that is being written, which is
60 * a data race that only "works" because the reader throws the result away.
61 * Fine for an 8-byte struct, unpleasant for 1.6 KB. The slot+refcount
62 * scheme gives the reader a stable buffer instead.
63 *
64 * PART 2 — hash map + on-demand min-heap
65 *
66 * wifi_lookup O(1) average (open addressing, linear probing, FNV-1a over
67 * the 6 BSSID bytes); O(capacity) worst case if the table is
68 * saturated. No chaining, so no malloc and one cache line per
69 * probe -- the right shape for an embedded target.
70 * wifi_top_k O(N log k) time, O(1) extra space: keep a size-k MIN-heap
71 * of the best-so-far in the caller's output buffer; the root
72 * is the weakest survivor, so each of the N slots costs one
73 * compare and only sometimes a log k sift. Sorting all N is
74 * O(N log N) and needs an N-sized scratch copy (you cannot
75 * sort the table in place - it is a hash table, moving
76 * entries breaks the probe chains). With N=256 and k=8 that
77 * is ~2000 compares versus ~256*3; the gap grows with N and
78 * the heap's memory does not grow at all.
79 * eviction lazy: nothing runs on a timer. An entry older than
80 * AP_STALE_US is invisible to readers and is the preferred
81 * victim when a new BSSID needs a slot. Deleting for real
82 * would mean tombstones or back-shifting to keep probe chains
83 * intact; overwriting a dead entry in place keeps the chain
84 * and costs nothing.
85 * EWMA s += alpha * (x - s), alpha = 1/4. One float per AP, no
86 * history buffer, and it is the right statistic here: a
87 * single scan's RSSI swings 5-10 dB from multipath and body
88 * blocking, so "last value" makes the roaming logic flap
89 * between APs. Averaging over a window would need per-AP
90 * storage and a decision about window length; the EWMA
91 * forgets exponentially with one multiply-add. alpha picks
92 * the trade: 1/4 settles within ~10 scans (~1 s here) and
93 * still rejects a single bad sample. First sighting seeds
94 * with the raw value - starting from 0 dBm would make a new
95 * AP look like the strongest thing in the building.
96 */

159/* Pick a slot that is neither published nor pinned by any reader.
160 * Only the scanner thread calls this, so g_cur cannot move underneath us. */
161static snap_slot_t *slot_acquire(void)
162{
163 int cur = atomic_load_explicit(&g_cur, memory_order_relaxed);
164 
165 for (int spin = 0; spin < 2000; spin++) {
166 for (int i = 0; i < SNAP_SLOTS; i++) {
167 if (i == cur)
168 continue;
169 /* acquire: pairs with the reader's release on unpin, so the
170 * reader's loads are ordered before our stores to this slot. */
171 if (atomic_load_explicit(&g_snap[i].pin, memory_order_acquire) == 0u)
172 return &g_snap[i];
173 }
174 atomic_fetch_add_explicit(&g_stalls, 1u, memory_order_relaxed);
175 usleep(ACQUIRE_SPIN_US); /* every reader is mid-memcpy; yield */
176 }
177 return NULL; /* only a stuck reader gets here */
178}

180static void snap_publish(const struct ApInfo *ap, size_t n)
181{
182 snap_slot_t *s = slot_acquire();
183 if (!s)
184 return; /* keep the old snapshot rather than tear it */
185 
186 s->n = n;
187 if (n)
188 memcpy(s->ap, ap, n * sizeof *ap);
189 
190 /* release: everything above is visible to any reader that acquire-loads
191 * this index. This single store is the "commit" of the whole collection. */
192 atomic_store_explicit(&g_cur, (int)(s - g_snap), memory_order_release);
193}

199size_t wifi_get_snapshot(struct ApInfo *out, size_t max)
200{
201 if (!out || max == 0)
202 return 0;
203 
204 for (;;) {
205 int i = atomic_load_explicit(&g_cur, memory_order_acquire);
206 if (i < 0)
207 return 0; /* no scan has completed yet */
208 
209 atomic_fetch_add_explicit(&g_snap[i].pin, 1u, memory_order_acq_rel);
210 
211 /* Re-check AFTER pinning. If the writer flipped away from this slot
212 * in the window between the load and the increment, it may already
213 * be refilling it -- drop the pin and start over. If the index still
214 * says i, the writer cannot be inside slot i: it only ever writes a
215 * slot that is not current and not pinned. */
216 if (atomic_load_explicit(&g_cur, memory_order_acquire) != i) {
217 atomic_fetch_sub_explicit(&g_snap[i].pin, 1u, memory_order_release);
218 continue;
219 }
220 
221 size_t n = g_snap[i].n;
222 size_t c = n < max ? n : max;
223 if (c)
224 memcpy(out, g_snap[i].ap, c * sizeof *out);
225 
226 /* release: the writer's acquire-load of pin==0 will see that our
227 * reads finished before it starts overwriting the slot. */
228 atomic_fetch_sub_explicit(&g_snap[i].pin, 1u, memory_order_release);
229 return c;
230 }
231}

237/* FNV-1a over the 6 MAC bytes, then an xor-fold.
238 * Why fold: we index with `h & (CAP-1)`, i.e. the LOW bits, and FNV's low
239 * bits are its weakest -- consecutive BSSIDs (a vendor hands out MACs in
240 * runs) would collide in clumps. The fold mixes the high bits down.
241 * Why not "just use the last 4 bytes of the MAC": the last bytes of a MAC
242 * are sequential per vendor, so with CAP=256 a rack of 300 APs from one
243 * vendor lands in a handful of buckets. */
244static uint32_t bssid_hash(const uint8_t b[6])
245{
246 uint32_t h = 2166136261u; /* FNV offset basis */
247 for (int i = 0; i < 6; i++) {
248 h ^= (uint32_t)b[i];
249 h *= 16777619u; /* FNV prime */
250 }
251 h ^= h >> 16;
252 return h;
253}

255static bool entry_stale(const struct ApEntry *e, uint64_t now)
256{
257 if (now < e->last_seen_us) /* wall clock stepped backwards */
258 return false;
259 return (now - e->last_seen_us) > AP_STALE_US;
260}

262/* Find a live-or-stale entry for this BSSID. Caller holds the lock. */
263static long tab_find(const uint8_t bssid[6])
264{
265 uint32_t start = bssid_hash(bssid) & (AP_TABLE_CAP - 1u);
266 for (uint32_t p = 0; p < AP_TABLE_CAP; p++) {
267 uint32_t j = (start + p) & (AP_TABLE_CAP - 1u);
268 if (!g_tab.slot[j].used)
269 return -1; /* EMPTY terminates the probe chain */
270 if (memcmp(g_tab.slot[j].e.bssid, bssid, 6) == 0)
271 return (long)j;
272 }
273 return -1; /* table saturated, key absent */
274}

276/* Insert or update one AP. Caller holds the lock. */
277static void tab_upsert(const struct ApInfo *a, uint64_t now)
278{
279 uint32_t start = bssid_hash(a->bssid) & (AP_TABLE_CAP - 1u);
280 long victim = -1, empty = -1;
281 
282 for (uint32_t p = 0; p < AP_TABLE_CAP; p++) {
283 uint32_t j = (start + p) & (AP_TABLE_CAP - 1u);
284 ap_slot_t *s = &g_tab.slot[j];
285 
286 if (!s->used) { empty = (long)j; break; }
287 
288 if (memcmp(s->e.bssid, a->bssid, 6) == 0) {
289 /* known AP: smooth the RSSI. Three statements, not one
290 * expression, so the compiler cannot fuse the multiply and the
291 * add into an FMA -- that keeps the result bit-identical to the
292 * harness's reference. */
293 float d = (float)a->rssi_dbm - s->e.rssi_smoothed;
294 d *= AP_EWMA_ALPHA;
295 s->e.rssi_smoothed += d;
296 memcpy(s->e.ssid, a->ssid, sizeof s->e.ssid); /* SSID can change */
297 s->e.ssid[sizeof s->e.ssid - 1] = '\0';
298 if (s->e.times_seen != UINT32_MAX)
299 s->e.times_seen++;
300 s->e.last_seen_us = now;
301 return;
302 }
303 
304 /* Remember the first dead entry on the chain: overwriting it in
305 * place keeps the chain intact, unlike a real delete. */
306 if (victim < 0 && entry_stale(&s->e, now))
307 victim = (long)j;
308 }
309 
310 long j = (victim >= 0) ? victim : empty;
311 if (j < 0) {
312 /* No empty slot and nothing dead: the table is genuinely full of
313 * APs we still care about. Refuse loudly instead of evicting
314 * something live at random -- a non-zero counter tells the operator
315 * AP_TABLE_CAP is undersized. */
316 g_tab.dropped++;
317 return;
318 }
319 if (victim < 0)
320 g_tab.occupied++; /* consumed a fresh slot */
321 
322 ap_slot_t *s = &g_tab.slot[j];
323 s->used = true;
324 memset(&s->e, 0, sizeof s->e);
325 memcpy(s->e.bssid, a->bssid, 6);
326 memcpy(s->e.ssid, a->ssid, sizeof s->e.ssid);
327 s->e.ssid[sizeof s->e.ssid - 1] = '\0';
328 s->e.rssi_smoothed = (float)a->rssi_dbm; /* seed, do not smooth from 0 */
329 s->e.times_seen = 1;
330 s->e.last_seen_us = now;
331}

333void wifi_table_observe(const struct ApInfo *aps, size_t n)
334{
335 if (!aps || n == 0)
336 return;
337 uint64_t now = wifi_now_us();
338 pthread_mutex_lock(&g_tab.lock);
339 for (size_t i = 0; i < n; i++)
340 tab_upsert(&aps[i], now);
341 pthread_mutex_unlock(&g_tab.lock);
342}

344int wifi_lookup(const uint8_t bssid[6], struct ApEntry *out)
345{
346 if (!bssid || !out)
347 return -1;
348 uint64_t now = wifi_now_us();
349 int rc = -1;
350 pthread_mutex_lock(&g_tab.lock);
351 long j = tab_find(bssid);
352 if (j >= 0 && !entry_stale(&g_tab.slot[j].e, now)) {
353 *out = g_tab.slot[j].e;
354 rc = 0;
355 }
356 pthread_mutex_unlock(&g_tab.lock);
357 return rc; /* stale == unknown to the caller */
358}

364/* "weaker" is the heap's ordering. The documented tie-break (equal smoothed
365 * RSSI -> smaller BSSID wins) makes the whole ranking deterministic, which
366 * matters: roaming logic that flips between two equally strong APs every
367 * scan is a bug report. */
368static bool weaker(const struct ApEntry *a, const struct ApEntry *b)
369{
370 if (a->rssi_smoothed != b->rssi_smoothed)
371 return a->rssi_smoothed < b->rssi_smoothed;
372 return memcmp(a->bssid, b->bssid, 6) > 0;
373}

375static void heap_swap(struct ApEntry *a, struct ApEntry *b)
376{
377 struct ApEntry t = *a; *a = *b; *b = t;
378}

380static void heap_up(struct ApEntry *h, size_t i)
381{
382 while (i > 0) {
383 size_t p = (i - 1) / 2;
384 if (!weaker(&h[i], &h[p]))
385 return;
386 heap_swap(&h[i], &h[p]);
387 i = p;
388 }
389}
390 
391static void heap_down(struct ApEntry *h, size_t m, size_t i)
392{
393 for (;;) {
394 size_t l = 2 * i + 1, r = l + 1, w = i;
395 if (l < m && weaker(&h[l], &h[w])) w = l;
396 if (r < m && weaker(&h[r], &h[w])) w = r;
397 if (w == i)
398 return;
399 heap_swap(&h[i], &h[w]);
400 i = w;
401 }
402}

404size_t wifi_top_k(struct ApEntry *out, size_t k)
405{
406 if (!out || k == 0)
407 return 0;
408 
409 uint64_t now = wifi_now_us();
410 size_t m = 0;
411 
412 pthread_mutex_lock(&g_tab.lock);
413 for (uint32_t j = 0; j < AP_TABLE_CAP; j++) {
414 const ap_slot_t *s = &g_tab.slot[j];
415 if (!s->used || entry_stale(&s->e, now))
416 continue;
417 if (m < k) {
418 out[m] = s->e; /* fill the heap */
419 heap_up(out, m);
420 m++;
421 } else if (weaker(&out[0], &s->e)) {
422 out[0] = s->e; /* beats the weakest survivor */
423 heap_down(out, m, 0);
424 }
425 /* else: weaker than the whole heap -- one compare and we are done,
426 * which is exactly why this is O(N log k) and not O(N log N). */
427 }
428 pthread_mutex_unlock(&g_tab.lock);
429 
430 /* The heap is only partially ordered. Sort strongest-first outside the
431 * lock -- `out` is the caller's private buffer and k is small. */
432 for (size_t i = 1; i < m; i++) {
433 struct ApEntry v = out[i];
434 size_t j = i;
435 while (j > 0 && weaker(&out[j - 1], &v)) {
436 out[j] = out[j - 1];
437 j--;
438 }
439 out[j] = v;
440 }
441 return m;
442}

468static bool scan_once(void)
469{
470 struct ApInfo buf[WIFI_MAX_APS];
471 size_t found = 0;
472 
473 if (wifi_scan(buf, WIFI_MAX_APS, &found) != 0)
474 return false; /* BUSY: buf and found are garbage */
475 
476 /* The vendor may report more APs than fit in the buffer. Trust the
477 * buffer size, never `found`, when deciding how much to read. */
478 size_t n = found < (size_t)WIFI_MAX_APS ? found : (size_t)WIFI_MAX_APS;
479 
480 snap_publish(buf, n); /* Part 1 first: cheapest, hottest */
481 wifi_table_observe(buf, n); /* Part 2 */
482 return true;
483}

485static void *scanner_main(void *arg)
486{
487 (void)arg;
488 while (atomic_load_explicit(&g_running, memory_order_acquire)) {
489 if (!scan_once())
490 usleep(BUSY_BACKOFF_US);
491 }

495int wifi_table_init(void)
496{
497 if (g_started)
498 return 0;
499 
500 /* One scan up front so a getter called right after init already has a
501 * complete snapshot. Costs one scan time; retry a couple of times in
502 * case the radio came back BUSY. */
503 for (int tries = 0; tries < 3; tries++)
504 if (scan_once())
505 break;
506 
507 atomic_store_explicit(&g_running, true, memory_order_release);
508 if (pthread_create(&g_thread, NULL, scanner_main, NULL) != 0) {
509 atomic_store_explicit(&g_running, false, memory_order_release);
510 return -1;
511 }
512 g_started = true;
513 return 0;
514}

516void wifi_table_deinit(void)
517{
518 if (!g_started)
519 return;
520 atomic_store_explicit(&g_running, false, memory_order_release);
521 /* The thread may be parked inside wifi_scan() for up to one scan
522 * (300 ms on real hardware). There is no way to cancel a vendor
523 * blocking call safely, so we wait for it. */
524 pthread_join(g_thread, NULL);
525 g_started = false;
526 /* The snapshot and the table stay valid and readable after deinit. */
