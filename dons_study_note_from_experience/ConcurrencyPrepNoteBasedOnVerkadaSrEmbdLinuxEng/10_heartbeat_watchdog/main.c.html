1/* main.c — test harness + fake hardware watchdog (GIVEN; do not modify).
2 *
3 * It is used unchanged by both watchdog.c (your stub) and watchdog_solution.c.
4 *
5 * How it judges the implementation
6 * - the fake hwwdt records EVERY hwwdt_kick() and hwwdt_report_fault() with a
7 * timestamp, so the tests can ask "was the dog kicked between t0 and t1?"
8 * and "how many times was worker 3 reported?" instead of trusting a counter
9 * inside the implementation.
10 * - ground truth for Part 2 is recomputed by brute force from the harness's
11 * own record of when it last called wd_heartbeat() for each worker.
12 * Nothing is hard-coded, so macOS and Linux give the same verdict.
13 * - randomness is a fixed-seed xorshift (never rand(): its sequence differs
14 * per platform).
15 *
16 * Timing: the real hardware watchdog window is 2 s. hwwdt.h scales it 20x down
17 * to 100 ms and every deadline in this file is scaled to match, so the whole run
18 * takes about 4 s instead of 80.
19 */

65void hwwdt_kick(void)
66{
67 uint64_t t = hwwdt_now_us();
68 pthread_mutex_lock(&g_log_lock);
69 if (g_kick_n < LOG_MAX) g_kick_ts[g_kick_n++] = t;
70 else g_kick_lost++;
71 pthread_mutex_unlock(&g_log_lock);
72}

74void hwwdt_report_fault(uint32_t worker_id)
75{
76 uint64_t t = hwwdt_now_us();

78 if (g_flt_n < LOG_MAX) { g_flt[g_flt_n].ts = t; g_flt[g_flt_n].id = worker_id; g_flt_n++; }
79 else g_flt_lost++;
80 pthread_mutex_unlock(&g_log_lock);

85static int kicks_between(uint64_t a, uint64_t b)
86{
87 int n = 0;
88 pthread_mutex_lock(&g_log_lock);
89 for (int i = 0; i < g_kick_n; i++)
90 if (g_kick_ts[i] >= a && g_kick_ts[i] <= b) n++;
91 pthread_mutex_unlock(&g_log_lock);
92 return n;
93}

95/* Worst interval in [a,b] during which the dog was not kicked — including the
96 * stretch before the first kick and after the last one. */
97static uint64_t max_kick_gap(uint64_t a, uint64_t b)
98{
99 uint64_t prev = a, worst = 0;
100 pthread_mutex_lock(&g_log_lock);
101 for (int i = 0; i < g_kick_n; i++) {
102 uint64_t t = g_kick_ts[i];
103 if (t < a || t > b) continue;
104 if (t - prev > worst) worst = t - prev;
105 prev = t;
106 }
107 pthread_mutex_unlock(&g_log_lock);
108 if (b > prev && b - prev > worst) worst = b - prev;
109 return worst;
110}

112static int faults_for(int id, uint64_t a, uint64_t b)
113{
114 int n = 0;
115 pthread_mutex_lock(&g_log_lock);
116 for (int i = 0; i < g_flt_n; i++)
117 if (g_flt[i].ts >= a && g_flt[i].ts <= b &&
118 (id < 0 || g_flt[i].id == (uint32_t)id)) n++;
119 pthread_mutex_unlock(&g_log_lock);
120 return n;
121}

157/* Harness copy of "when did we last tell the watchdog this worker is alive".
158 * The timestamp is taken BEFORE calling wd_heartbeat(), so it is never newer
159 * than what the implementation stored -> the brute-force minimum is never later
160 * than the true one, and the only error is the few ns between the two clock
161 * reads (covered by ND_TOL_US). */
162static _Atomic uint64_t g_hb_seen[WD_MAX_WORKERS];

164static void hb(int id)
165{
166 uint64_t t = hwwdt_now_us();
167 wd_heartbeat((uint32_t)id);
168 /* guard: with the unfinished stub wd_register() returns -1, and the tests
169 * below still call hb(-1). Writing g_hb_seen[-1] would corrupt the harness. */
170 if (id >= 0 && id < WD_MAX_WORKERS)
171 atomic_store_explicit(&g_hb_seen[id], t, memory_order_relaxed);
172}

179static int reg_add(const char *name, uint64_t deadline_us)
180{

201/* ground truth for Part 2 */
202static uint64_t brute_min(void)
203{
204 uint64_t m = UINT64_MAX;
205 for (int i = 0; i < g_reg_n; i++) {
206 uint64_t d = atomic_load_explicit(&g_hb_seen[g_reg[i].id], memory_order_relaxed)
207 + g_reg[i].deadline;
208 if (d < m) m = d;
209 }
210 return m;
211}
212static int brute_min_holder(void)
213{
214 uint64_t m = UINT64_MAX;
215 int who = -1;
216 for (int i = 0; i < g_reg_n; i++) {
217 uint64_t d = atomic_load_explicit(&g_hb_seen[g_reg[i].id], memory_order_relaxed)
218 + g_reg[i].deadline;
219 if (d < m) { m = d; who = g_reg[i].id; }
220 }
221 return who;
222}

258/* per-call nanoseconds; thread create/join overhead is identical for both
259 * variants and is ~0.1% of the run, so the ratio is meaningful */
260static double bench_run(void *(*fn)(void *), const int *ids)
261{
262 pthread_t th[BENCH_THREADS];
263 uint64_t t0 = hwwdt_now_us();
264 for (int i = 0; i < BENCH_THREADS; i++)
265 pthread_create(&th[i], NULL, fn, (void *)&ids[i]);
266 for (int i = 0; i < BENCH_THREADS; i++) pthread_join(th[i], NULL);
267 uint64_t t1 = hwwdt_now_us();
268 return (double)(t1 - t0) * 1000.0 / ((double)BENCH_THREADS * BENCH_ITERS);
269}

284{

286 while (atomic_load_explicit(&g_wk_run, memory_order_relaxed)) {
287 /* g_wk_quiet freezes the beats for a few ms so that the brute-force
288 * minimum stands still while we compare it with wd_next_deadline(). */
289 if (!atomic_load_explicit(&g_wk_quiet, memory_order_relaxed)) {
290 hb(w->id);

292 }

294 }
