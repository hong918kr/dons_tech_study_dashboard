2 *
3 * ./main.sh build main.c + frame_store.c (your code)
4 * ./main.sh sol build main.c + frame_store_solution.c (reference)
5 *
6 * GIVEN — do not edit. The same file builds against the stub and the solution.
7 *
8 * What is faked here (bottom of the file):
9 * capture_next_frame() blocks ~25 ms, then fills the frame with a pattern
10 * derived from the frame's sequence number, IN TWO HALVES with a 0.5 ms gap
11 * the way a DMA engine would. So a reader that is handed a buffer the sensor
12 * is still writing sees a frame whose body does not match its header, and
13 * frame_check() catches it. That is the tearing test.
14 *
15 * Ground truth: every frame the fake sensor produced (seq, ts) is recorded
16 * here, and every PASS/FAIL is computed against that — never against
17 * hard-coded numbers (timings differ per machine).
18 *
19 * Randomness is a fixed-seed xorshift, never rand(): rand() differs between
20 * glibc and Apple libc.
21 */

69/* true if the frame is internally consistent: body matches the seq in its
70 * header. *seq_out gets that seq. A torn frame fails this. */
71static bool frame_check(const uint8_t *f, uint64_t *seq_out)
72{
73 uint64_t seq;
74 memcpy(&seq, f, SEQ_BYTES);
75 if (seq == 0)
76 return false; /* never-written buffer */
77 for (size_t i = SEQ_BYTES; i < FRAME_BYTES; i++)
78 if (f[i] != frame_byte(seq, i))
79 return false;
80 if (seq_out)
81 *seq_out = seq;
82 return true;
83}

148/* The newest frame strictly older than `cut`, plus the timestamp of the first
149 * frame at or after `cut`. Used to build a window-edge query whose answer is a
150 * frame that is itself older than the window. */
151static bool truth_edge(uint64_t cut, uint64_t *fseq, uint64_t *fts, uint64_t *gts)
152{
153 bool ok = false;
154 pthread_mutex_lock(&g_truth_lock);
155 for (size_t i = 0; i < g_truth_n; i++) {
156 if (g_truth[i].ts < cut) {
157 if (fseq) *fseq = g_truth[i].seq;
158 if (fts) *fts = g_truth[i].ts;
159 ok = true;
160 } else {
161 if (gts) *gts = g_truth[i].ts;
162 break;
163 }
164 }
165 pthread_mutex_unlock(&g_truth_lock);
166 return ok;
167}

213static void *stress_reader(void *arg)
214{
215 reader_stat_t *st = arg;
216 uint32_t rng = 0x1234567u + (uint32_t)st->id * 0x9E3779B9u;
217 uint8_t copy[FRAME_BYTES];
218 
219 while (!atomic_load(&g_stress_stop)) {
220 rng ^= rng << 13; rng ^= rng >> 17; rng ^= rng << 5;
221 
222 /* ---- Part 1: borrow the latest frame ---- */
223 uint64_t ts = 0, seq = 0, t0 = capture_now_us();
224 const uint8_t *f = frame_acquire_latest(&ts);
225 uint64_t t1 = capture_now_us();
226 if (t1 - t0 > st->max_acquire_us)
227 st->max_acquire_us = t1 - t0;
228 
229 if (!f) {
230 st->nulls++;
231 } else {
232 st->acquires++;
233 if (!frame_check(f, &seq)) {
234 st->torn++; /* saw a half-written frame */
235 } else {
236 uint64_t real_ts = 0;
237 if (!truth_ts_of_seq(seq, &real_ts) || real_ts != ts)
238 st->bad_ts++; /* invented frame or wrong ts */
239 }
240 /* hold it a while: the writer must keep going without us */
241 if (st->id > 0)
242 usleep(200u * (unsigned)st->id);
243 /* re-check after the hold: the borrow must still be stable */
244 uint64_t seq2 = 0;
245 if (!frame_check(f, &seq2) || seq2 != seq)
246 st->torn++;
247 frame_release(f);
248 }
249 
250 /* ---- Part 2: replay a point 200..800 ms in the past ---- */
251 uint64_t now = capture_now_us();
252 uint64_t qt = now - (200000u + rng % 600000u);
253 uint64_t ats = 0;
254 t0 = capture_now_us();
255 int r = frame_at(qt, copy, &ats);
256 t1 = capture_now_us();
257 if (t1 - t0 > st->max_at_us)
258 st->max_at_us = t1 - t0;
259 if (r == 0) {
260 uint64_t rseq = 0, real_ts = 0;
261 if (!frame_check(copy, &rseq))
262 st->replay_torn++;
263 else if (ats > qt || !truth_ts_of_seq(rseq, &real_ts) || real_ts != ats)
264 st->replay_bad++;
265 else
266 st->replay_ok++;
267 }
268 }
269 return NULL;
270}
