1/* temp_cache_solution.c — reference solution (Part 1 + Part 2).
2 *
3 * ./main.sh sol
4 *
5 * Why this shape
6 * Part 1 is NOT "one sampler thread forever". The probe is read on demand,
7 * so the thing to get right is DEMAND COALESCING ("single flight"):
8 * mutex + condvar + an `inflight` flag + a generation counter. One caller
9 * becomes the leader, drops the mutex, does the slow vendor read, then
10 * publishes and broadcasts; everyone who arrived meanwhile took the
11 * waiter path and consumes the leader's result. The mutex is NEVER held
12 * across tempdev_read() — holding it would serialise 8 callers into 8
13 * sequential 60 ms reads, which is both slow and the classic mistake.
14 * An atomic/seqlock cannot replace the condvar here because waiters must
15 * SLEEP until a specific event (this refresh finished), not just read a
16 * value; the generation counter is what makes "finished" unambiguous and
17 * immune to spurious wakeups.
18 *
19 * Part 2 is a fixed grid of 60 one-minute buckets, each carrying
20 * min/max/sum/count, plus a ring of the raw samples so the two PARTIAL
21 * buckets at the ends of a range are exact. A query folds whole buckets in
22 * O(1) each (<= 60 of them) and scans raw samples only inside the two edge
23 * minutes, so cost is bounded by the WINDOW, never by the sample count.
24 *
25 * Failure policy (documented, because the harness checks it)
26 * A waiter consumes the result of the refresh it waited on and returns.
27 * If that refresh NACKed it gets TEMP_STALE with the last good value (or
28 * TEMP_NODATA if there never was one). It does NOT promote itself to
29 * leader and start another read — otherwise 8 waiters on a wedged bus
30 * would turn into 8 more vendor reads, i.e. a retry storm. Retrying is
31 * the caller's decision, one read per call.
32 */

47static struct {
48 pthread_mutex_t lock;
49 pthread_cond_t cv;
50 
51 float value; /* last good reading */
52 uint64_t stamp; /* when it was taken (us, tempdev_now_us clock) */
53 bool have; /* a reading ever succeeded */
54 
55 bool inflight; /* somebody is inside tempdev_read() right now */
56 uint64_t gen; /* ++ every time a refresh COMPLETES (ok or not) */
57 bool last_ok; /* did the most recently completed refresh work? */
58 
59 bool up; /* between init and deinit */
60 
61 /* counters, all mutated under g.lock so there is nothing to race on */
62 uint64_t st_vendor_calls, st_hits, st_leaders, st_coalesced, st_nacks;
63} g = {
64 .lock = PTHREAD_MUTEX_INITIALIZER,
65 .cv = PTHREAD_COND_INITIALIZER,
66};

72/* Raw sample ring. Sized for the worst rate we promise to keep exactly at
73 * the range edges: 16384 samples / 3600 s ~= 4.5 Hz, 16 B each = 256 KiB.
74 * Beyond that rate the oldest raw samples get overwritten before they age
75 * out; the per-minute AGGREGATES are never lost, so min/max/avg of whole
76 * minutes stay exact and only the two partial edge minutes degrade. Those
77 * queries are counted in stats.raw_lost instead of being silently wrong. */
78#define RAW_CAP 16384u
79_Static_assert((RAW_CAP & (RAW_CAP - 1u)) == 0u, "RAW_CAP must be a power of 2");
80#define RAW_MASK (RAW_CAP - 1u)

114int temp_cache_init(void)
115{
116 pthread_mutex_lock(&g.lock);
117 if (g.up) {
118 pthread_mutex_unlock(&g.lock);
119 return 0;
120 }
121 g.value = NAN;
122 g.stamp = 0;
123 g.have = false;
124 g.inflight = false;
125 g.gen = 0;
126 g.last_ok = false;
127 g.up = true;
128 g.st_vendor_calls = g.st_hits = g.st_leaders = g.st_coalesced = g.st_nacks = 0;
129 pthread_mutex_unlock(&g.lock);
130 
131 /* Deliberately no device access here: the whole point of this API is
132 * that the bus is touched only when somebody actually wants a value. */
133 
134 pthread_mutex_lock(&h.lock);
135 memset(h.b, 0, sizeof h.b);
136 for (unsigned i = 0; i < TEMP_NBUCKETS; i++)
137 h.b[i].minute = UINT64_MAX; /* UINT64_MAX == "never used" */
138 h.head = 0;
139 h.newest_minute = 0;
140 h.last_ts = 0;
141 h.any = false;
142 h.range_queries = h.last_scanned = h.raw_lost = 0;
143 pthread_mutex_unlock(&h.lock);
144 
145 return 0;
146}

148void temp_cache_deinit(void)
149{
150 pthread_mutex_lock(&g.lock);
151 if (!g.up) {
152 pthread_mutex_unlock(&g.lock);
153 return;
154 }
155 g.up = false;
156 pthread_cond_broadcast(&g.cv); /* release parked waiters */
157 
158 /* A leader is somewhere inside the vendor read; it still owns the bus.
159 * Wait for it, or we would return while the device is mid-transaction. */
160 while (g.inflight)
161 pthread_cond_wait(&g.cv, &g.lock);
162 
163 g.have = false;
164 g.value = NAN;
165 pthread_mutex_unlock(&g.lock);
166}

172/* Called with g.lock held. "fresher than max_age_us" is strict, so
173 * max_age_us == 0 always means "refresh". */
174static bool fresh_enough(uint64_t max_age_us)
175{
176 if (!g.have)
177 return false;
178 uint64_t now = tempdev_now_us();
179 if (now < g.stamp) /* clock stepped back: treat as fresh, the */
180 return true; /* sample cannot be older than "now" */
181 return (now - g.stamp) < max_age_us;
182}

184int temp_get(float *out, uint64_t max_age_us)
185{
186 if (out == NULL)
187 return TEMP_NODATA;
188 
189 pthread_mutex_lock(&g.lock);
190 
191 if (!g.up) {
192 pthread_mutex_unlock(&g.lock);
193 return TEMP_NODATA;
194 }
195 
196 /* (1) cache hit — the common case, and it must not touch the bus. */
197 if (fresh_enough(max_age_us)) {
198 *out = g.value;
199 g.st_hits++;
200 pthread_mutex_unlock(&g.lock);
201 return TEMP_OK;
202 }
203 
204 /* (2) a refresh is already running — ride along instead of starting a
205 * second one. Remember WHICH refresh we are waiting for: the
206 * generation counter turns "it finished" into a plain comparison,
207 * so spurious wakeups and a second refresh racing past us are both
208 * harmless. */
209 if (g.inflight) {
210 uint64_t seen = g.gen;
211 g.st_coalesced++;
212 while (g.gen == seen && g.up)
213 pthread_cond_wait(&g.cv, &g.lock);
214 
215 int rc;
216 if (g.have && g.last_ok) { /* the read we waited on worked */
217 *out = g.value;
218 rc = TEMP_OK;
219 } else if (g.have) { /* it NACKed: hand back stale */
220 *out = g.value;
221 rc = TEMP_STALE;
222 } else {
223 rc = TEMP_NODATA; /* never had a good reading */
224 }
225 pthread_mutex_unlock(&g.lock);
226 return rc;
227 }
228 
229 /* (3) nobody is reading: I am the leader. Claim the slot, then DROP THE
230 * MUTEX before the slow call. Holding it here is the classic bug —
231 * it would serialise every caller behind a 60 ms bus transaction and
232 * also block the cheap cache-hit path. */
233 g.inflight = true;
234 g.st_leaders++;
235 g.st_vendor_calls++;
236 pthread_mutex_unlock(&g.lock);
237 
238 float v = 0.0f;
239 int rc = tempdev_read(&v);
240 uint64_t t = tempdev_now_us();
241 
242 pthread_mutex_lock(&g.lock);
243 if (rc == 0) {
244 g.value = v;
245 g.stamp = t;
246 g.have = true;
247 g.last_ok = true;
248 } else {
249 g.last_ok = false;
250 g.st_nacks++;
251 }
252 g.inflight = false;
253 g.gen++; /* publish BEFORE broadcasting */
254 pthread_cond_broadcast(&g.cv);
255 
256 int ret;
257 if (rc == 0) {
258 *out = g.value;
259 ret = TEMP_OK;
260 } else if (g.have) {
261 *out = g.value;
262 ret = TEMP_STALE;
263 } else {
264 ret = TEMP_NODATA;
265 }
266 pthread_mutex_unlock(&g.lock);
267 
268 /* History is a different lock; take it AFTER releasing g.lock so the two
269 * are never nested and there is no lock-order to get wrong. */
270 if (rc == 0)
271 temp_history_add(t, v);
272 
273 return ret;
274}

280int temp_history_add(uint64_t ts, float c)
281{
282 if (isnan(c))
283 return -1;
284 
285 pthread_mutex_lock(&h.lock);
286 
287 /* The clock is gettimeofday: NTP can step it backwards. Clamp so the
288 * ring stays sorted — a bucket scan and the minute index both rely on
289 * non-decreasing timestamps. */
290 if (h.any && ts < h.last_ts)
291 ts = h.last_ts;
292 
293 uint64_t m = ts / TEMP_BUCKET_US;
294 bucket_t *b = &h.b[m % TEMP_NBUCKETS];
295 
296 if (b->minute != m) { /* slot belongs to an older */
297 b->minute = m; /* hour (or was never used): */
298 b->first = h.head; /* recycle it. That IS the */
299 b->count = 0; /* eviction — no pass needed. */
300 b->sum = 0.0;
301 b->min = b->max = c;
302 }
303 
304 h.raw[h.head & RAW_MASK] = (sample_t){ .ts = ts, .c = c };
305 h.head++;
306 
307 if (b->count == 0) {
308 b->min = b->max = c;
309 } else {
310 if (c < b->min) b->min = c;
311 if (c > b->max) b->max = c;
312 }
313 b->count++;
314 b->sum += (double)c;
315 
316 if (!h.any || m > h.newest_minute)
317 h.newest_minute = m;
318 h.last_ts = ts;
319 h.any = true;
320 
321 pthread_mutex_unlock(&h.lock);
322 return 0;
323}

325int temp_range_stats(uint64_t t0, uint64_t t1, float *out_min, float *out_max,
326 float *out_avg)
327{
328 if (t0 > t1)
329 return -1;
330 
331 pthread_mutex_lock(&h.lock);
332 if (!h.any) {
333 pthread_mutex_unlock(&h.lock);
334 return -1;
335 }
336 
337 /* Clamp to the window FIRST. Without this, t0 = 0 would make the loop
338 * below walk ~28 million minute indices. */
339 uint64_t newest = h.newest_minute;
340 uint64_t oldest = (newest >= TEMP_NBUCKETS - 1u) ? newest - (TEMP_NBUCKETS - 1u) : 0;
341 uint64_t m0 = t0 / TEMP_BUCKET_US;
342 uint64_t m1 = t1 / TEMP_BUCKET_US;
343 if (m1 < oldest || m0 > newest) { /* entirely outside history */
344 h.range_queries++;
345 h.last_scanned = 0;
346 pthread_mutex_unlock(&h.lock);
347 return -1;
348 }
349 if (m0 < oldest) m0 = oldest;
350 if (m1 > newest) m1 = newest;
351 
352 uint64_t avail_from = (h.head > RAW_CAP) ? h.head - RAW_CAP : 0;
353 uint64_t n = 0, scanned = 0;
354 bool lost = false;
355 double sum = 0.0;
356 float lo = 0.0f, hi = 0.0f;
357 
358 for (uint64_t m = m0; m <= m1; m++) { /* at most TEMP_NBUCKETS steps */
359 bucket_t *b = &h.b[m % TEMP_NBUCKETS];
360 if (b->minute != m || b->count == 0)
361 continue;
362 
363 uint64_t bs = m * TEMP_BUCKET_US;
364 uint64_t be = bs + TEMP_BUCKET_US - 1u;
365 
366 if (t0 <= bs && be <= t1) {
367 /* whole minute inside the range: O(1), this is the whole point */
368 if (n == 0) { lo = b->min; hi = b->max; }
369 else {
370 if (b->min < lo) lo = b->min;
371 if (b->max > hi) hi = b->max;
372 }
373 n += b->count;
374 sum += b->sum;
375 continue;
376 }
377 
378 /* partial minute (only ever the first and/or last of the range):
379 * walk this bucket's raw samples, bounded by one minute of data. */
380 uint64_t from = b->first;
381 uint64_t to = b->first + b->count;
382 if (from < avail_from) { /* overwritten: degraded, said so */
383 from = avail_from;
384 lost = true;
385 }
386 for (uint64_t i = from; i < to; i++) {
387 const sample_t *s = &h.raw[i & RAW_MASK];
388 scanned++;
389 if (s->ts < t0 || s->ts > t1)
390 continue;
391 if (n == 0) { lo = s->c; hi = s->c; }
392 else {
393 if (s->c < lo) lo = s->c;
394 if (s->c > hi) hi = s->c;
395 }
396 n++;
397 sum += (double)s->c;
398 }
399 }
400 
401 h.range_queries++;
402 h.last_scanned = scanned;
403 if (lost) h.raw_lost++;
404 pthread_mutex_unlock(&h.lock);
405 
406 if (n == 0)
407 return -1;
408 if (out_min) *out_min = lo;
409 if (out_max) *out_max = hi;
410 if (out_avg) *out_avg = (float)(sum / (double)n);
411 return 0;
412}

414/* Why buckets, and when the alternatives win
415 *
416 * (a) scan all samples: one array, O(1) append, zero bookkeeping, and the
417 * query is O(n) — at 4.5 Hz that is 16k floats per query while a mutex
418 * is held. Right answer when queries are RARE compared to appends, or
419 * when the window is small. Wrong here: a dashboard polls min/max far
420 * more often than the probe produces samples, and the lock hold time
421 * grows with history length.
422 *
423 * (b) sparse table / segment tree over the samples: O(1) (sparse) or
424 * O(log n) (segment) range min/max regardless of range width. But a
425 * sparse table is built on an IMMUTABLE array — it must be rebuilt
426 * O(n log n) after every append, so it fits offline analysis, not a
427 * live ring. A segment tree does support O(log n) append, but it costs
428 * 2n nodes, cannot carry a mean without extra sums, and eviction from
429 * the front means either rebuilding or a tombstoned tree. Worth it when
430 * the window is huge and queries need arbitrary sub-ranges exactly.
431 *
432 * (c) fixed buckets (this file): O(1) append, O(window/bucket + samples in
433 * two edge buckets) query, fixed memory, and eviction is free because a
434 * slot is recycled the moment its absolute minute changes. The cost is
435 * resolution: exactness at the edges only survives while the raw ring
436 * still holds those minutes. This is what real telemetry does (RRDtool,
437 * Prometheus recording rules, Graphite whisper) for exactly this shape:
438 * high-rate appends, bounded window, coarse range queries.
439 */

445void temp_cache_stats(temp_stats_t *out)
446{
447 if (!out)
448 return;
449 memset(out, 0, sizeof *out);
450 
451 pthread_mutex_lock(&g.lock);
452 out->vendor_calls = g.st_vendor_calls;
453 out->hits = g.st_hits;
454 out->leaders = g.st_leaders;
455 out->coalesced = g.st_coalesced;
456 out->nacks = g.st_nacks;
457 pthread_mutex_unlock(&g.lock);
458 
459 /* Two locks, taken one after the other, never nested. */
460 pthread_mutex_lock(&h.lock);
461 out->range_queries = h.range_queries;
462 out->last_scanned = h.last_scanned;
463 out->raw_lost = h.raw_lost;
464 pthread_mutex_unlock(&h.lock);
