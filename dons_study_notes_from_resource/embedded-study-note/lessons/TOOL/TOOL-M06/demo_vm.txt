===== 4. RPATH vs RUNPATH vs LD_LIBRARY_PATH (search order) =====

$ gcc -fPIC -shared -Wl,-soname,libwho.so -x c - -o order/b/libwho.so <<< "const char *who(void){return \"b (LD_LIBRARY_PATH dir)\";}"
$ gcc who.c -Lorder/a -lwho -Wl,-rpath,$PWD/order/a -Wl,--enable-new-dtags -o order/app_runpath

$ patchelf --set-soname libfoo.so.1 pe/lib/libfoo.so.1.2.0 && ln -s libfoo.so.1.2.0 pe/lib/libfoo.so.1

$ readelf -dW pe/bin/app | grep -E "NEEDED.*foo|RUNPATH"
