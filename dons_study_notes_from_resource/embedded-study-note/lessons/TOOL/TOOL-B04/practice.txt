# TOOL-B04 practice — same anatomy.c, different flags. One row per prediction:
#   sym <build> <symbol> <nm letter> <section>      (section: nm -m (...) for Mach-O, objdump -t for ELF)
#   rel <build> <ref>    <reloc types, comma-separated or ->
# builds
#   macho_nocommon : cc -O2 -fno-common -c
#   elf_common     : clang --target=aarch64-linux-gnu -O2 -fcommon -c
#   elf_O0         : clang --target=aarch64-linux-gnu -O0 -c
#   elf_fsect      : clang --target=aarch64-linux-gnu -O2 -ffunction-sections -fdata-sections -c
#   elf_nopic      : clang --target=aarch64-linux-gnu -O2 -fno-pic -c

# 연습 1 — common 을 뒤집기
sym macho_nocommon  g_tent     S  __DATA,__common
rel macho_nocommon  ref:g_tent ARM64_RELOC_PAGE21,ARM64_RELOC_PAGEOFF12
sym elf_common      g_tent     C  *COM*
rel elf_common      ref:g_tent R_AARCH64_ADR_PREL_PG_HI21,R_AARCH64_LDST32_ABS_LO12_NC

# 연습 2 — -O0 에서 다시 나타나는 것 / 여전히 없는 것
sym elf_O0  s_file         d  .data
sym elf_O0  s_file_zero    b  .bss
sym elf_O0  unused_static  -  -
rel elf_O0  ref:g_const    -

# 연습 3 — 함수/데이터마다 섹션 하나씩
sym elf_fsect  api_entry             T  .text.api_entry
sym elf_fsect  board_init            W  .text.board_init
sym elf_fsect  g_init                D  .data.g_init
sym elf_fsect  g_zero                B  .bss.g_zero
sym elf_fsect  g_tent                B  .bss.g_tent
sym elf_fsect  g_const               R  .rodata.g_const
sym elf_fsect  counter_next.s_local  b  .bss.counter_next.s_local

# 연습 4 — PIC 를 끄면 GOT 가 사라진다
rel elf_nopic  ref:ext_counter  R_AARCH64_ADR_PREL_PG_HI21,R_AARCH64_LDST32_ABS_LO12_NC
