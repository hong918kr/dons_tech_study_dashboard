# ARCH-A04 교과서 트레이스 (solution.c 의 TEXTBOOK 과 같다)
# 실행: ./so trace_textbook.txt mesi      (msi / moesi 로 바꿔 비교)
c0 R 0        # c0 혼자 읽기 → E
sync
c1 R 0        # 둘째 읽기 → 둘 다 S
sync
c2 W 0 7      # 쓰기 미스 = RFO(BusRdX) → 나머지 무효화
sync
c0 R 0        # M 소유자가 Flush (cache-to-cache)
sync
c0 W 0 8      # S 에서 쓰기 → BusUpgr
sync
c3 R 1        # 같은 라인의 다른 워드
sync
c3 W 1 5
sync
c3 R 2        # 같은 라인 → 히트
sync
c1 R 8        # 다른 라인: 혼자 읽고
sync
c1 W 8 9      # 혼자 쓴다 (MESI: 버스 없음, MSI: BusUpgr)
sync
c2 R 1
sync
c1 R 2        # MOESI: O 가 공급, MESI: 메모리가 공급
